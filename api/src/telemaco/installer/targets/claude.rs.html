6/// Claude Code's own config directory: `~/.claude` unless `$CLAUDE_CONFIG_DIR`
7/// says otherwise.
8///
9/// "To keep the home-directory files somewhere else, set
10/// `CLAUDE_CONFIG_DIR`; Claude Code then stores your settings, session
11/// history, and plugins there instead" (docs.claude.com/en/docs/claude-code/settings).
12/// With it set, everything written under `~/.claude` is read by nothing - the
13/// same trap `CODEX_HOME` and `HERMES_HOME` were.
14fn claude_config_dir(home: &PathBuf) -> PathBuf {
15    home_env_var("CLAUDE_CONFIG_DIR").unwrap_or_else(|| home.join(".claude"))
16}
17
18/// `~/.claude.json`, Claude Code's own state file: "Claude Code also keeps a
19/// fifth file, `~/.claude.json`, that it writes for itself; [...] It holds
20/// your sign-in session, MCP server configurations, per-project state such as
21/// trust decisions, and the global config keys" (same page).
22///
23/// Unlike `settings.json`, this one is *not* nested one level inside the
24/// config directory by default - it sits next to `.claude/` at `$HOME`. But
25/// once `CLAUDE_CONFIG_DIR` relocates that directory, it moves inside it: on a
26/// machine with `CLAUDE_CONFIG_DIR` set, this file lives at
27/// `$CLAUDE_CONFIG_DIR/.claude.json`, confirmed against a real Claude Code
28/// session's own live trust state, not only the docs' wording.
29fn claude_json_path(home: &PathBuf) -> PathBuf {
30    match home_env_var("CLAUDE_CONFIG_DIR") {
31        Some(dir) => dir.join(".claude.json"),
32        None => home.join(".claude.json"),
33    }
34}

36pub fn detect(loc: &Location, home: Option<&PathBuf>) -> DetectionResult {
37    let (mcp_path, config_dir, claude_md) = match loc {
38        Location::Global => {
39            let h = home.cloned().unwrap_or_else(|| PathBuf::from("."));
40            let config_dir = claude_config_dir(&h);
41            (claude_json_path(&h), config_dir.clone(), config_dir.join("CLAUDE.md"))
42        }
43        Location::Folder(folder) => {
44            let mcp = if folder.join(".mcp.json").exists() {
45                folder.join(".mcp.json")
46            } else if folder.join(".claude.json").exists() {
47                folder.join(".claude.json")
48            } else {
49                folder.join(".mcp.json")
50            };
51            (mcp, folder.join(".claude"), project_instructions_path(folder))
52        }
53    };
54    let installed = config_dir.exists() || mcp_path.exists() || claude_md.exists();
55    let mut already_configured = false;
56    if mcp_path.exists() {
57        let json = read_json_file(&mcp_path);
58        if let Some(servers) = json.get("mcpServers").and_then(|v| v.as_object()) {
59            already_configured = servers.contains_key("telemaco");
60        }
61    }
62    let hint = if already_configured {
63        "already configured".to_string()
64    } else if installed {
65        if loc.is_global() {
66            "detected".to_string()
67        } else {
68            let mut markers = Vec::new();
69            if config_dir.exists() { markers.push(".claude/"); }
70            if mcp_path.exists() { markers.push(".mcp.json"); }
71            if claude_md.exists() { markers.push("CLAUDE.md"); }
72            if markers.is_empty() { "detected".to_string() } else { markers.join(", ") }
73        }
74    } else {
75        String::new()
76    };
77    DetectionResult {
78        installed,
79        already_configured,
80        config_path: Some(mcp_path),
81        hint,
82    }
83}

85/// Where a project keeps its instructions.
86///
87/// A project CLAUDE.md lives at `./CLAUDE.md` or `./.claude/CLAUDE.md`, and
88/// every discovered file is concatenated rather than one overriding another
89/// (docs.claude.com/en/docs/claude-code/memory). So both work, and the one the
90/// project already uses wins: a repository that keeps its instructions in
91/// `.claude/` has no use for a second file appearing at its root.
92fn project_instructions_path(folder: &Path) -> PathBuf {
93    let nested = folder.join(".claude").join("CLAUDE.md");
94    if nested.exists() {
95        nested
96    } else {
97        folder.join("CLAUDE.md")
98    }
99}

103    vec![
104        folder.join("CLAUDE.md"),
105        folder.join(".claude").join("CLAUDE.md"),

109/// The allow rule that auto-approves Telemaco's MCP tools.
110///
111/// A glob is only accepted after a literal `mcp__<server>__` prefix, which this
112/// has (docs.claude.com/en/docs/claude-code/permissions).
113const MCP_PERMISSION: &str = "mcp__telemaco__*";
114
115/// Drops the top-level `autoApprove` array earlier versions wrote.
116///
117/// Claude Code has no such setting: it reads `permissions.allow`, so the key
118/// sat in settings.json auto-approving nothing. Only our own entry is taken
119/// out, and the key goes with it once it is empty.
120fn remove_legacy_auto_approve(settings_json: &mut serde_json::Value) -> bool {
121    let Some(obj) = settings_json.as_object_mut() else {
122        return false;
123    };
124    let Some(arr) = obj.get_mut("autoApprove").and_then(|v| v.as_array_mut()) else {
125        return false;
126    };
127    let before = arr.len();
128    arr.retain(|item| item.as_str() != Some(MCP_PERMISSION));
129    let changed = arr.len() != before;
130    if arr.is_empty() {
131        obj.remove("autoApprove");
132    }
133    changed
134}

136/// Adds or removes auto-approval for Telemaco's MCP tools.
137///
138/// The removal half matters: `--no-permissions` on an install that already
139/// granted it has to take the grant back, not merely stop re-adding it.
140fn sync_permission_allow(settings_json: &mut serde_json::Value, enabled: bool) -> bool {
141    let mut changed = remove_legacy_auto_approve(settings_json);
142
143    if !enabled {
144        if let Some(arr) = settings_json
145            .get_mut("permissions")
146            .and_then(|p| p.get_mut("allow"))
147            .and_then(|v| v.as_array_mut())
148        {
149            let before = arr.len();
150            arr.retain(|item| item.as_str() != Some(MCP_PERMISSION));
151            changed |= arr.len() != before;
152        }
153        return changed;
154    }
155
156    let permissions = settings_json
157        .as_object_mut()
158        .expect("load_json guarantees an object")
159        .entry("permissions")
160        .or_insert_with(|| json!({}));
161    let Some(perm_obj) = permissions.as_object_mut() else {
162        return changed;
163    };
164    let allow = perm_obj.entry("allow").or_insert_with(|| json!([]));
165    let Some(arr) = allow.as_array_mut() else {
166        return changed;
167    };
168    if arr.iter().any(|item| item.as_str() == Some(MCP_PERMISSION)) {
169        return changed;
170    }
171    arr.push(json!(MCP_PERMISSION));
172    true
173}

175/// Approves our own server from the project's `.mcp.json`.
176///
177/// A project-scoped server is not connected until it is approved: Claude Code
178/// shows it as "Pending approval" and its tools are simply absent from the
179/// session, so writing `.mcp.json` alone left the install looking done and
180/// doing nothing (docs.claude.com/en/docs/claude-code/mcp, project scope).
181/// `enabledMcpjsonServers` is the key Claude Code itself writes when you
182/// approve one, and `.claude/settings.local.json` is where it writes it: that
183/// file is personal and untracked, so the approval is ours alone rather than
184/// something every teammate who clones the repository inherits, and it is
185/// honored even in a folder whose trust dialog has not been accepted
186/// (settings reference, `enabledMcpjsonServers`).
187///
188/// User-scoped servers in `~/.claude.json` need no approval, so this is for
189/// project installs only.
190///
191/// The approval does not take effect before the folder is trusted: in a folder
192/// whose trust dialog has not been accepted, `claude mcp list` and `claude mcp
193/// get` ignore these keys, and checking with the real CLI in a scratch project
194/// shows `Pending approval` for `enabledMcpjsonServers` and for the blanket
195/// `enableAllProjectMcpServers` alike. Nothing an installer writes can accept
196/// that dialog, so the note says what is left to do.
197fn sync_project_mcp_approval(out: &mut Outcome, folder: &Path, enabled: bool) {
198    let path = folder.join(".claude").join("settings.local.json");
199    let existed = path.exists();
200    if !enabled && !existed {
201        return;
202    }
203    let Some(mut json) = out.load_json(&path) else {
204        return;
205    };
206    let Some(obj) = json.as_object_mut() else {
207        return;
208    };
209
210    let changed = if enabled {
211        let list = obj
212            .entry("enabledMcpjsonServers")
213            .or_insert_with(|| json!([]));
214        match list.as_array_mut() {
215            Some(arr) if !arr.iter().any(|s| s.as_str() == Some("telemaco")) => {
216                arr.push(json!("telemaco"));
217                true
218            }
219            _ => false,
220        }
221    } else {
222        match obj
223            .get_mut("enabledMcpjsonServers")
224            .and_then(|v| v.as_array_mut())
225        {
226            Some(arr) => {
227                let before = arr.len();
228                arr.retain(|s| s.as_str() != Some("telemaco"));
229                arr.len() != before
230            }
231            None => false,
232        }
233    };
234    if !changed {
235        return;
236    }
237
238    if enabled && !existed {
239        out.note(format!(
240            "Approved the telemaco server in {}. Claude Code applies it once you accept the \
241             folder's trust dialog on the first interactive run. That file is personal: add \
242             it to .gitignore if git does not ignore it already.",
243            path.display()
244        ));
245    }
246    let action = if existed { Action::Updated } else { Action::Created };
247    out.write_json_or_remove(&path, &json, action);
248}

250/// True when a PreToolUse group is the guard *we* installed.
251///
252/// Matching on the matcher alone is not enough: a user may well have written
253/// their own `WebSearch|WebFetch` guard, and uninstalling Telemaco must not
254/// take it away with it.
255fn is_our_web_block(group: &serde_json::Value) -> bool {
256    const MATCHER: &str = "WebSearch|WebFetch";
257    if group.get("matcher").and_then(|m| m.as_str()) != Some(MATCHER) {
258        return false;
259    }
260    group
261        .get("hooks")
262        .and_then(|h| h.as_array())
263        .map_or(false, |arr| {
264            arr.iter().any(|entry| {
265                entry
266                    .get("command")
267                    .and_then(|c| c.as_str())
268                    .map_or(false, |cmd| cmd.to_lowercase().contains("telemaco"))
269            })
270        })
271}

273/// Adds or removes the PreToolUse guard that refuses Claude Code's built-in
274/// WebSearch and WebFetch. Returns whether the settings changed.
275///
276/// Removing on `enabled == false` matters: reinstalling with the guard turned
277/// off has to undo an earlier install that turned it on.
278fn sync_web_block(settings_json: &mut serde_json::Value, enabled: bool) -> bool {
279    if !enabled {
280        let Some(groups) = settings_json
281            .get_mut("hooks")
282            .and_then(|h| h.get_mut("PreToolUse"))
283            .and_then(|p| p.as_array_mut())
284        else {
285            return false;
286        };
287        let before = groups.len();
288        groups.retain(|g| !is_our_web_block(g));
289        return groups.len() != before;
290    }
291
292    let hooks_obj = settings_json
293        .as_object_mut()
294        .expect("load_json guarantees an object")
295        .entry("hooks")
296        .or_insert_with(|| json!({}));
297    let Some(hooks_map) = hooks_obj.as_object_mut() else {
298        return false;
299    };
300    let pre_tool = hooks_map.entry("PreToolUse").or_insert_with(|| json!([]));
301    let Some(groups) = pre_tool.as_array_mut() else {
302        return false;
303    };
304
305    for g in groups.iter_mut() {
306        if !is_our_web_block(g) {
307            continue;
308        }
309        let mut changed = false;
310        if let Some(first) = g.get_mut("hooks").and_then(|h| h.as_array_mut()).and_then(|a| a.first_mut()) {
311            if first.get("command").and_then(|c| c.as_str()) != Some(WEB_BLOCK_COMMAND) {
312                first["command"] = json!(WEB_BLOCK_COMMAND);
313                changed = true;
314            }
315        }
316        return changed;
317    }
318
319    groups.push(json!({
320        "matcher": "WebSearch|WebFetch",
321        "hooks": [
322            {
323                "type": "command",
324                "command": WEB_BLOCK_COMMAND
325            }
326        ]
327    }));
328    true
329}

331pub fn install(loc: &Location, opts: &TargetInstallOptions, home: &PathBuf) -> TargetResult {
332    let mut out = Outcome::new(opts.dry_run);
333
334    // 1. MCP entry in ~/.claude.json (global) or <folder>/.mcp.json (folder)
335    let mcp_path = match loc {
336        Location::Global => claude_json_path(home),
337        Location::Folder(folder) => folder.join(".mcp.json"),
338    };
339    upsert_mcp_server(
340        &mut out,
341        &mcp_path,
342        "telemaco",
343        stdio_typed_mcp_entry(&opts.binary_path, opts.stealth),
344    );
345
346    // 2. Settings.json (autoApprove + prompt-hook + PreToolUse)
347    let settings_path = match loc {
348        Location::Global => claude_config_dir(home).join("settings.json"),
349        Location::Folder(folder) => folder.join(".claude").join("settings.json"),
350    };
351    let settings_existed = settings_path.exists();
352    if let Some(mut settings_json) = out.load_json(&settings_path) {
353        let mut modified = false;
354
355        if sync_permission_allow(&mut settings_json, opts.auto_allow) {
356            modified = true;
357        }
358
359        // Register UserPromptSubmit hook
360        if opts.prompt_hook && add_user_prompt_hook(&mut settings_json, &prompt_hook_command(&opts.binary_path)) {
361            modified = true;
362        } else if !opts.prompt_hook && remove_user_prompt_hook(&mut settings_json) {
363            // Reinstalling with the hook declined takes back the one an
364            // earlier install added.
365            modified = true;
366        }
367
368        // PreToolUse guard on the agent's own web tools
369        if sync_web_block(&mut settings_json, opts.block_builtin_web) {
370            modified = true;
371        }
372
373        if modified {
374            let action = if settings_existed { Action::Updated } else { Action::Created };
375            out.write_json(&settings_path, &settings_json, action);
376        }
377    }
378
379    // 3. A project server has to be approved before Claude Code connects it.
380    if let Location::Folder(folder) = loc {
381        sync_project_mcp_approval(&mut out, folder, opts.auto_allow);
382    }
383
384    // 4. Instructions in CLAUDE.md
385    let instructions_path = match loc {
386        Location::Global => claude_config_dir(home).join("CLAUDE.md"),
387        Location::Folder(folder) => project_instructions_path(folder),
388    };
389    update_instructions(&mut out, &instructions_path, opts.stealth);
390
391    out.finish(TargetId::Claude)
392}

394pub fn uninstall(loc: &Location, home: &PathBuf, dry_run: bool) -> TargetResult {
395    let mut out = Outcome::new(dry_run);
396
397    let mcp_path = match loc {
398        Location::Global => claude_json_path(home),
399        Location::Folder(folder) => folder.join(".mcp.json"),
400    };
401    remove_mcp_server(&mut out, &mcp_path, "telemaco");
402
403    let settings_path = match loc {
404        Location::Global => claude_config_dir(home).join("settings.json"),
405        Location::Folder(folder) => folder.join(".claude").join("settings.json"),
406    };
407    if settings_path.exists() {
408        if let Some(mut settings_json) = out.load_json(&settings_path) {
409            let mut modified = false;
410
411            if sync_permission_allow(&mut settings_json, false) {
412                modified = true;
413            }
414
415            if remove_user_prompt_hook(&mut settings_json) {
416                modified = true;
417            }
418
419            if sync_web_block(&mut settings_json, false) {
420                modified = true;
421            }
422
423            if modified {
424                out.write_json_or_remove(&settings_path, &settings_json, Action::Updated);
425            }
426        }
427    }
428
429    if let Location::Folder(folder) = loc {
430        sync_project_mcp_approval(&mut out, folder, false);
431    }
432
433    match loc {
434        Location::Global => {
435            remove_instructions(&mut out, &claude_config_dir(home).join("CLAUDE.md"))
436        }
437        Location::Folder(folder) => {
438            for path in all_project_instructions(folder) {
439                if path.exists() {
440                    remove_instructions(&mut out, &path);
441                }
442            }
443        }
444    }
445
446    out.finish(TargetId::Claude)
