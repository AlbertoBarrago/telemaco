7pub fn detect(loc: &Location, home: Option<&PathBuf>) -> DetectionResult {
8    let (kiro_dir, mcp_path) = match loc {
9        Location::Global => {
10            let h = home.cloned().unwrap_or_else(|| PathBuf::from("."));
11            let k_dir = h.join(".kiro");
12            let mcp = k_dir.join("settings").join("mcp.json");
13            (k_dir, mcp)
14        }
15        Location::Folder(folder) => {
16            let k_dir = folder.join(".kiro");
17            let mcp = k_dir.join("settings").join("mcp.json");
18            (k_dir, mcp)
19        }
20    };
21    let hooks_path = kiro_dir.join("hooks").join("telemaco.json");
22    // Only `.kiro/` identifies Kiro. Looking for "kiro" inside AGENTS.md
23    // matched every repository whose instructions merely mention the agent -
24    // this one included - so `--target auto` configured Kiro for people who do
25    // not run it. Same trap as the `.pi` check in the Pi target.
26    let installed = kiro_dir.exists();
27
28    let mut already_configured = false;
29    if mcp_path.exists() {
30        let json = read_json_file(&mcp_path);
31        if let Some(servers) = json.get("mcpServers").and_then(|v| v.as_object()) {
32            already_configured = servers.contains_key("telemaco");
33        }
34    }
35    if !already_configured && hooks_path.exists() {
36        // The file is ours by name, but an empty or hand-edited one is not a
37        // working install.
38        if let Ok(content) = fs::read_to_string(&hooks_path) {
39            already_configured = text_has_telemaco_hook(&content);
40        }
41    }
42    if !already_configured {
43        let steering = match loc {
44            Location::Global => kiro_dir.join("steering").join("telemaco.md"),
45            Location::Folder(folder) => {
46                if kiro_dir.join("steering").exists() {
47                    kiro_dir.join("steering").join("telemaco.md")
48                } else {
49                    folder.join("AGENTS.md")
50                }
51            }
52        };
53        already_configured = fs::read_to_string(&steering)
54            .map_or(false, |c| c.contains("<!-- TELEMACO_START -->"));
55    }
56
57    let hint = if already_configured {
58        "already configured".to_string()
59    } else if installed {
60        if loc.is_global() {
61            "detected".to_string()
62        } else {
63            let mut markers = Vec::new();
64            if kiro_dir.exists() { markers.push(".kiro/"); }
65            if markers.is_empty() { "detected".to_string() } else { markers.join(", ") }
66        }
67    } else {
68        String::new()
69    };
70
71    DetectionResult {
72        installed,
73        already_configured,
74        config_path: Some(mcp_path),
75        hint,
76    }
77}

79/// Kiro's `autoApprove` list.
80///
81/// Always written, empty when the user declined: leaving the key out let an
82/// earlier `["*"]` survive the merge, so `--no-permissions` revoked nothing.
83pub fn mcp_entry(binary_path: &str, stealth: bool, auto_allow: bool) -> serde_json::Value {
84    json!({
85        "command": binary_path,
86        "args": stdio_mcp_args(stealth),
87        "disabled": false,
88        "autoApprove": if auto_allow { json!(["*"]) } else { json!([]) }
89    })
90}

92fn hook_command_of(entry: &serde_json::Value) -> Option<&str> {
93    entry.get("action").and_then(|a| a.get("command")).and_then(|c| c.as_str())
94}

96pub fn install(loc: &Location, opts: &TargetInstallOptions, home: &PathBuf) -> TargetResult {
97    let mut out = Outcome::new(opts.dry_run);
98
99    let kiro_dir = match loc {
100        Location::Global => home.join(".kiro"),
101        Location::Folder(folder) => folder.join(".kiro"),
102    };
103
104    // 1. MCP entry in ~/.kiro/settings/mcp.json or <folder>/.kiro/settings/mcp.json
105    let mcp_path = kiro_dir.join("settings").join("mcp.json");
106    upsert_mcp_server(
107        &mut out,
108        &mcp_path,
109        "telemaco",
110        mcp_entry(&opts.binary_path, opts.stealth, opts.auto_allow),
111    );
112
113    // 2. Hook in ~/.kiro/hooks/telemaco.json or <folder>/.kiro/hooks/telemaco.json
114    let hooks_path = kiro_dir.join("hooks").join("telemaco.json");
115    let hook_cmd = prompt_hook_command(&opts.binary_path);
116    // Kiro passes the prompt in `USER_PROMPT` rather than in the stdin payload
117    // (kiro.dev/docs/hooks/types, Prompt Submit), which `prompt-hook` reads as
118    // a fallback. Its stdout on exit 0 "is added to the agent's context"
119    // (kiro.dev/docs/hooks/actions), so plain text is the right format here.
120    // `UserPromptSubmit` is the event name Kiro documents for the prompt
121    // trigger (kiro.dev/docs/hooks/actions); "PromptSubmit" is the label in the
122    // UI, not the identifier, so the hook never fired.
123    let desired_hook = json!({
124        "name": "Telemaco Prompt Hook",
125        "trigger": "UserPromptSubmit",
126        "action": {
127            "type": "command",
128            "command": hook_cmd
129        }
130    });
131
132    // Kiro loads hook files from `.kiro/hooks/` in a project root, and
133    // documents no user-level hooks directory (kiro.dev/docs/hooks): a global
134    // install gets the directive through the steering file instead.
135    if !opts.prompt_hook {
136        // Declined: the steering file still carries the instructions. Take
137        // back the dedicated hook file an earlier install wrote.
138        if hooks_path.exists() {
139            out.remove_config_file(&hooks_path, "");
140        }
141    } else if loc.is_global() {
142        out.note(
143            "Kiro reads hooks from a project's .kiro/hooks/ only, so no global prompt hook \
144             is installed; the steering file covers every project.",
145        );
146    } else if !hooks_path.exists() {
147        let content = json!({ "version": "v1", "hooks": [desired_hook] });
148        out.write_json(&hooks_path, &content, Action::Created);
149    } else if let Some(mut cur_json) = out.load_json(&hooks_path) {
150        match cur_json.get_mut("hooks").and_then(|h| h.as_array_mut()) {
151            Some(hooks_arr) => {
152                let mut changed = false;
153                let existing = hooks_arr
154                    .iter_mut()
155                    .find(|h| hook_command_of(h).map_or(false, is_telemaco_hook_command));
156                match existing {
157                    Some(entry) => {
158                        if hook_command_of(entry) != Some(hook_cmd.as_str()) {
159                            entry["action"]["command"] = json!(hook_cmd);
160                            changed = true;
161                        }
162                        // Repair the trigger an older Telemaco wrote.
163                        if entry.get("trigger").and_then(|t| t.as_str()) != Some("UserPromptSubmit") {
164                            entry["trigger"] = json!("UserPromptSubmit");
165                            changed = true;
166                        }
167                    }
168                    None => {
169                        hooks_arr.push(desired_hook);
170                        changed = true;
171                    }
172                }
173                if changed {
174                    out.write_json(&hooks_path, &cur_json, Action::Updated);
175                } else {
176                    out.push(FileResult { path: hooks_path, action: Action::Unchanged });
177                }
178            }
179            None => {
180                let content = json!({ "version": "v1", "hooks": [desired_hook] });
181                out.write_json(&hooks_path, &content, Action::Updated);
182            }
183        }
184    }
185
186    // 3. Instructions: steering file or AGENTS.md
187    let instructions_path = match loc {
188        Location::Global => kiro_dir.join("steering").join("telemaco.md"),
189        Location::Folder(folder) => {
190            if folder.join(".kiro").join("steering").exists() {
191                folder.join(".kiro").join("steering").join("telemaco.md")
192            } else {
193                folder.join("AGENTS.md")
194            }
195        }
196    };
197    update_instructions(&mut out, &instructions_path, opts.stealth);
198
199    out.finish(TargetId::Kiro)
200}

202pub fn uninstall(loc: &Location, home: &PathBuf, dry_run: bool) -> TargetResult {
203    let mut out = Outcome::new(dry_run);
204
205    let kiro_dir = match loc {
206        Location::Global => home.join(".kiro"),
207        Location::Folder(folder) => folder.join(".kiro"),
208    };
209
210    // 1. MCP
211    remove_mcp_server(&mut out, &kiro_dir.join("settings").join("mcp.json"), "telemaco");
212
213    // 2. Hooks
214    let hooks_path = kiro_dir.join("hooks").join("telemaco.json");
215    if hooks_path.exists() {
216        if let Some(mut cur_json) = out.load_json(&hooks_path) {
217            let mut modified = false;
218            let mut now_empty = false;
219            if let Some(hooks_arr) = cur_json.get_mut("hooks").and_then(|h| h.as_array_mut()) {
220                let prev_len = hooks_arr.len();
221                hooks_arr.retain(|h| !hook_command_of(h).map_or(false, is_telemaco_hook_command));
222                if hooks_arr.len() != prev_len {
223                    modified = true;
224                }
225                now_empty = hooks_arr.is_empty();
226            }
227            if now_empty {
228                out.remove_file(&hooks_path);
229            } else if modified {
230                out.write_json_or_remove(&hooks_path, &cur_json, Action::Updated);
231            }
232        }
233    }
234
235    // 3. Instructions
236    let instructions_paths = match loc {
237        Location::Global => vec![kiro_dir.join("steering").join("telemaco.md")],
238        Location::Folder(folder) => vec![
239            folder.join(".kiro").join("steering").join("telemaco.md"),
240            folder.join("AGENTS.md"),
241        ],
242    };
243    for p in instructions_paths {
244        remove_instructions(&mut out, &p);
245    }
246
247    out.finish(TargetId::Kiro)
248}
