5fn config_dir_for(loc: &Location, home: &PathBuf) -> PathBuf {
6    match loc {
7        Location::Global => home.join(".qwen"),
8        Location::Folder(folder) => folder.join(".qwen"),
9    }
10}

12/// The context file names Qwen was told to load, if the default was replaced.
13///
14/// `QWEN.md` is the default, "configurable via the `context.fileName` setting"
15/// (QwenLM/qwen-code docs/users/configuration/settings.md). A project's
16/// `.qwen/settings.json` overrides the user's.
17fn configured_names(loc: &Location, home: &PathBuf) -> Option<Vec<String>> {
18    let mut sources = Vec::new();
19    if let Location::Folder(folder) = loc {
20        sources.push(folder.join(".qwen").join("settings.json"));
21    }
22    sources.push(home.join(".qwen").join("settings.json"));
23    configured_context_file_names(&sources)
24}
25
26/// The context file to write.
27///
28/// With a configured name, that name and nothing else: the default is not
29/// loaded any more. Without one, an existing `QWEN.md` if the project has one,
30/// otherwise `AGENTS.md`, which Qwen reads as well ("If your repository already
31/// has an `AGENTS.md` file for other AI tools, Qwen reads that too",
32/// docs/users/features/memory.md) and which the other agents share.
33fn instructions_path(loc: &Location, home: &PathBuf) -> PathBuf {
34    let configured = configured_names(loc, home);
35    match loc {
36        Location::Global => {
37            let name = configured
38                .and_then(|n| n.into_iter().next())
39                .unwrap_or_else(|| "QWEN.md".to_string());
40            home.join(".qwen").join(name)
41        }
42        Location::Folder(folder) => match configured {
43            Some(names) => names
44                .iter()
45                .map(|n| folder.join(n))
46                .find(|p| p.exists())
47                .unwrap_or_else(|| folder.join(&names[0])),
48            None => {
49                if folder.join("QWEN.md").exists() {
50                    folder.join("QWEN.md")
51                } else {
52                    folder.join("AGENTS.md")
53                }
54            }
55        },
56    }
57}

59/// Every context file a previous install may have written into.
60fn all_instructions_paths(loc: &Location, home: &PathBuf) -> Vec<PathBuf> {
61    let mut names = configured_names(loc, home).unwrap_or_default();
62    for default in ["QWEN.md", "AGENTS.md"] {
63        if !names.iter().any(|n| n == default) {
64            names.push(default.to_string());
65        }
66    }
67    match loc {
68        // A global install never writes AGENTS.md: `~/.qwen/AGENTS.md` is not
69        // a file Qwen reads.
70        Location::Global => names
71            .into_iter()
72            .filter(|n| n != "AGENTS.md")
73            .map(|n| home.join(".qwen").join(n))
74            .collect(),
75        Location::Folder(folder) => names.into_iter().map(|n| folder.join(n)).collect(),
76    }
77}

79pub fn detect(loc: &Location, home: Option<&PathBuf>) -> DetectionResult {
80    let h = home.cloned().unwrap_or_else(|| PathBuf::from("."));
81    let qwen_dir = config_dir_for(loc, &h);
82    let settings_path = qwen_dir.join("settings.json");
83    let installed = match loc {
84        Location::Global => qwen_dir.exists() || settings_path.exists(),
85        Location::Folder(folder) => {
86            qwen_dir.exists() || settings_path.exists() || folder.join("QWEN.md").exists()
87        }
88    };
89    let mut already_configured = false;
90    if settings_path.exists() {
91        let json = read_json_file(&settings_path);
92        if let Some(servers) = json.get("mcpServers").and_then(|v| v.as_object()) {
93            already_configured = servers.contains_key("telemaco");
94        }
95        if !already_configured {
96            // Our hook, not any UserPromptSubmit hook: the user's own live
97            // in the same array.
98            already_configured = json
99                .get("hooks")
100                .and_then(|hooks| hooks.get("UserPromptSubmit"))
101                .map_or(false, |v| text_has_telemaco_hook(&v.to_string()));
102        }
103    }
104    let hint = if already_configured {
105        "already configured".to_string()
106    } else if installed {
107        if loc.is_global() {
108            "detected".to_string()
109        } else {
110            let mut markers = Vec::new();
111            if let Location::Folder(folder) = loc {
112                if folder.join(".qwen").exists() { markers.push(".qwen/"); }
113                if folder.join("QWEN.md").exists() { markers.push("QWEN.md"); }
114            }
115            if markers.is_empty() { "detected".to_string() } else { markers.join(", ") }
116        }
117    } else {
118        String::new()
119    };
120    DetectionResult {
121        installed,
122        already_configured,
123        config_path: Some(settings_path),
124        hint,
125    }
126}

128pub fn install(loc: &Location, opts: &TargetInstallOptions, home: &PathBuf) -> TargetResult {
129    let mut out = Outcome::new(opts.dry_run);
130
131    let config_dir = config_dir_for(loc, home);
132    let settings_path = config_dir.join("settings.json");
133    let settings_existed = settings_path.exists();
134    if let Some(mut settings_json) = out.load_json(&settings_path) {
135        let target_entry = stdio_mcp_entry(&opts.binary_path, opts.stealth);
136
137        // Through the shared helper, so the entry is merged rather than
138        // replaced: this target hand-rolled the insert and so never got the
139        // fix that keeps a user's `env` or `timeout` across a reinstall.
140        let mut modified = match upsert_server_entry(
141            &mut settings_json,
142            "mcpServers",
143            "telemaco",
144            target_entry,
145        ) {
146            Ok(changed) => changed,
147            Err(why) => {
148                // Recorded, then carry on: the instructions file does not
149                // depend on this key being usable.
150                out.note(format!("{}: {}", settings_path.display(), why));
151                false
152            }
153        };
154
155        // Qwen parses a command hook's stdout as JSON and ignores anything
156        // else, so the plain-text directive never reached the model
157        // (QwenLM/qwen-code docs/users/features/hooks.md).
158        if opts.prompt_hook && add_user_prompt_hook(&mut settings_json, &prompt_hook_command_json(&opts.binary_path)) {
159            modified = true;
160        } else if !opts.prompt_hook && remove_user_prompt_hook(&mut settings_json) {
161            // Reinstalling with the hook declined takes back the one an
162            // earlier install added.
163            modified = true;
164        }
165
166        if modified {
167            let action = if settings_existed { Action::Updated } else { Action::Created };
168            out.write_json(&settings_path, &settings_json, action);
169        } else {
170            out.push(FileResult { path: settings_path, action: Action::Unchanged });
171        }
172    }
173
174    update_instructions(&mut out, &instructions_path(loc, home), opts.stealth);
175
176    // Folder trust is off by default, but once it is on an untrusted project
177    // has its `.qwen/settings.json` ignored outright
178    // (docs/users/configuration/trusted-folders.md).
179    if loc.folder().is_some() {
180        out.note(
181            "With folder trust enabled, Qwen Code ignores a project's .qwen/settings.json until \
182             you trust the folder, so the MCP entry and the prompt hook start working after that.",
183        );
184    }
185
186    out.finish(TargetId::QwenCode)
187}

189pub fn uninstall(loc: &Location, home: &PathBuf, dry_run: bool) -> TargetResult {
190    let mut out = Outcome::new(dry_run);
191
192    let settings_path = config_dir_for(loc, home).join("settings.json");
193    if settings_path.exists() {
194        if let Some(mut json) = out.load_json(&settings_path) {
195            let mut modified = false;
196            if let Some(servers) = json.get_mut("mcpServers").and_then(|v| v.as_object_mut()) {
197                if servers.remove("telemaco").is_some() {
198                    modified = true;
199                }
200            }
201            if remove_user_prompt_hook(&mut json) {
202                modified = true;
203            }
204            if modified {
205                out.write_json_or_remove(&settings_path, &json, Action::Updated);
206            }
207        }
208    }
209
210    for p in all_instructions_paths(loc, home) {
211        remove_instructions(&mut out, &p);
212    }
213
214    out.finish(TargetId::QwenCode)
215}
