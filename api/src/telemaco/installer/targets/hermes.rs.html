6/// Hermes' home directory: `~/.hermes` unless `$HERMES_HOME` says otherwise.
7///
8/// The docs use `$HERMES_HOME` throughout ("`~/.hermes/SOUL.md` or
9/// `$HERMES_HOME/SOUL.md` if you run Hermes with a custom home directory"), and
10/// the config, the secrets file and the MCP tokens all live under it
11/// (hermes-agent.nousresearch.com/docs/user-guide/features/context-files,
12/// /docs/reference/mcp-config-reference).
13fn hermes_home(home: &PathBuf) -> PathBuf {
14    home_env_var("HERMES_HOME").unwrap_or_else(|| home.join(".hermes"))
15}
16
17/// Both the MCP servers and the shell hooks live in one YAML file.
18fn config_path(home: &PathBuf) -> PathBuf {
19    hermes_home(home).join("config.yaml")
20}

22/// The project context file Hermes actually reads.
23///
24/// Hermes scans the working directory for `.hermes.md`, then `AGENTS.md`, then
25/// `CLAUDE.md`, then `.cursorrules`, and the **first match wins** - only one of
26/// them is loaded (hermes-agent.nousresearch.com/docs/user-guide/features/context-files).
27/// So the block has to go in whichever file Hermes would pick, or it is written
28/// somewhere that is never read. With none of them present, `AGENTS.md` is the
29/// one to create: it is the file the other agents share.
30fn project_context_file(folder: &Path) -> PathBuf {
31    for name in [".hermes.md", "AGENTS.md", "CLAUDE.md", ".cursorrules"] {
32        let candidate = folder.join(name);
33        if candidate.exists() {
34            return candidate;
35        }
36    }
37    folder.join("AGENTS.md")
38}
39
40/// Every project file we may have written into, for cleanup.
41fn all_project_context_files(folder: &Path) -> Vec<PathBuf> {
42    [".hermes.md", "AGENTS.md", "CLAUDE.md", ".cursorrules"]
43        .iter()
44        .map(|n| folder.join(n))
45        .collect()
46}

48pub fn detect(loc: &Location, home: Option<&PathBuf>) -> DetectionResult {
49    let h = home.cloned().unwrap_or_else(|| PathBuf::from("."));
50    let cfg = config_path(&h);
51    let installed = match loc {
52        Location::Global => hermes_home(&h).exists() || cfg.exists(),
53        // `.hermes.md` is the only project file Hermes owns; the other three it
54        // reads belong to the cross-tool conventions or to Cursor.
55        Location::Folder(folder) => folder.join(".hermes.md").exists(),
56    };
57
58    let mut already_configured = false;
59    if cfg.exists() {
60        if let Ok(content) = fs::read_to_string(&cfg) {
61            already_configured = content.lines().any(|l| l.trim() == "telemaco:");
62        }
63    }
64    if !already_configured {
65        if let Location::Folder(folder) = loc {
66            let ctx = project_context_file(folder);
67            if ctx.exists() {
68                already_configured = fs::read_to_string(&ctx)
69                    .map_or(false, |c| c.contains("<!-- TELEMACO_START -->"));
70            }
71        }
72    }
73
74    let hint = if already_configured {
75        "already configured".to_string()
76    } else if installed {
77        if loc.is_global() {
78            "detected".to_string()
79        } else {
80            ".hermes.md".to_string()
81        }
82    } else {
83        String::new()
84    };
85
86    DetectionResult {
87        installed,
88        already_configured,
89        config_path: Some(cfg),
90        hint,
91    }
92}

94/// The `pre_llm_call` entry, written at indent zero.
95///
96/// `pre_llm_call` is the event that can "inject context into the next LLM
97/// turn"; a hook answers with `{"context": "..."}` on stdout, which is what
98/// `--format hermes` prints. `timeout` is in seconds, default 60, capped at 300
99/// (hermes-agent.nousresearch.com/docs/user-guide/features/hooks).
100fn hook_item(binary_path: &str) -> String {
101    format!(
102        "- command: \"{} prompt-hook --format hermes\"\n  timeout: 30",
103        binary_path
104    )
105}

107pub fn install(loc: &Location, opts: &TargetInstallOptions, home: &PathBuf) -> TargetResult {
108    let mut out = Outcome::new(opts.dry_run);
109
110    // Hermes has no project-level config: the servers and the hooks are read
111    // from the one file under its home, so a folder install configures the
112    // context file only.
113    if loc.is_global() {
114        let cfg = config_path(home);
115        let mcp_entry = yaml_mcp_entry("telemaco", &opts.binary_path, &stdio_mcp_args(opts.stealth));
116        let hook = hook_item(&opts.binary_path);
117
118        if !cfg.exists() {
119            let mut content = format!("mcp_servers:\n{}\n", indent_block(&mcp_entry, 2));
120            if opts.prompt_hook {
121                content.push_str(&format!(
122                    "\nhooks:\n  pre_llm_call:\n{}\n",
123                    indent_block(&hook, 4)
124                ));
125            }
126            out.write_text(&cfg, &content, Action::Created);
127        } else if let Some(original) = out.load_text(&cfg) {
128            let mut content = original.clone();
129            let mut modified = false;
130
131            let has_entry = content.lines().any(|l| l.trim() == "telemaco:");
132            if has_entry {
133                if let Some(updated) = refresh_yaml_entry(&content, "telemaco", &mcp_entry) {
134                    if updated != content {
135                        content = updated;
136                        modified = true;
137                    }
138                }
139            } else {
140                match upsert_yaml_path(&content, &["mcp_servers"], &mcp_entry) {
141                    Ok(updated) => {
142                        content = updated;
143                        modified = true;
144                    }
145                    Err(e) => out.note(format!("{}: {}", cfg.display(), e)),
146                }
147            }
148
149            // Our hook is replaced rather than skipped, so a moved binary is
150            // followed the way every other target follows it. Declining the
151            // hook takes back the one an earlier install added.
152            let (without, removed) = remove_yaml_block(&content, |l| {
153                l.trim().starts_with("- command:") && is_telemaco_hook_command(l)
154            });
155            if removed {
156                content = without;
157            }
158            if opts.prompt_hook {
159                match upsert_yaml_path(&content, &["hooks", "pre_llm_call"], &hook) {
160                    Ok(updated) => {
161                        if updated != content {
162                            content = updated;
163                            modified = true;
164                        }
165                    }
166                    Err(e) => out.note(format!("{}: {}", cfg.display(), e)),
167                }
168            } else if removed {
169                content = prune_empty_yaml_keys(&content, &["hooks", "pre_llm_call"]);
170                modified = true;
171            }
172
173            if modified {
174                let rebuilt = with_line_ending(&original, &content);
175                out.write_text(&cfg, &rebuilt, Action::Updated);
176            } else {
177                out.push(FileResult { path: cfg, action: Action::Unchanged });
178            }
179        }
180
181        // `SOUL.md` is the only global context file, and it holds the agent's
182        // personality: the directive does not belong there. The hook covers
183        // every session instead.
184    }
185
186    let instructions_path = match loc {
187        Location::Global => None,
188        Location::Folder(folder) => Some(project_context_file(folder)),
189    };
190    if let Some(path) = instructions_path {
191        update_instructions(&mut out, &path, opts.stealth);
192    }
193
194    out.finish(TargetId::Hermes)
195}

197pub fn uninstall(loc: &Location, home: &PathBuf, dry_run: bool) -> TargetResult {
198    let mut out = Outcome::new(dry_run);
199
200    if loc.is_global() {
201        let cfg = config_path(home);
202        if cfg.exists() {
203            if let Some(original) = out.load_text(&cfg) {
204                let (content, removed_mcp) =
205                    remove_yaml_block(&original, |l| l.trim() == "telemaco:");
206                let (content, removed_hook) = remove_yaml_block(&content, |l| {
207                    l.trim().starts_with("- command:") && is_telemaco_hook_command(l)
208                });
209                if removed_mcp || removed_hook {
210                    let pruned =
211                        prune_empty_yaml_keys(&content, &["mcp_servers", "hooks", "pre_llm_call"]);
212                    if pruned.trim().is_empty() {
213                        out.remove_config_file(&cfg, "");
214                    } else {
215                        let rebuilt = with_line_ending(&original, &format!("{}\n", pruned.trim_end()));
216                        out.write_text(&cfg, &rebuilt, Action::Removed);
217                    }
218                }
219            }
220        }
221    }
222
223    if let Location::Folder(folder) = loc {
224        for path in all_project_context_files(folder) {
225            if path.exists() {
226                remove_instructions(&mut out, &path);
227            }
228        }
229    }
230
231    out.finish(TargetId::Hermes)
232}
