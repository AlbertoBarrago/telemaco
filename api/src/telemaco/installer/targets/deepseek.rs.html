6/// The harness home: `$DSH_HOME`, else `~/.dsh`.
7///
8/// Every path the harness reads is relative to it - "defaults to `$DSH_HOME`
9/// or `~/.dsh`" appears on the instruction loader, the skill roots and the
10/// patch layer alike (deepseek-ai/deepseek-harness docs/config-catalog.md).
11fn dsh_home(home: &PathBuf) -> PathBuf {
12    home_env_var("DSH_HOME").unwrap_or_else(|| home.join(".dsh"))
13}
14
15/// The user patch layer, where a plugin is turned on for every profile.
16///
17/// "To keep the selection across runs, merge the chosen file's single `insert`
18/// patch into a user patch layer - `$DSH_HOME/profiles/<profile>/cordis.patch.yml`
19/// for one profile, or `$DSH_HOME/cordis.patch.yml` for every profile on the
20/// machine" (docs/user/guide/mcp-memory.md). The file is a sequence of patch
21/// operations, so ours is appended to it rather than written over it: "Do not
22/// copy over an existing file: it may already contain unrelated user patches."
23fn patch_path(home: &PathBuf) -> PathBuf {
24    dsh_home(home).join("cordis.patch.yml")
25}

27const PATCH_ID: &str = "telemaco-mcp";
28
29/// The MCP server, in the shape `@deepseek-ai/dsh-mcp-client` declares:
30/// `serverName` (`[A-Za-z0-9_-]{1,32}`), `transport`, `command`, `args`
31/// (docs/config-catalog.md, `StdioConfig`).
32fn mcp_patch_item(binary_path: &str, stealth: bool) -> String {
33    let mut item = String::from("- insert:\n");
34    item.push_str(&format!("    - id: {}\n", PATCH_ID));
35    item.push_str("      name: \"@deepseek-ai/dsh-mcp-client\"\n");
36    item.push_str("      config:\n");
37    item.push_str("        serverName: telemaco\n");
38    item.push_str("        transport: stdio\n");
39    item.push_str(&format!("        command: \"{}\"\n", binary_path));
40    item.push_str("        args:\n");
41    for arg in stdio_mcp_args(stealth) {
42        item.push_str(&format!("          - \"{}\"\n", arg));
43    }
44    item
45}

47/// Whether a line carries our patch id, in a nested sequence entry
48/// (`    - id: telemaco-mcp`) or a plain mapping key (`      id: telemaco-mcp`).
49fn line_has_patch_id(line: &str) -> bool {
50    let trimmed = line.trim();
51    let trimmed = trimmed.strip_prefix("- ").unwrap_or(trimmed);
52    trimmed.trim() == format!("id: {}", PATCH_ID)
53}
54
55/// Line ranges of the top-level sequence items, one per patch operation.
56fn top_level_items(lines: &[&str]) -> Vec<(usize, usize)> {
57    let mut items: Vec<(usize, usize)> = Vec::new();
58    for (i, line) in lines.iter().enumerate() {
59        if line.starts_with("- ") || *line == "-" {
60            if let Some(last) = items.last_mut() {
61                last.1 = i;
62            }
63            items.push((i, lines.len()));
64        }
65    }
66    items
67}

69/// Replaces our patch operation with the one we would write today, or appends
70/// it. A stale copy names a binary that has moved, the same way every other
71/// target's entry would.
72fn upsert_patch_item(content: &str, fresh_item: &str) -> String {
73    let lines: Vec<&str> = content.lines().collect();
74    let ours = top_level_items(&lines)
75        .into_iter()
76        .find(|(start, end)| lines[*start..*end].iter().any(|l| line_has_patch_id(l)));
77
78    match ours {
79        Some((start, end)) => {
80            let mut out: Vec<String> = lines[..start].iter().map(|l| l.to_string()).collect();
81            out.extend(fresh_item.lines().map(|l| l.to_string()));
82            out.extend(lines[end..].iter().map(|l| l.to_string()));
83            format!("{}\n", out.join("\n"))
84        }
85        None => {
86            let mut out = content.trim_end().to_string();
87            if !out.is_empty() {
88                out.push('\n');
89            }
90            out.push_str(fresh_item);
91            out
92        }
93    }
94}

96/// Drops our patch operation, leaving every other one untouched.
97fn remove_patch_item(content: &str) -> (String, bool) {
98    let lines: Vec<&str> = content.lines().collect();
99    let ours = top_level_items(&lines)
100        .into_iter()
101        .find(|(start, end)| lines[*start..*end].iter().any(|l| line_has_patch_id(l)));
102
103    match ours {
104        Some((start, end)) => {
105            let mut out: Vec<String> = lines[..start].iter().map(|l| l.to_string()).collect();
106            out.extend(lines[end..].iter().map(|l| l.to_string()));
107            let joined = out.join("\n");
108            if joined.trim().is_empty() {
109                (String::new(), true)
110            } else {
111                (format!("{}\n", joined.trim_end()), true)
112            }
113        }
114        None => (content.to_string(), false),
115    }
116}

118/// Whether anything but comments and blank lines is left.
119fn is_effectively_empty(content: &str) -> bool {
120    content
121        .lines()
122        .all(|l| l.trim().is_empty() || l.trim_start().starts_with('#'))
123}

125pub fn detect(loc: &Location, home: Option<&PathBuf>) -> DetectionResult {
126    let h = home.cloned().unwrap_or_else(|| PathBuf::from("."));
127    let (dsh_dir, instr_path, installed) = match loc {
128        Location::Global => {
129            let dsh_dir = dsh_home(&h);
130            let instr = dsh_dir.join("AGENTS.md");
131            let installed = dsh_dir.exists();
132            (dsh_dir, instr, installed)
133        }
134        Location::Folder(folder) => {
135            let dsh_dir = folder.join(".dsh");
136            let instr = folder.join("AGENTS.md");
137            let installed = dsh_dir.exists();
138            (dsh_dir, instr, installed)
139        }
140    };
141    let mut already_configured = false;
142    if instr_path.exists() {
143        if let Ok(content) = fs::read_to_string(&instr_path) {
144            already_configured = content.contains("<!-- TELEMACO_START -->");
145        }
146    }
147    let hooks_path = dsh_dir.join("hooks.json");
148    if !already_configured && hooks_path.exists() {
149        if let Ok(content) = fs::read_to_string(&hooks_path) {
150            already_configured = text_has_telemaco_hook(&content);
151        }
152    }
153    if !already_configured && loc.is_global() {
154        let patch = patch_path(&h);
155        if let Ok(content) = fs::read_to_string(&patch) {
156            already_configured = content.contains(&format!("id: {}", PATCH_ID));
157        }
158    }
159    let hint = if already_configured {
160        "already configured".to_string()
161    } else if installed {
162        if loc.is_global() {
163            "detected".to_string()
164        } else {
165            let mut markers = Vec::new();
166            if let Location::Folder(folder) = loc {
167                if folder.join(".dsh").exists() { markers.push(".dsh/"); }
168            }
169            if markers.is_empty() { "detected".to_string() } else { markers.join(", ") }
170        }
171    } else {
172        String::new()
173    };
174    DetectionResult {
175        installed,
176        already_configured,
177        config_path: Some(instr_path),
178        hint,
179    }
180}

182pub fn install(loc: &Location, opts: &TargetInstallOptions, home: &PathBuf) -> TargetResult {
183    let mut out = Outcome::new(opts.dry_run);
184
185    let dsh_dir = match loc {
186        Location::Global => dsh_home(home),
187        Location::Folder(folder) => folder.join(".dsh"),
188    };
189
190    // The MCP server goes in the user patch layer, which is the documented way
191    // to turn a plugin on for every run: the memory guide has the user merge
192    // exactly this kind of `insert` into it
193    // (docs/user/guide/mcp-memory.md). There is no project-level patch layer,
194    // so a folder install gets the instructions block only.
195    if loc.is_global() {
196        let patch = patch_path(home);
197        let fresh = mcp_patch_item(&opts.binary_path, opts.stealth);
198        if !patch.exists() {
199            out.write_text(&patch, &fresh, Action::Created);
200        } else if let Some(original) = out.load_text(&patch) {
201            let updated = upsert_patch_item(&original, &fresh);
202            if updated != original {
203                let rebuilt = with_line_ending(&original, &updated);
204                out.write_text(&patch, &rebuilt, Action::Updated);
205            } else {
206                out.push(FileResult { path: patch, action: Action::Unchanged });
207            }
208        }
209    }
210
211    // The file is in Claude Code's shape, which is what dsh's hook plugin
212    // reads, but the harness does not discover it on its own: the plugin takes
213    // an explicit `configPath` (deepseek-ai/deepseek-harness
214    // docs/config-catalog.md, `@deepseek-ai/dsh-hooks-claude-code`). Say so
215    // rather than let the user believe the hook is live.
216    let hooks_path = dsh_dir.join("hooks.json");
217    let hooks_existed = hooks_path.exists();
218    if let Some(mut hooks_json) = out.load_json(&hooks_path) {
219        if opts.prompt_hook && add_user_prompt_hook(&mut hooks_json, &prompt_hook_command(&opts.binary_path)) {
220            let action = if hooks_existed { Action::Updated } else { Action::Created };
221            out.write_json(&hooks_path, &hooks_json, action);
222        } else if !opts.prompt_hook && remove_user_prompt_hook(&mut hooks_json) {
223            // Reinstalling with the hook declined takes back the one an
224            // earlier install added, and takes the file with it when the hook
225            // was the only thing in it: this file exists only to hold the hook,
226            // so leaving `{"hooks":{"UserPromptSubmit":[]}}` behind would drop
227            // a config the user never wrote into their agent's directory. Same
228            // call the uninstall path uses.
229            out.write_json_or_remove(&hooks_path, &hooks_json, Action::Updated);
230        }
231        // Said on every run, not only the one that writes the file: an
232        // unchanged hook is just as inert until the plugin points at it.
233        out.note(format!(
234            "The prompt hook runs once you enable the \
235             @deepseek-ai/dsh-hooks-claude-code plugin with configPath: {}. \
236             The instructions block works without it.",
237            hooks_path.display()
238        ));
239    }
240
241    let instructions_path = match loc {
242        Location::Global => dsh_home(home).join("AGENTS.md"),
243        Location::Folder(folder) => folder.join("AGENTS.md"),
244    };
245    update_instructions(&mut out, &instructions_path, opts.stealth);
246
247    out.finish(TargetId::DeepSeek)
248}

250pub fn uninstall(loc: &Location, home: &PathBuf, dry_run: bool) -> TargetResult {
251    let mut out = Outcome::new(dry_run);
252
253    let dsh_dir = match loc {
254        Location::Global => dsh_home(home),
255        Location::Folder(folder) => folder.join(".dsh"),
256    };
257
258    if loc.is_global() {
259        let patch = patch_path(home);
260        if patch.exists() {
261            if let Some(original) = out.load_text(&patch) {
262                let (content, removed) = remove_patch_item(&original);
263                if removed {
264                    if is_effectively_empty(&content) {
265                        out.remove_config_file(&patch, "");
266                    } else {
267                        let rebuilt = with_line_ending(&original, &content);
268                        out.write_text(&patch, &rebuilt, Action::Removed);
269                    }
270                }
271            }
272        }
273    }
274
275    let hooks_path = dsh_dir.join("hooks.json");
276    if hooks_path.exists() {
277        if let Some(mut hooks_json) = out.load_json(&hooks_path) {
278            if remove_user_prompt_hook(&mut hooks_json) {
279                out.write_json_or_remove(&hooks_path, &hooks_json, Action::Updated);
280            }
281        }
282    }
283
284    let instructions_path = match loc {
285        Location::Global => dsh_home(home).join("AGENTS.md"),
286        Location::Folder(folder) => folder.join("AGENTS.md"),
287    };
288    remove_instructions(&mut out, &instructions_path);
289
290    out.finish(TargetId::DeepSeek)
291}
