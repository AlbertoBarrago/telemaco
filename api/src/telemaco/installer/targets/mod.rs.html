208pub fn get_home_dir() -> Option<PathBuf> {
209    std::env::var_os("HOME")
210        .or_else(|| std::env::var_os("USERPROFILE"))
211        .and_then(|h| if h.is_empty() { None } else { Some(PathBuf::from(h)) })
212        .or_else(|| {
213            #[cfg(windows)]
214            {
215                std::env::var_os("USERPROFILE").map(PathBuf::from)
216            }
217            #[cfg(not(windows))]
218            {
219                None
220            }
221        })
222}

233/// Absolute path to write into agent configs.
234///
235/// The running binary wins. Preferring `~/.local/bin/telemaco` instead meant a
236/// stale copy there got wired in while a newer binary was doing the install,
237/// and hooks from a version that lacks `prompt-hook` fail silently.
238pub fn resolve_telemaco_binary() -> String {
239    if let Ok(cur_exe) = std::env::current_exe() {
240        if cur_exe.exists() {
241            return cur_exe.display().to_string();
242        }
243    }
244
245    if let Some(home) = get_home_dir() {
246        let local_bin = home.join(".local").join("bin").join("telemaco");
247        if local_bin.exists() {
248            return local_bin.display().to_string();
249        }
250        let cargo_bin = home.join(".cargo").join("bin").join("telemaco");
251        if cargo_bin.exists() {
252            return cargo_bin.display().to_string();
253        }
254    }
255
256    "telemaco".to_string()
257}

259/// The MCP config snippet for a target, built from the very entry builders the
260/// installer writes. Printing and installing cannot drift apart.
261pub fn config_snippet(target: TargetId, binary: &str, stealth: bool, auto_allow: bool) -> String {
262    use common::{stdio_mcp_args, stdio_mcp_entry, stdio_typed_mcp_entry, yaml_mcp_entry};
263
264    fn wrap(container: &str, entry: serde_json::Value) -> String {
265        let doc = serde_json::json!({ container: { "telemaco": entry } });
266        serde_json::to_string_pretty(&doc).unwrap_or_default()
267    }
268
269    match target {
270        TargetId::Codex => {
271            let args = stdio_mcp_args(stealth)
272                .iter()
273                .map(|a| format!("\"{}\"", a))
274                .collect::<Vec<_>>()
275                .join(", ");
276            format!(
277                "[mcp_servers.telemaco]\ncommand = \"{}\"\nargs = [{}]",
278                binary, args
279            )
280        }
281        TargetId::Hermes => {
282            let entry = yaml_mcp_entry("telemaco", binary, &stdio_mcp_args(stealth));
283            let indented: Vec<String> = entry.lines().map(|l| format!("  {}", l)).collect();
284            format!("mcp_servers:\n{}", indented.join("\n"))
285        }
286        TargetId::Poolside => {
287            let entry = poolside::mcp_yaml_entry(binary, stealth);
288            let indented: Vec<String> = entry.lines().map(|l| format!("  {}", l)).collect();
289            format!("mcp_servers:\n{}", indented.join("\n"))
290        }
291        TargetId::OpenCode => wrap("mcp", opencode::mcp_entry(binary, stealth)),
292        TargetId::RooCline => wrap("mcpServers", roocline::mcp_entry(binary, stealth)),
293        TargetId::Kiro => wrap("mcpServers", kiro::mcp_entry(binary, stealth, auto_allow)),
294        TargetId::Claude | TargetId::Droid => {
295            wrap("mcpServers", stdio_typed_mcp_entry(binary, stealth))
296        }
297        TargetId::Pi | TargetId::DeepSeek => {
298            "Telemaco is configured through AGENTS.md instructions only; this agent has no MCP config file.".to_string()
299        }
300        _ => wrap("mcpServers", stdio_mcp_entry(binary, stealth)),
301    }
302}

304/// Detection, install and uninstall for one target, in one row.
305///
306/// These used to be three separate fourteen-arm matches. A target wired into
307/// one of them and forgotten in another still compiled, which is how Windsurf
308/// grew a `.devin/` write path that detection never looked at. One table means
309/// a target is either fully wired or it does not build.
310struct TargetOps {
311    detect: fn(&Location, Option<&PathBuf>) -> DetectionResult,
312    install: fn(&Location, &TargetInstallOptions, &PathBuf) -> TargetResult,
313    uninstall: fn(&Location, &PathBuf, bool) -> TargetResult,
314}
315
316macro_rules! target_ops {
317    ($m:ident) => {
318        TargetOps {
319            detect: $m::detect,
320            install: $m::install,
321            uninstall: $m::uninstall,
322        }
323    };
324}
325
326fn ops_for(target: TargetId) -> TargetOps {
327    match target {
328        TargetId::Claude => target_ops!(claude),
329        TargetId::Cursor => target_ops!(cursor),
330        TargetId::Codex => target_ops!(codex),
331        TargetId::Gemini => target_ops!(gemini),
332        TargetId::Antigravity => target_ops!(antigravity),
333        TargetId::Windsurf => target_ops!(windsurf),
334        TargetId::OpenCode => target_ops!(opencode),
335        TargetId::RooCline => target_ops!(roocline),
336        TargetId::Pi => target_ops!(pi),
337        TargetId::DeepSeek => target_ops!(deepseek),
338        TargetId::QwenCode => target_ops!(qwen),
339        TargetId::Droid => target_ops!(droid),
340        TargetId::Poolside => target_ops!(poolside),
341        TargetId::Kiro => target_ops!(kiro),
342        TargetId::Hermes => target_ops!(hermes),
343    }
344}

444    /// Serializes every test that sets one of the fake `TELEMACO_TEST_*`
445    /// home-dir overrides `home_env_var` reads in a test build, so two tests
446    /// never race setting the same process-global variable, and restores
447    /// whatever was there before (nothing, normally) once the test is done.
448    static ENV_VAR_LOCK: std::sync::Mutex<()> = std::sync::Mutex::new(());
449
450    struct FakeHomeVar {
451        key: String,
452        prev: Option<std::ffi::OsString>,
453        _guard: std::sync::MutexGuard<'static, ()>,
454    }
455
456    impl FakeHomeVar {
457        /// Sets `TELEMACO_TEST_<name>`, the test-build stand-in for the real
458        /// override `home_env_var` would read in production (`CODEX_HOME`,
459        /// `CLAUDE_CONFIG_DIR`, ...). Never touches the real variable, so a
460        /// real one exported in the ambient shell cannot leak into the test
461        /// and cannot be disturbed by it either.
462        fn set(name: &str, value: &Path) -> Self {
463            let guard = ENV_VAR_LOCK.lock().unwrap_or_else(|e| e.into_inner());
464            let key = format!("TELEMACO_TEST_{name}");
465            let prev = std::env::var_os(&key);
466            std::env::set_var(&key, value);
467            Self { key, prev, _guard: guard }
468        }
469    }
470
471    impl Drop for FakeHomeVar {
472        fn drop(&mut self) {
473            match self.prev.take() {
474                Some(v) => std::env::set_var(&self.key, v),
475                None => std::env::remove_var(&self.key),
476            }
477        }
478    }
