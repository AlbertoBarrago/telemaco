5/// Rules file for a project.
6///
7/// `.devin/rules/` is the preferred directory and `.windsurf/rules/` the
8/// fallback; the single-file `.windsurfrules` at the root is legacy but "also
9/// still read" (docs.windsurf.com/windsurf/cascade/memories). A rule file in
10/// either directory declares its activation mode in frontmatter, while the
11/// legacy root file is always on.
12///
13/// The test is the rules directory, never the bare `.devin/`: the MCP entry
14/// creates that directory itself, so keying off it made the answer depend on
15/// whether the install had already run - `--dry-run` named `.windsurfrules`
16/// and the install that followed wrote somewhere else.
17fn workspace_rules_path(folder: &Path) -> PathBuf {
18    if folder.join(".windsurf").join("rules").exists() {
19        folder.join(".windsurf").join("rules").join("telemaco.md")
20    } else if !folder.join(".devin").join("rules").exists()
21        && folder.join(".windsurfrules").exists()
22    {
23        folder.join(".windsurfrules")
24    } else {
25        folder.join(".devin").join("rules").join("telemaco.md")
26    }
27}

29/// Every MCP config a Windsurf user's agent actually reads.
30///
31/// Cascade reads `~/.codeium/windsurf/mcp_config.json` and has no project file
32/// at all, and that page says so itself: the MCP configuration there "applies
33/// to the legacy Cascade agent only", while the Devin Local agent, the default
34/// for new tabs, "configures MCP servers in the Devin CLI config files"
35/// (docs.windsurf.com/windsurf/cascade/mcp). Those are
36/// `~/.config/devin/mcp_config.json` and `.devin/mcp_config.json`, both keyed
37/// by `mcpServers` (docs.devin.ai/cli/extensibility/mcp/configuration). A
38/// project install used to write `.codeium/windsurf/mcp_config.json`, which
39/// neither agent opens.
40fn mcp_paths(loc: &Location, home: &PathBuf) -> Vec<PathBuf> {
41    match loc {
42        Location::Global => vec![
43            home.join(".codeium").join("windsurf").join("mcp_config.json"),
44            home.join(".config").join("devin").join("mcp_config.json"),
45        ],
46        Location::Folder(folder) => vec![folder.join(".devin").join("mcp_config.json")],
47    }
48}

50/// MCP paths earlier versions wrote, taken back out on uninstall.
51fn legacy_mcp_paths(loc: &Location) -> Vec<PathBuf> {
52    match loc {
53        Location::Global => Vec::new(),
54        Location::Folder(folder) => vec![
55            folder.join(".windsurf").join("mcp.json"),
56            folder.join(".codeium").join("windsurf").join("mcp_config.json"),
57        ],
58    }

70/// Where the prompt hook goes, and whether that file wraps it.
71///
72/// The Devin CLI runs `UserPromptSubmit` hooks and injects whatever they print
73/// as `hookSpecificOutput.additionalContext`, the same shape Claude Code and
74/// Qwen use (docs.devin.ai/cli/extensibility/hooks/lifecycle-hooks). Cascade
75/// has no hook of its own, which is why this target had none at all.
76///
77/// The file decides the shape: in `.devin/hooks.v1.json` "the hooks object is
78/// the entire file", while every other location nests it under a `hooks` key
79/// (docs.devin.ai/cli/extensibility/hooks/overview).
80fn hooks_target(loc: &Location, home: &PathBuf) -> (PathBuf, bool) {
81    match loc {
82        Location::Global => (
83            home.join(".config").join("devin").join("config.json"),
84            true,
85        ),
86        Location::Folder(folder) => (folder.join(".devin").join("hooks.v1.json"), false),
87    }
88}

90/// Global rule files, one per agent that reads one.
91///
92/// Cascade keeps its global rules in `~/.codeium/windsurf/memories/`. The Devin
93/// CLI does not read that file: its user-level rules are `~/.devin/rules/*.md`
94/// and `~/.devin/global_rules.md`, with the same `trigger` frontmatter a
95/// project rule uses (docs.devin.ai/cli/extensibility/rules). A global install
96/// that wrote only the Cascade file left the default agent with nothing.
97fn global_rule_paths(home: &PathBuf) -> Vec<(PathBuf, Option<&'static str>)> {
98    vec![
99        (
100            home.join(".codeium")
101                .join("windsurf")
102                .join("memories")
103                .join("global_rules.md"),
104            None,
105        ),
106        (
107            home.join(".devin").join("rules").join("telemaco.md"),
108            Some("trigger: always_on"),
109        ),
110    ]
111}

113pub fn detect(loc: &Location, home: Option<&PathBuf>) -> DetectionResult {
114    let h = home.cloned().unwrap_or_else(|| PathBuf::from("."));
115    let candidates: Vec<PathBuf> = mcp_paths(loc, &h)
116        .into_iter()
117        .chain(legacy_mcp_paths(loc))
118        .collect();
119    let installed = match loc {
120        Location::Global => {
121            h.join(".codeium").exists() || h.join(".config").join("devin").exists()
122        }
123        Location::Folder(folder) => {
124            folder.join(".codeium").exists()
125                || folder.join(".windsurf").exists()
126                || folder.join(".devin").exists()
127                || folder.join(".windsurfrules").exists()
128        }
129    };
130    let has_us = |path: &PathBuf| {
131        if !path.exists() {
132            return false;
133        }
134        let json = read_json_file(path);
135        json.get("mcpServers")
136            .and_then(|v| v.as_object())
137            .map_or(false, |s| s.contains_key("telemaco"))
138    };
139    let already_configured = candidates.iter().any(has_us);
140    let mcp_path = candidates
141        .iter()
142        .find(|p| p.exists())
143        .cloned()
144        .unwrap_or_else(|| candidates[0].clone());
145    let hint = if already_configured {
146        "already configured".to_string()
147    } else if installed {
148        if loc.is_global() {
149            "detected".to_string()
150        } else {
151            let mut markers = Vec::new();
152            if let Location::Folder(folder) = loc {
153                if folder.join(".devin").exists() { markers.push(".devin/"); }
154                if folder.join(".windsurf").exists() { markers.push(".windsurf/"); }
155                if folder.join(".codeium").exists() { markers.push(".codeium/"); }
156                if folder.join(".windsurfrules").exists() { markers.push(".windsurfrules"); }
157            }
158            if markers.is_empty() { "detected".to_string() } else { markers.join(", ") }
159        }
160    } else {
161        String::new()
162    };
163    DetectionResult {
164        installed,
165        already_configured,
166        config_path: Some(mcp_path),
167        hint,
168    }
169}

171pub fn install(loc: &Location, opts: &TargetInstallOptions, home: &PathBuf) -> TargetResult {
172    let mut out = Outcome::new(opts.dry_run);
173
174    for mcp_path in mcp_paths(loc, home) {
175        upsert_mcp_server(
176            &mut out,
177            &mcp_path,
178            "telemaco",
179            stdio_mcp_entry(&opts.binary_path, opts.stealth),
180        );
181    }
182
183    let (hooks_path, wrapped) = hooks_target(loc, home);
184    let hooks_existed = hooks_path.exists();
185    if let Some(mut hooks_json) = opts.prompt_hook.then(|| out.load_json(&hooks_path)).flatten() {
186        let cmd = prompt_hook_command_json(&opts.binary_path);
187        let changed = if wrapped {
188            add_user_prompt_hook(&mut hooks_json, &cmd)
189        } else {
190            add_user_prompt_hook_flat(&mut hooks_json, &cmd)
191        };
192        if changed {
193            let action = if hooks_existed { Action::Updated } else { Action::Created };
194            out.write_json(&hooks_path, &hooks_json, action);
195        } else {
196            out.push(FileResult { path: hooks_path, action: Action::Unchanged });
197        }
198    } else if !opts.prompt_hook {
199        // Reinstalling with the hook declined takes back the one an earlier
200        // install added.
201        if let Some(mut hooks_json) = out.load_json(&hooks_path) {
202            let removed = if wrapped {
203                remove_user_prompt_hook(&mut hooks_json)
204            } else {
205                remove_user_prompt_hook_flat(&mut hooks_json)
206            };
207            if removed {
208                // Takes the file too when the hook was all it held: see codex.
209                out.write_json_or_remove(&hooks_path, &hooks_json, Action::Updated);
210            }
211        }
212    }
213
214    match loc {
215        Location::Global => {
216            // Cascade's global rules file takes no frontmatter: it is always
217            // on. The Devin one declares its trigger like any other rule.
218            for (path, frontmatter) in global_rule_paths(home) {
219                match frontmatter {
220                    Some(fm) => update_rule_file(&mut out, &path, fm, opts.stealth),
221                    None => update_instructions(&mut out, &path, opts.stealth),
222                }
223            }
224        }
225        Location::Folder(folder) => {
226            let path = workspace_rules_path(folder);
227            if path.file_name().map_or(false, |n| n == "telemaco.md") {
228                update_rule_file(&mut out, &path, "trigger: always_on", opts.stealth);
229            } else {
230                update_instructions(&mut out, &path, opts.stealth);
231            }
232        }
233    }
234
235    out.finish(TargetId::Windsurf)
236}

238pub fn uninstall(loc: &Location, home: &PathBuf, dry_run: bool) -> TargetResult {
239    let mut out = Outcome::new(dry_run);
240
241    for mcp_path in mcp_paths(loc, home).into_iter().chain(legacy_mcp_paths(loc)) {
242        remove_mcp_server(&mut out, &mcp_path, "telemaco");
243    }
244
245    let (hooks_path, wrapped) = hooks_target(loc, home);
246    if hooks_path.exists() {
247        if let Some(mut hooks_json) = out.load_json(&hooks_path) {
248            let removed = if wrapped {
249                remove_user_prompt_hook(&mut hooks_json)
250            } else {
251                remove_user_prompt_hook_flat(&mut hooks_json)
252            };
253            if removed {
254                out.write_json_or_remove(&hooks_path, &hooks_json, Action::Updated);
255            }
256        }
257    }
258
259    match loc {
260        Location::Global => {
261            for (path, frontmatter) in global_rule_paths(home) {
262                if frontmatter.is_some() {
263                    // A rule file we created outright.
264                    if path.exists() {
265                        out.remove_file(&path);
266                    }
267                } else {
268                    remove_instructions(&mut out, &path);
269                }
270            }
271        }
272        Location::Folder(folder) => {
273            for path in all_workspace_rules(folder) {
274                if !path.exists() {
275                    continue;
276                }
277                if path.file_name().map_or(false, |n| n == "telemaco.md") {
278                    // A rule file we created outright: taking only the block
279                    // out would leave the frontmatter behind as a husk.
280                    out.remove_file(&path);
281                } else {
282                    remove_instructions(&mut out, &path);
283                }
284            }
285        }
286    }
287
288    out.finish(TargetId::Windsurf)
