6/// Pi's configuration directory: `~/.pi/agent` unless `PI_CODING_AGENT_DIR`
7/// says otherwise, which is the variable the CLI itself reads
8/// (nix-community.github.io/home-manager, programs.pi-coding-agent.configDir,
9/// "matching the upstream pi CLI default"). With the variable set, everything
10/// written under `~/.pi` is read by nothing, the same trap `CODEX_HOME` was.
11fn agent_dir(home: &PathBuf) -> PathBuf {
12    home_env_var("PI_CODING_AGENT_DIR").unwrap_or_else(|| home.join(".pi").join("agent"))
13}
14
15/// The context file Pi loads from a directory.
16///
17/// "Pi loads `AGENTS.md` (or `CLAUDE.md`) at startup from `~/.pi/agent/`,
18/// parent directories and the current directory [...] If a directory contains
19/// `AGENTS.override.md`, Pi loads it instead of `AGENTS.md` or `CLAUDE.md`
20/// from that directory" (earendil-works/pi packages/coding-agent/README.md).
21/// Written to `AGENTS.md` next to an override, the block is loaded by nothing.
22fn instructions_path(loc: &Location, home: &PathBuf) -> PathBuf {
23    match loc {
24        Location::Global => agents_file_in(&agent_dir(home)),
25        Location::Folder(folder) => agents_file_in(folder),
26    }
27}

29/// Both names, for cleanup.
30fn all_instructions_paths(loc: &Location, home: &PathBuf) -> Vec<PathBuf> {
31    match loc {
32        Location::Global => all_agents_files_in(&agent_dir(home)),
33        Location::Folder(folder) => all_agents_files_in(folder),
34    }
35}

37pub fn detect(loc: &Location, home: Option<&PathBuf>) -> DetectionResult {
38    let h = home.cloned().unwrap_or_else(|| PathBuf::from("."));
39    let instr_path = instructions_path(loc, &h);
40    // Only the .pi directory identifies Pi. Looking for "pi" inside AGENTS.md
41    // matched "pipeline", "API" and "expiry", so every repo with an AGENTS.md
42    // came back as a Pi install.
43    let installed = match loc {
44        Location::Global => h.join(".pi").exists() || agent_dir(&h).exists(),
45        Location::Folder(folder) => folder.join(".pi").exists(),
46    };
47    let mut already_configured = false;
48    if instr_path.exists() {
49        if let Ok(content) = fs::read_to_string(&instr_path) {
50            already_configured = content.contains("<!-- TELEMACO_START -->");
51        }
52    }
53    let hint = if already_configured {
54        "already configured".to_string()
55    } else if installed {
56        if loc.is_global() {
57            "detected".to_string()
58        } else {
59            let mut markers = Vec::new();
60            if let Location::Folder(folder) = loc {
61                if folder.join(".pi").exists() { markers.push(".pi/"); }
62            }
63            if markers.is_empty() { "detected".to_string() } else { markers.join(", ") }
64        }
65    } else {
66        String::new()
67    };
68    DetectionResult {
69        installed,
70        already_configured,
71        config_path: Some(instr_path),
72        hint,
73    }
74}

76pub fn install(loc: &Location, opts: &TargetInstallOptions, home: &PathBuf) -> TargetResult {
77    let mut out = Outcome::new(opts.dry_run);
78    update_instructions(&mut out, &instructions_path(loc, home), opts.stealth);
79    out.finish(TargetId::Pi)
80}
81
82pub fn uninstall(loc: &Location, home: &PathBuf, dry_run: bool) -> TargetResult {
83    let mut out = Outcome::new(dry_run);
84    for path in all_instructions_paths(loc, home) {
85        remove_instructions(&mut out, &path);
86    }
87    out.finish(TargetId::Pi)
