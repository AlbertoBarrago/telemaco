5/// Gemini CLI's user-level directory.
6///
7/// `~/.gemini` normally, but `GEMINI_CLI_HOME` moves the root the CLI creates
8/// that `.gemini` folder inside ("Specifies the root directory for Gemini
9/// CLI's user-level configuration and storage [...] The CLI will create a
10/// `.gemini` folder inside this directory",
11/// google-gemini/gemini-cli docs/reference/configuration.md). With it set,
12/// everything written under `~/.gemini` is read by nothing.
13fn gemini_dir(home: &PathBuf) -> PathBuf {
14    home_env_var("GEMINI_CLI_HOME")
15        .unwrap_or_else(|| home.clone())
16        .join(".gemini")
17}
18
19fn config_dir_for(loc: &Location, home: &PathBuf) -> PathBuf {
20    match loc {
21        Location::Global => gemini_dir(home),
22        Location::Folder(folder) => folder.join(".gemini"),
23    }
24}

26/// The context file names Gemini actually loads.
27///
28/// `GEMINI.md` is only the default: `context.fileName` replaces it, and takes
29/// either a name or a list of names that are all loaded and concatenated
30/// (docs/cli/gemini-md.md, "Customize the context file name"). A project's
31/// `.gemini/settings.json` wins over the user's. Written to the wrong name,
32/// the block is loaded by nothing.
33fn context_file_names(loc: &Location, home: &PathBuf) -> Vec<String> {
34    let mut sources = Vec::new();
35    if let Location::Folder(folder) = loc {
36        sources.push(folder.join(".gemini").join("settings.json"));
37    }
38    sources.push(gemini_dir(home).join("settings.json"));
39
40    configured_context_file_names(&sources).unwrap_or_else(|| vec!["GEMINI.md".to_string()])
41}
42
43/// The one file an install writes: the first name Gemini would load.
44fn instructions_path(loc: &Location, home: &PathBuf) -> PathBuf {
45    let name = context_file_names(loc, home)
46        .into_iter()
47        .next()
48        .unwrap_or_else(|| "GEMINI.md".to_string());
49    match loc {
50        Location::Global => gemini_dir(home).join(name),
51        Location::Folder(folder) => folder.join(name),
52    }
53}
54
55/// Every context file a previous install may have written into.
56fn all_instructions_paths(loc: &Location, home: &PathBuf) -> Vec<PathBuf> {
57    let mut names = context_file_names(loc, home);
58    if !names.iter().any(|n| n == "GEMINI.md") {
59        names.push("GEMINI.md".to_string());
60    }
61    names
62        .into_iter()
63        .map(|name| match loc {
64            Location::Global => gemini_dir(home).join(name),
65            Location::Folder(folder) => folder.join(name),
66        })
67        .collect()
68}

70pub fn detect(loc: &Location, home: Option<&PathBuf>) -> DetectionResult {
71    let h = home.cloned().unwrap_or_else(|| PathBuf::from("."));
72    let config_dir = config_dir_for(loc, &h);
73    let settings_path = config_dir.join("settings.json");
74    let gemini_md = instructions_path(loc, &h);
75    let installed = config_dir.exists() || settings_path.exists() || gemini_md.exists();
76    let mut already_configured = false;
77    if settings_path.exists() {
78        let json = read_json_file(&settings_path);
79        if let Some(servers) = json.get("mcpServers").and_then(|v| v.as_object()) {
80            already_configured = servers.contains_key("telemaco");
81        }
82        if !already_configured {
83            // Not "a BeforeAgent hook exists": the user's own hooks live there
84            // too, and any one of them used to mark us as installed.
85            already_configured = json
86                .get("hooks")
87                .and_then(|h| h.get("BeforeAgent"))
88                .map_or(false, |v| text_has_telemaco_hook(&v.to_string()));
89        }
90    }
91    let hint = if already_configured {
92        "already configured".to_string()
93    } else if installed {
94        if loc.is_global() {
95            "detected".to_string()
96        } else {
97            let mut markers: Vec<String> = Vec::new();
98            if config_dir.exists() { markers.push(".gemini/".to_string()); }
99            if gemini_md.exists() {
100                if let Some(name) = gemini_md.file_name() {
101                    markers.push(name.to_string_lossy().to_string());
102                }
103            }
104            if markers.is_empty() { "detected".to_string() } else { markers.join(", ") }
105        }
106    } else {
107        String::new()
108    };
109    DetectionResult {
110        installed,
111        already_configured,
112        config_path: Some(settings_path),
113        hint,
114    }
115}

117pub fn install(loc: &Location, opts: &TargetInstallOptions, home: &PathBuf) -> TargetResult {
118    let mut out = Outcome::new(opts.dry_run);
119
120    let config_dir = config_dir_for(loc, home);
121    // The MCP entry and the hook share one file, so they share one write:
122    // upserting each in turn reported settings.json twice and rewrote it twice.
123    let settings_path = config_dir.join("settings.json");
124    let existed = settings_path.exists();
125    if let Some(mut settings_json) = out.load_json(&settings_path) {
126        let mut modified = match upsert_server_entry(
127            &mut settings_json,
128            "mcpServers",
129            "telemaco",
130            stdio_mcp_entry(&opts.binary_path, opts.stealth),
131        ) {
132            Ok(changed) => changed,
133            Err(why) => {
134                out.note(format!("{}: {}", settings_path.display(), why));
135                false
136            }
137        };
138
139        // Gemini CLI's prompt event is `BeforeAgent`, and it parses a command
140        // hook's stdout as JSON: "your script must not print any plain text to
141        // stdout other than the final JSON"
142        // (google-gemini/gemini-cli docs/hooks/reference.md).
143        if opts.prompt_hook
144            && add_prompt_hook_for_event(
145                &mut settings_json,
146                "BeforeAgent",
147                &prompt_hook_command_json(&opts.binary_path),
148            )
149        {
150            modified = true;
151        } else if !opts.prompt_hook
152            && remove_prompt_hook_for_event(&mut settings_json, "BeforeAgent")
153        {
154            // Reinstalling with the hook declined takes back the one an
155            // earlier install added.
156            modified = true;
157        }
158
159        if modified {
160            let action = if existed { Action::Updated } else { Action::Created };
161            out.write_json(&settings_path, &settings_json, action);
162        } else {
163            out.push(FileResult { path: settings_path.clone(), action: Action::Unchanged });
164        }
165    }
166
167    update_instructions(&mut out, &instructions_path(loc, home), opts.stealth);
168
169    // Project hooks are fingerprinted: "If a hook's name or command changes
170    // [...] it is treated as a new, untrusted hook and you will be warned
171    // before it executes" (docs/hooks/index.md).
172    if loc.folder().is_some() {
173        out.note(
174            "Gemini CLI warns you before running a project hook it has not seen before, and \
175             again whenever its command changes, so the prompt hook starts working after you \
176             accept it.",
177        );
178    }
179
180    out.finish(TargetId::Gemini)
181}

183pub fn uninstall(loc: &Location, home: &PathBuf, dry_run: bool) -> TargetResult {
184    let mut out = Outcome::new(dry_run);
185
186    let settings_path = config_dir_for(loc, home).join("settings.json");
187    if settings_path.exists() {
188        if let Some(mut settings_json) = out.load_json(&settings_path) {
189            let hook = remove_prompt_hook_for_event(&mut settings_json, "BeforeAgent");
190            let server = settings_json
191                .get_mut("mcpServers")
192                .and_then(|v| v.as_object_mut())
193                .map_or(false, |s| s.remove("telemaco").is_some());
194            if hook || server {
195                out.write_json_or_remove(&settings_path, &settings_json, Action::Updated);
196            }
197        }
198    }
199
200    // Both the configured name and the default: a settings change between
201    // install and uninstall must not strand the block in the old file.
202    for path in all_instructions_paths(loc, home) {
203        remove_instructions(&mut out, &path);
204    }
205
206    out.finish(TargetId::Gemini)
207}
