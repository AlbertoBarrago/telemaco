6/// Codex's home directory: `~/.codex` "unless you set `CODEX_HOME`"
7/// (learn.chatgpt.com/docs/agent-configuration/agents-md). With the variable
8/// set, everything we wrote under `~/.codex` was read by nothing.
9fn codex_home(home: &PathBuf) -> PathBuf {
10    home_env_var("CODEX_HOME").unwrap_or_else(|| home.join(".codex"))
11}
12
13/// True when this config layer already declares hooks inline.
14///
15/// Codex merges an inline `[hooks]` table with a `hooks.json` in the same
16/// layer and warns at startup, asking for one representation per layer
17/// (developers.openai.com/codex/hooks). We keep writing the JSON file, which
18/// is the form the other targets share, and say where the warning comes from.
19fn has_inline_hooks(content: &str) -> bool {
20    content.lines().any(|l| {
21        let t = l.trim();
22        t == "[hooks]" || t.starts_with("[hooks.") || t.starts_with("[[hooks.")
23    })
24}

26/// The instruction file Codex reads in a directory.
27///
28/// At every level Codex checks `AGENTS.override.md` first and falls back to
29/// `AGENTS.md` only when the override is absent: it "uses only the first
30/// non-empty file at this level", and a project directory contributes at most
31/// one file (learn.chatgpt.com/docs/agent-configuration/agents-md). A user
32/// with an override file therefore never saw the block written to `AGENTS.md`.
33fn instructions_file(dir: &Path) -> PathBuf {
34    agents_file_in(dir)
35}
36
37/// Both instruction files for a directory, for cleanup.
38fn all_instruction_files(dir: &Path) -> Vec<PathBuf> {
39    all_agents_files_in(dir)
40}

42pub fn detect(loc: &Location, home: Option<&PathBuf>) -> DetectionResult {
43    let (config_dir, toml_path, agents_md) = match loc {
44        Location::Global => {
45            let h = home.cloned().unwrap_or_else(|| PathBuf::from("."));
46            let dir = codex_home(&h);
47            (dir.clone(), dir.join("config.toml"), dir.join("AGENTS.md"))
48        }
49        Location::Folder(folder) => {
50            (folder.join(".codex"), folder.join(".codex").join("config.toml"), folder.join("AGENTS.md"))
51        }
52    };
53    // Not `|| agents_md.exists()`: AGENTS.md is the cross-tool convention, so
54    // most repositories have one and every one of them came back as a Codex
55    // install. Same reasoning as the `.pi` check in the Pi target.
56    //
57    // A `codex.toml` at the project root is not checked either: the config
58    // layers are `.codex/config.toml`, `~/.codex/config.toml` and
59    // `/etc/codex/config.toml`, and nothing reads a file by that name at a
60    // repository root (developers.openai.com/codex/config-basic).
61    let installed = config_dir.exists() || toml_path.exists();
62    let mut already_configured = false;
63    if toml_path.exists() {
64        if let Ok(content) = fs::read_to_string(&toml_path) {
65            already_configured = content.contains("[mcp_servers.telemaco]");
66        }
67    }
68    let hint = if already_configured {
69        "already configured".to_string()
70    } else if installed {
71        if loc.is_global() {
72            "detected".to_string()
73        } else {
74            let mut markers = Vec::new();
75            if config_dir.exists() { markers.push(".codex/"); }
76            if agents_md.exists() { markers.push("AGENTS.md"); }
77            if markers.is_empty() { "detected".to_string() } else { markers.join(", ") }
78        }
79    } else {
80        String::new()
81    };
82    DetectionResult {
83        installed,
84        already_configured,
85        config_path: Some(toml_path),
86        hint,
87    }
88}

90pub fn install(loc: &Location, opts: &TargetInstallOptions, home: &PathBuf) -> TargetResult {
91    let mut out = Outcome::new(opts.dry_run);
92
93    // 1. Config in ~/.codex/config.toml or <folder>/.codex/config.toml
94    let toml_path = match loc {
95        Location::Global => codex_home(home).join("config.toml"),
96        Location::Folder(folder) => folder.join(".codex").join("config.toml"),
97    };
98    let mcp_args = stdio_mcp_args(opts.stealth);
99    let toml_args_str = mcp_args
100        .iter()
101        .map(|a| format!("\"{}\"", a))
102        .collect::<Vec<_>>()
103        .join(", ");
104    let toml_existed = toml_path.exists();
105    if let Some(original) = out.load_text(&toml_path) {
106        let content = original.clone();
107        // A bare top-level `hooks` key is where inline hook *tables* live, so
108        // the `hooks = true` earlier versions wrote there handed Codex a
109        // boolean where it expects a table
110        // (developers.openai.com/codex/config-reference). Take it back.
111        let (content, legacy_removed) = remove_top_level_key_with_value(&content, "hooks", "true");
112        // `[features] hooks` is not ours to set. Hooks are on by default and
113        // that key is how a user turns them off
114        // (developers.openai.com/codex/hooks), so writing `true` pinned a
115        // default at best and overrode a deliberate `false` at worst. Say what
116        // is happening instead.
117        if toml_table_key_value(&content, "features", "hooks").as_deref() == Some("false") {
118            out.note(format!(
119                "{} sets [features] hooks = false, so Codex loads no hooks and the Telemaco \
120                 prompt hook stays idle until you remove that line.",
121                toml_path.display()
122            ));
123        }
124        if has_inline_hooks(&content) {
125            out.note(format!(
126                "{} already declares hooks inline. Codex merges those with the hooks.json \
127                 next to it and warns at startup; move one of the two if you want the \
128                 warning gone.",
129                toml_path.display()
130            ));
131        }
132        let flag_changed = legacy_removed;
133        let managed = [
134            ("command", format!("\"{}\"", opts.binary_path)),
135            ("args", format!("[{}]", toml_args_str)),
136        ];
137        let (new_content, action) = upsert_toml_table_keys(&content, "mcp_servers.telemaco", &managed);
138        if action != Action::Unchanged || flag_changed {
139            let effective_action = match (action, toml_existed) {
140                (Action::Unchanged, _) => Action::Updated,
141                (Action::Created, true) => Action::Updated,
142                (a, _) => a,
143            };
144            let new_content = with_line_ending(&original, &new_content);
145            out.write_text(&toml_path, &new_content, effective_action);
146        } else {
147            out.push(FileResult { path: toml_path, action: Action::Unchanged });
148        }
149    }
150
151    // 2. Hooks in ~/.codex/hooks.json or <folder>/.codex/hooks.json
152    let hooks_path = match loc {
153        Location::Global => codex_home(home).join("hooks.json"),
154        Location::Folder(folder) => folder.join(".codex").join("hooks.json"),
155    };
156    let hooks_existed = hooks_path.exists();
157    if let Some(mut hooks_json) = out.load_json(&hooks_path) {
158        if opts.prompt_hook && add_user_prompt_hook(&mut hooks_json, &prompt_hook_command(&opts.binary_path)) {
159            let action = if hooks_existed { Action::Updated } else { Action::Created };
160            out.write_json(&hooks_path, &hooks_json, action);
161        } else if !opts.prompt_hook && remove_user_prompt_hook(&mut hooks_json) {
162            // Reinstalling with the hook declined takes back the one an
163            // earlier install added, and takes the file with it when the hook
164            // was the only thing in it: this file exists only to hold the hook,
165            // so leaving `{"hooks":{"UserPromptSubmit":[]}}` behind would drop
166            // a config the user never wrote into their agent's directory. Same
167            // call the uninstall path uses.
168            out.write_json_or_remove(&hooks_path, &hooks_json, Action::Updated);
169        }
170    }
171
172    // 3. Instructions in ~/.codex/AGENTS.md or <folder>/AGENTS.md
173    let instructions_path = match loc {
174        Location::Global => instructions_file(&codex_home(home)),
175        Location::Folder(folder) => instructions_file(folder),
176    };
177    update_instructions(&mut out, &instructions_path, opts.stealth);
178
179    // Two review steps stand between this install and a hook that runs: a
180    // project `.codex/` layer loads only in a project you trust
181    // (developers.openai.com/codex/config-basic), and every non-managed hook
182    // is trusted by hash, so it is reviewed again whenever its command changes
183    // (developers.openai.com/codex/hooks).
184    if loc.folder().is_some() {
185        out.note(
186            "Codex reads a project's .codex/ layer only once you trust the project, so the \
187             MCP entry and the prompt hook start working after you answer that prompt.",
188        );
189    }
190    out.note(
191        "Codex asks you to review the prompt hook before it runs, and again whenever its \
192         command changes, because it records trust against the hook's exact definition.",
193    );
194
195    out.finish(TargetId::Codex)
196}

198pub fn uninstall(loc: &Location, home: &PathBuf, dry_run: bool) -> TargetResult {
199    let mut out = Outcome::new(dry_run);
200
201    // Hooks first: whether anything still needs `hooks = true` depends on
202    // whether the hooks file survives.
203    let hooks_path = match loc {
204        Location::Global => codex_home(home).join("hooks.json"),
205        Location::Folder(folder) => folder.join(".codex").join("hooks.json"),
206    };
207    // Whether a hook file is left decides whether `features.hooks` still has a
208    // job. Under --dry-run nothing is written, so asking the filesystem again
209    // answered for the run that did not happen and the plan came out wrong.
210    let mut hooks_remain = hooks_path.exists();
211    if hooks_path.exists() {
212        if let Some(mut hooks_json) = out.load_json(&hooks_path) {
213            if remove_user_prompt_hook(&mut hooks_json) {
214                let mut pruned = hooks_json.clone();
215                prune_installer_scaffolding(&mut pruned);
216                hooks_remain = !pruned.as_object().map_or(false, |o| o.is_empty());
217                out.write_json_or_remove(&hooks_path, &hooks_json, Action::Updated);
218            }
219        }
220    }
221
222    let toml_path = match loc {
223        Location::Global => codex_home(home).join("config.toml"),
224        Location::Folder(folder) => folder.join(".codex").join("config.toml"),
225    };
226    if toml_path.exists() {
227        if let Some(original) = out.load_text(&toml_path) {
228            let (content, table_action) = remove_toml_table(&original, "mcp_servers.telemaco");
229            // `hooks = true` was ours; take it back only once no hook file is
230            // left that would need it.
231            let (content, flag_removed) = if hooks_remain {
232                (content, false)
233            } else {
234                let (c, legacy) = remove_top_level_key_with_value(&content, "hooks", "true");
235                // Only the `true` an older Telemaco wrote comes out. A `false`
236                // is the user's own switch and stays.
237                let ours = toml_table_key_value(&c, "features", "hooks").as_deref() == Some("true");
238                let (c, feature) = if ours {
239                    remove_toml_table_key(&c, "features", "hooks")
240                } else {
241                    (c, false)
242                };
243                (c, legacy || feature)
244            };
245            if table_action == Action::Removed || flag_removed {
246                if content.trim().is_empty() {
247                    out.remove_config_file(&toml_path, "");
248                } else {
249                    let rebuilt = with_line_ending(&original, &content);
250                    out.write_text(&toml_path, &rebuilt, Action::Removed);
251                }
252            }
253        }
254    }
255
256    let dir = match loc {
257        Location::Global => codex_home(home),
258        Location::Folder(folder) => folder.clone(),
259    };
260    for path in all_instruction_files(&dir) {
261        if path.exists() {
262            remove_instructions(&mut out, &path);
263        }
264    }
265
266    out.finish(TargetId::Codex)
267}
