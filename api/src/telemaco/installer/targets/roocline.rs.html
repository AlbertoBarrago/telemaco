6/// Project-level MCP config.
7///
8/// Roo Code reads `.roo/mcp.json` at the project root (docs: "Project-level
9/// Configuration: Defined in a `.roo/mcp.json` file within your project's root
10/// directory"). We used to write `.vscode/mcp.json`, which is VS Code's own
11/// native MCP file and is read by neither Roo Code nor Cline.
12fn project_mcp_path(folder: &Path) -> PathBuf {
13    folder.join(".roo").join("mcp.json")
14}

16/// VS Code's per-user directory, the parent of `globalStorage`.
17fn vscode_user_dir(home: &PathBuf) -> PathBuf {
18    #[cfg(target_os = "macos")]
19    {
20        home.join("Library/Application Support/Code/User")
21    }
22    #[cfg(not(target_os = "macos"))]
23    {
24        home.join(".config/Code/User")
25    }
26}

28fn get_base_storage(home: &PathBuf) -> PathBuf {
29    vscode_user_dir(home).join("globalStorage")
30}

32/// Roo Code's storage base.
33///
34/// Normally the extension's own `globalStorage` directory, but the
35/// `roo-cline.customStoragePath` VS Code setting replaces it wholesale
36/// (`getStorageBasePath`, Roo-Code `src/utils/storage.ts`). With that setting
37/// pointing elsewhere, everything written under `globalStorage` is read by
38/// nothing, the same trap `CODEX_HOME` and `HERMES_HOME` were.
39fn roo_storage_base(home: &PathBuf) -> PathBuf {
40    let settings = vscode_user_dir(home).join("settings.json");
41    if settings.exists() {
42        if let Some(custom) = read_json_file(&settings)
43            .get("roo-cline.customStoragePath")
44            .and_then(|v| v.as_str())
45        {
46            if !custom.trim().is_empty() {
47                return PathBuf::from(custom.trim());
48            }
49        }
50    }
51    get_base_storage(home).join("rooveterinaryinc.roo-cline")
52}

60/// The MCP files read at the user level, each with the directory that proves
61/// its product is actually installed.
62///
63/// The marker is never a directory this installer creates: `~/.cline/rules/`
64/// is ours, `~/.cline/data/` is Cline's, and guarding on the wrong one makes
65/// the second install differ from the first.
66///
67/// - Roo Code: `<storage base>/settings/mcp_settings.json`
68///   (`GlobalFileNames.mcpSettings`, Roo-Code `src/shared/globalFileNames.ts`).
69///   Not `cline_mcp_settings.json`: that is the name Roo inherited from the
70///   fork and stopped reading.
71/// - Cline, VS Code extension: `<globalStorage>/saoudrizwan.claude-dev/settings/cline_mcp_settings.json`.
72/// - Cline, CLI and SDK: `~/.cline/data/settings/cline_mcp_settings.json`.
73///   The MCP page still documents `~/.cline/mcp.json`; the shipped CLI
74///   (`cline mcp install`) writes the path above and reads only that one.
75fn global_mcp_targets(home: &PathBuf) -> Vec<(PathBuf, PathBuf)> {
76    let roo_base = roo_storage_base(home);
77    let cline_extension = get_base_storage(home).join("saoudrizwan.claude-dev");
78    let cline_root = cline_home(home);
79    vec![
80        (roo_base.join("settings").join("mcp_settings.json"), roo_base),
81        (
82            cline_extension.join("settings").join("cline_mcp_settings.json"),
83            cline_extension,
84        ),
85        (
86            cline_root
87                .join("data")
88                .join("settings")
89                .join("cline_mcp_settings.json"),
90            cline_root.join("data"),
91        ),
92    ]
93}

106/// Rules files, one per product: Roo Code and Cline read different paths, and
107/// neither of them reads the `~/.clinerules` we used to write.
108///
109/// Roo Code: `.roo/rules/` in a workspace, `~/.roo/rules/` globally
110/// (docs.roocode.com/features/custom-instructions).
111///
112/// Cline: `.clinerules/` in a workspace, and globally both `~/.cline/rules/`
113/// (CLI and SDK) and `~/Documents/Cline/Rules` (the VS Code extension's
114/// documented directory, also read by the CLI). Cline keys a rule by its file
115/// stem and loads the first one it finds, so the same `telemaco.md` in both
116/// directories is one rule, not two.
117fn instructions_paths(loc: &Location, home: &PathBuf) -> Vec<PathBuf> {
118    match loc {
119        Location::Global => vec![
120            home.join(".roo").join("rules").join("telemaco.md"),
121            cline_home(home).join("rules").join("telemaco.md"),
122            home.join("Documents")
123                .join("Cline")
124                .join("Rules")
125                .join("telemaco.md"),
126        ],
127        Location::Folder(folder) => vec![
128            folder.join(".roo").join("rules").join("telemaco.md"),
129            cline_workspace_rules(folder),
130        ],
131    }
132}

134/// `.clinerules` is a directory of rule files, but plenty of projects still
135/// carry the older single file. Creating the directory on top of it would fail,
136/// so an existing file is updated where it is.
137fn cline_workspace_rules(folder: &Path) -> PathBuf {
138    let base = folder.join(".clinerules");
139    if base.is_file() {
140        base
141    } else {
142        base.join("telemaco.md")
143    }
144}

146/// Rules paths earlier versions wrote, cleaned up on the way out.
147fn legacy_instructions(loc: &Location, home: &PathBuf) -> Vec<PathBuf> {
148    match loc {
149        Location::Global => vec![home.join(".clinerules")],
150        // The workspace file is already the target of `cline_workspace_rules`.
151        Location::Folder(_) => Vec::new(),

155pub fn detect(loc: &Location, home: Option<&PathBuf>) -> DetectionResult {
156    let h = home.cloned().unwrap_or_else(|| PathBuf::from("."));
157    let (chosen, installed) = match loc {
158        Location::Global => {
159            let targets = global_mcp_targets(&h);
160            let chosen = targets
161                .iter()
162                .map(|(p, _)| p)
163                .find(|p| p.exists())
164                .cloned()
165                .or_else(|| targets.first().map(|(p, _)| p.clone()));
166            // Not `|| globalStorage.exists()`: every VS Code user has that
167            // directory, and it says nothing about these extensions. The
168            // markers are each product's own directory, so a Cline CLI that
169            // has never had an MCP server still counts as installed.
170            let installed = targets
171                .iter()
172                .any(|(path, marker)| path.exists() || marker.exists())
173                || legacy_global_mcp_paths(&h).iter().any(|p| p.exists());
174            (chosen, installed)
175        }
176        Location::Folder(folder) => {
177            let mcp_path = project_mcp_path(folder);
178            let installed = folder.join(".clinerules").exists()
179                || folder.join(".roomodes").exists()
180                || folder.join(".roo").exists()
181                || folder.join(".cline").exists()
182                || mcp_path.exists();
183            (Some(mcp_path), installed)
184        }
185    };
186    let mut already_configured = false;
187    if let Some(ref p) = chosen {
188        if p.exists() {
189            let json = read_json_file(p);
190            if let Some(servers) = json.get("mcpServers").and_then(|v| v.as_object()) {
191                already_configured = servers.contains_key("telemaco");
192            }
193        }
194    }
195    if !already_configured {
196        // The rules files are the other half of the install, and for a folder
197        // without an MCP file they are the whole of it.
198        already_configured = instructions_paths(loc, &h).iter().any(|p| {
199            std::fs::read_to_string(p).map_or(false, |c| c.contains("<!-- TELEMACO_START -->"))
200        });
201    }
202    let hint = if already_configured {
203        "already configured".to_string()
204    } else if installed {
205        if loc.is_global() {
206            "detected".to_string()
207        } else {
208            let mut markers = Vec::new();
209            if let Location::Folder(folder) = loc {
210                if folder.join(".clinerules").exists() { markers.push(".clinerules"); }
211                if folder.join(".roomodes").exists() { markers.push(".roomodes"); }
212                if folder.join(".roo").exists() { markers.push(".roo/"); }
213                if folder.join(".cline").exists() { markers.push(".cline/"); }
214            }
215            if markers.is_empty() { "detected".to_string() } else { markers.join(", ") }
216        }
217    } else {
218        String::new()
219    };
220    DetectionResult {
221        installed,
222        already_configured,
223        config_path: chosen,
224        hint,
225    }
226}

229pub fn mcp_entry(binary_path: &str, stealth: bool) -> serde_json::Value {
230    json!({
231        "command": binary_path,
232        "args": stdio_mcp_args(stealth),

238pub fn install(loc: &Location, opts: &TargetInstallOptions, home: &PathBuf) -> TargetResult {
239    let mut out = Outcome::new(opts.dry_run);
240
241    let target_entry = mcp_entry(&opts.binary_path, opts.stealth);
242
243    match loc {
244        Location::Global => {
245            for (path, marker) in global_mcp_targets(home) {
246                if marker.exists() || path.exists() {
247                    upsert_mcp_server(&mut out, &path, "telemaco", target_entry.clone());
248                }
249            }
250        }
251        // Only Roo Code reads a project MCP file. Cline's `.cline/` holds
252        // rules, hooks and plugins; its servers come from the user-level file
253        // whatever directory the session runs in.
254        Location::Folder(folder) => {
255            upsert_mcp_server(&mut out, &project_mcp_path(folder), "telemaco", target_entry);
256        }
257    }
258
259    for path in instructions_paths(loc, home) {
260        update_instructions(&mut out, &path, opts.stealth);
261    }
262
263    out.finish(TargetId::RooCline)
264}

266pub fn uninstall(loc: &Location, home: &PathBuf, dry_run: bool) -> TargetResult {
267    let mut out = Outcome::new(dry_run);
268
269    let paths = match loc {
270        Location::Global => global_mcp_targets(home)
271            .into_iter()
272            .map(|(path, _)| path)
273            .chain(legacy_global_mcp_paths(home))
274            .collect(),
275        Location::Folder(folder) => vec![
276            project_mcp_path(folder),
277            // Earlier builds wrote here; clean it up on the way out.
278            folder.join(".vscode").join("mcp.json"),
279        ],
280    };
281
282    for path in paths {
283        remove_mcp_server(&mut out, &path, "telemaco");
284    }
285
286    for path in instructions_paths(loc, home)
287        .into_iter()
288        .chain(legacy_instructions(loc, home))
289    {
290        remove_instructions(&mut out, &path);
291    }
292
293    out.finish(TargetId::RooCline)
