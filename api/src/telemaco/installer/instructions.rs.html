9/// The block written into an agent's instructions file.
10///
11/// Kept deliberately short: it sits in the agent's context on every single
12/// turn. The MCP server repeats the same directive in its `initialize`
13/// response, and the prompt hook injects a fuller version only on prompts that
14/// actually look web-bound, so this copy only has to carry what an agent needs
15/// before it has touched either.
16pub fn get_instructions_block(stealth_default: bool) -> String {
17    let stealth_flag = if stealth_default { " --stealth" } else { "" };
18    format!(
19r#"{TELEMACO_START_MARKER}
20## Web access: use Telemaco
21
22For any web work (visiting a URL, reading online docs, inspecting or scraping a
23page, searching), use Telemaco instead of built-in web search or `curl`/`wget`.
24Telemaco runs a real browser: V8, a full DOM, and stealth against bot detection,
25so it sees pages that a plain fetch cannot.
26
27MCP tools, when connected: `browser_navigate` (open a URL), `browser_markdown`
28(page as clean Markdown), `browser_snapshot` (DOM and interactive elements),
29`browser_extract` (CSS selectors), `browser_search` (find in page),
30`browser_click` / `browser_fill` / `browser_type` (forms).
31
32CLI, when they are not:
33- `telemaco{stealth_flag} fetch <url> --dump markdown` (also `text`, `html`)
34- `telemaco{stealth_flag} fetch <url> --screenshot <path.png>` (needs a build
35  with the `render` feature; without it the command says so and exits)
36
37Rules:
381. To search, navigate to a search engine's results page (for example
39   `https://duckduckgo.com/html/?q=<query>`), read it with `browser_markdown`,
40   then open the target page.
412. Loopback and RFC1918 are blocked by default (SSRF guard). Pass
42   `--allow-private-network` only when deliberately testing a local URL.
433. State the URL before visiting it (`Navigating to: <url>`) and cite every
44   source URL you consulted.
45{TELEMACO_END_MARKER}"#
46    )
47}

49#[derive(Debug, Clone, Copy, PartialEq, Eq)]
50pub enum Action {
51    Created,
52    Updated,
53    Unchanged,
54    Removed,
55    NotFound,
56}

58/// What `replace_or_append_marked_section` would do, without doing it.
59pub fn plan_marked_section(file_path: &Path, block: &str) -> std::io::Result<Action> {
60    if !file_path.exists() {
61        return Ok(Action::Created);
62    }
63    let content = fs::read_to_string(file_path)?;
64    match locate_marked_section(&content) {
65        MarkedSection::Malformed(why) => Err(malformed(why)),
66        // Compared in LF: a CRLF file holds the same block with `\r\n`, and
67        // calling that a change rewrote the file on every single install.
68        MarkedSection::One(s, e) if normalize(&content[s..e]) == block => Ok(Action::Unchanged),
69        _ => Ok(Action::Updated),
70    }
71}

73pub fn replace_or_append_marked_section(file_path: &Path, block: &str) -> std::io::Result<Action> {
74    if !file_path.exists() {
75        if let Some(parent) = file_path.parent() {
76            fs::create_dir_all(parent)?;
77        }
78        let content = format!("{}\n", block);
79        crate::installer::json_utils::atomic_write_file(file_path, &content)?;
80        return Ok(Action::Created);
81    }
82
83    let content = fs::read_to_string(file_path)?;
84
85    match locate_marked_section(&content) {
86        MarkedSection::Malformed(why) => return Err(malformed(why)),
87        MarkedSection::One(s, e) => {
88            if normalize(&content[s..e]) == block {
89                return Ok(Action::Unchanged);
90            }
91            let new_content = format!("{}{}{}", &content[..s], block, &content[e..]);
92            let new_content = with_line_ending(&content, &new_content);
93            crate::installer::json_utils::atomic_write_file(file_path, &new_content)?;
94            return Ok(Action::Updated);
95        }
96        MarkedSection::Absent => {}
97    }
98
99    // Append to existing content
100    let trimmed = content.trim_end();
101    let sep = if trimmed.is_empty() { "" } else { "\n\n" };
102    let new_content = format!("{}{}{}\n", trimmed, sep, block);
103    let new_content = with_line_ending(&content, &new_content);
104    crate::installer::json_utils::atomic_write_file(file_path, &new_content)?;
105    Ok(Action::Updated)
106}

108/// What `remove_marked_section` would do, without doing it.
109pub fn plan_remove_marked_section(file_path: &Path) -> std::io::Result<Action> {
110    if !file_path.exists() {
111        return Ok(Action::NotFound);
112    }
113    let content = fs::read_to_string(file_path)?;
114    match locate_marked_section(&content) {
115        MarkedSection::Malformed(why) => Err(malformed(why)),
116        MarkedSection::One(_, _) => Ok(Action::Removed),
117        MarkedSection::Absent => Ok(Action::NotFound),
118    }
119}

121pub fn remove_marked_section(file_path: &Path) -> std::io::Result<Action> {
122    if !file_path.exists() {
123        return Ok(Action::NotFound);
124    }
125
126    let content = fs::read_to_string(file_path)?;
127
128    let (s, full_end) = match locate_marked_section(&content) {
129        MarkedSection::Malformed(why) => return Err(malformed(why)),
130        MarkedSection::Absent => return Ok(Action::NotFound),
131        MarkedSection::One(s, e) => (s, e),
132    };
133
134    let before = content[..s].trim_end();
135    let after = content[full_end..].trim_start();
136    let joined = if before.is_empty() {
137        after.to_string()
138    } else if after.is_empty() {
139        before.to_string()
140    } else {
141        format!("{}\n\n{}", before, after)
142    };
143
144    let symlinked = fs::symlink_metadata(file_path)
145        .map_or(false, |m| m.file_type().is_symlink());
146    if joined.trim().is_empty() && !symlinked {
147        // Nothing of the user's left: the file was ours to begin with. A
148        // symlink is the user's dotfiles setup, so it is emptied instead.
149        if let Err(e) = fs::remove_file(file_path) {
150            return Err(e);
151        }
152    } else {
153        let new_content = format!("{}\n", joined.trim());
154        let new_content = with_line_ending(&content, &new_content);
155        crate::installer::json_utils::atomic_write_file(file_path, &new_content)?;
156    }
157    Ok(Action::Removed)
158}

160/// What the Telemaco markers in a file look like.
161enum MarkedSection {
162    /// No markers: the block gets appended.
163    Absent,
164    /// Exactly one well-formed block, as a byte range.
165    One(usize, usize),
166    /// Anything else. Guessing here is how the user's text gets eaten: a stray
167    /// start marker with no end used to pair up with the end of a block
168    /// appended later, and everything in between was replaced.
169    Malformed(String),
170}
171
172fn locate_marked_section(content: &str) -> MarkedSection {
173    let starts = content.matches(TELEMACO_START_MARKER).count();
174    let ends = content.matches(TELEMACO_END_MARKER).count();
175
176    match (starts, ends) {
177        (0, 0) => MarkedSection::Absent,
178        (1, 1) => {
179            let start = content.find(TELEMACO_START_MARKER).expect("counted one");
180            let end = content.find(TELEMACO_END_MARKER).expect("counted one");
181            if end < start {
182                MarkedSection::Malformed(
183                    "the Telemaco end marker comes before the start marker".to_string(),
184                )
185            } else {
186                MarkedSection::One(start, end + TELEMACO_END_MARKER.len())
187            }
188        }
189        _ => MarkedSection::Malformed(format!(
190            "found {} Telemaco start and {} end markers, expected one of each",
191            starts, ends
192        )),
193    }
194}

196fn malformed(msg: String) -> std::io::Error {
197    std::io::Error::new(
198        std::io::ErrorKind::InvalidData,
199        format!("{}; fix or remove them by hand and re-run", msg),
200    )
