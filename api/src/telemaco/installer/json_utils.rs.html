6/// Appended to a config file before we rewrite it in a way that loses
7/// formatting, so the original stays recoverable. Same suffix the shell
8/// installer used.
9const BACKUP_SUFFIX: &str = ".telemaco-backup";
10
11/// The file a write should actually land on.
12///
13/// Config files are routinely symlinks into a dotfiles repo (stow, chezmoi,
14/// yadm). Renaming over the link would replace it with a regular file and
15/// orphan the copy the user actually edits, so the link is followed first.
16fn resolve_write_target(file_path: &Path) -> PathBuf {
17    match fs::symlink_metadata(file_path) {
18        Ok(meta) if meta.file_type().is_symlink() => {
19            fs::canonicalize(file_path).unwrap_or_else(|_| file_path.to_path_buf())
20        }
21        _ => file_path.to_path_buf(),
22    }
23}
24
25pub fn atomic_write_file(file_path: &Path, content: &str) -> std::io::Result<()> {
26    let target = resolve_write_target(file_path);
27
28    if let Some(parent) = target.parent() {
29        if !parent.exists() {
30            fs::create_dir_all(parent)?;
31        }
32    }
33
34    let pid = std::process::id();
35    let tmp_path = target.with_extension(format!("tmp.{}", pid));
36    fs::write(&tmp_path, content)?;
37
38    // A fresh temp file is created with the process umask. Carry the original
39    // mode over instead: these configs hold credentials and are often 0600.
40    if let Ok(meta) = fs::metadata(&target) {
41        let _ = fs::set_permissions(&tmp_path, meta.permissions());
42    }
43
44    if let Err(e) = fs::rename(&tmp_path, &target) {
45        let _ = fs::remove_file(&tmp_path);
46        return Err(e);
47    }
48    Ok(())
49}

51/// How a JSON config file parsed.
52#[derive(Debug, Clone, Copy, PartialEq, Eq)]
53pub enum JsonSource {
54    /// Absent or empty: the caller starts from an empty object.
55    Missing,
56    /// Strict JSON. Rewriting it preserves everything but the formatting.
57    Strict,
58    /// Parsed only after stripping comments and trailing commas, which VS Code,
59    /// Cursor and Zed all accept in their config files. Rewriting it as strict
60    /// JSON drops those comments, so the file must be backed up first.
61    Jsonc,
62}
63
64#[derive(Debug)]
65pub struct JsonConfig {
66    pub value: Value,
67    pub source: JsonSource,
68}

70/// Reads a JSON config for inspection only. Never fails: detection has nothing
71/// to lose from an unreadable file and never writes one back.
72pub fn read_json_file(file_path: &Path) -> Value {
73    match parse_json_file(file_path) {
74        Ok(cfg) if cfg.value.is_object() => cfg.value,
75        _ => empty_object(),
76    }
77}

79/// Reads a JSON config that is about to be modified.
80///
81/// `Err` means the file exists but cannot be parsed, or does not hold an
82/// object. The caller must then leave it alone: overwriting it would discard
83/// the user's other MCP servers and editor settings.
84pub fn read_json_for_update(file_path: &Path) -> Result<JsonConfig, String> {
85    let cfg = parse_json_file(file_path)?;
86    if !cfg.value.is_object() {
87        return Err(format!(
88            "{}: top level is not a JSON object; leaving it alone",
89            file_path.display()
90        ));
91    }
92    Ok(cfg)
93}

95/// Where `backup_file` puts its copy.
96pub fn backup_path(file_path: &Path) -> PathBuf {
97    let mut name = file_path.as_os_str().to_os_string();
98    name.push(BACKUP_SUFFIX);
99    PathBuf::from(name)
100}

102/// Copies a file next to itself before a rewrite that would lose content.
103pub fn backup_file(file_path: &Path) -> Result<PathBuf, String> {
104    let backup = backup_path(file_path);
105    fs::copy(file_path, &backup)
106        .map_err(|e| format!("Could not back up {}: {}", file_path.display(), e))?;
107    Ok(backup)
108}

110pub fn write_json_file(file_path: &Path, data: &Value) -> Result<(), String> {
111    let mut json_str = serde_json::to_string_pretty(data)
112        .map_err(|e| format!("Could not serialize JSON for {}: {}", file_path.display(), e))?;
113    json_str.push('\n');
114    // A config the user keeps with CRLF stays that way. Serializing always
115    // emits LF, which turns adding one server into a whole-file diff for
116    // anyone on Windows or with core.autocrlf. Newlines inside JSON strings
117    // are escaped, so only the structural ones are touched.
118    if let Ok(original) = fs::read_to_string(file_path) {
119        json_str = with_line_ending(&original, &json_str);
120    }
121    atomic_write_file(file_path, &json_str)
122        .map_err(|e| format!("Could not write {}: {}", file_path.display(), e))
123}

125pub fn json_deep_equal(a: &Value, b: &Value) -> bool {
126    match (a, b) {
127        (Value::Null, Value::Null) => true,
128        (Value::Bool(a), Value::Bool(b)) => a == b,
129        (Value::Number(a), Value::Number(b)) => a == b,
130        (Value::String(a), Value::String(b)) => a == b,
131        (Value::Array(a), Value::Array(b)) => {
132            if a.len() != b.len() {
133                return false;
134            }
135            a.iter().zip(b.iter()).all(|(x, y)| json_deep_equal(x, y))
136        }
137        (Value::Object(a), Value::Object(b)) => {
138            if a.len() != b.len() {
139                return false;
140            }
141            for (k, v_a) in a {
142                match b.get(k) {
143                    Some(v_b) => {
144                        if !json_deep_equal(v_a, v_b) {
145                            return false;
146                        }
147                    }
148                    None => return false,
149                }
150            }
151            true
152        }
153        _ => false,
154    }
155}

161fn parse_json_file(file_path: &Path) -> Result<JsonConfig, String> {
162    if !file_path.exists() {
163        return Ok(JsonConfig { value: empty_object(), source: JsonSource::Missing });
164    }
165
166    let text = fs::read_to_string(file_path)
167        .map_err(|e| format!("Could not read {}: {}", file_path.display(), e))?;
168    // Editors on Windows leave a UTF-8 BOM in front; serde_json rejects it and
169    // the file would be refused as malformed.
170    let text = text.strip_prefix('\u{feff}').unwrap_or(&text).to_string();
171    if text.trim().is_empty() {
172        return Ok(JsonConfig { value: empty_object(), source: JsonSource::Missing });
173    }
174
175    match serde_json::from_str(&text) {
176        Ok(value) => Ok(JsonConfig { value, source: JsonSource::Strict }),
177        Err(strict_err) => match serde_json::from_str(&strip_jsonc(&text)) {
178            Ok(value) => Ok(JsonConfig { value, source: JsonSource::Jsonc }),
179            Err(_) => Err(format!(
180                "{} is not valid JSON ({}); leaving it alone",
181                file_path.display(),
182                strict_err
183            )),
184        },
185    }
186}

188/// Strips `//` and `/* */` comments and trailing commas, leaving string
189/// literals untouched. Newlines inside comments are kept so parse errors still
190/// point at the right line.
191fn strip_jsonc(input: &str) -> String {
192    let mut out = String::with_capacity(input.len());
193    let mut chars = input.chars().peekable();
194    let mut in_string = false;
195    let mut escaped = false;
196
197    while let Some(c) = chars.next() {
198        if in_string {
199            out.push(c);
200            if escaped {
201                escaped = false;
202            } else if c == '\\' {
203                escaped = true;
204            } else if c == '"' {
205                in_string = false;
206            }
207            continue;
208        }
209
210        match c {
211            '"' => {
212                in_string = true;
213                out.push(c);
214            }
215            '/' => match chars.peek().copied() {
216                Some('/') => {
217                    for c in chars.by_ref() {
218                        if c == '\n' {
219                            out.push('\n');
220                            break;
221                        }
222                    }
223                }
224                Some('*') => {
225                    chars.next();
226                    let mut prev = '\0';
227                    for c in chars.by_ref() {
228                        if prev == '*' && c == '/' {
229                            break;
230                        }
231                        if c == '\n' {
232                            out.push('\n');
233                        }
234                        prev = c;
235                    }
236                }
237                _ => out.push(c),
238            },
239            _ => out.push(c),
240        }
241    }
242
243    strip_trailing_commas(&out)
244}

246fn strip_trailing_commas(input: &str) -> String {
247    let mut out = String::with_capacity(input.len());
248    let mut in_string = false;
249    let mut escaped = false;
250    // Byte offset in `out` of the last comma that is still only followed by
251    // whitespace; dropped if the next real character closes the container.
252    let mut pending_comma: Option<usize> = None;
253
254    for c in input.chars() {
255        if in_string {
256            out.push(c);
257            if escaped {
258                escaped = false;
259            } else if c == '\\' {
260                escaped = true;
261            } else if c == '"' {
262                in_string = false;
263            }
264            continue;
265        }
266
267        match c {
268            '"' => {
269                pending_comma = None;
270                in_string = true;
271                out.push(c);
272            }
273            ',' => {
274                pending_comma = Some(out.len());
275                out.push(c);
276            }
277            ']' | '}' => {
278                if let Some(idx) = pending_comma.take() {
279                    out.remove(idx);
280                }
281                out.push(c);
282            }
283            _ if c.is_whitespace() => out.push(c),
284            _ => {
285                pending_comma = None;
286                out.push(c);
287            }
288        }
289    }
290
291    out
292}

301    fn temp_path(name: &str) -> PathBuf {
302        let id = COUNTER.fetch_add(1, Ordering::SeqCst);
303        std::env::temp_dir().join(format!("telemaco_json_{}_{}_{}", std::process::id(), id, name))
