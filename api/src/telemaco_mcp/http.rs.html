9/// Hard cap on a single MCP request body. The client-supplied `Content-Length`
10/// is used to pre-size the read buffer; without a ceiling a request advertising
11/// e.g. `Content-Length: 4294967296` makes the server allocate and zero-fill
12/// that many bytes before reading any body — an unauthenticated OOM/DoS. 16 MiB
13/// is far above any real JSON-RPC tool call.
14const MAX_BODY_BYTES: usize = 16 * 1024 * 1024;
15
16/// Maximum time allowed to receive one complete HTTP request (request line,
17/// headers, and body). The deadline is shared across all reads so a client
18/// cannot keep the sequential MCP server occupied by slowly dribbling data.
19const REQUEST_READ_TIMEOUT: Duration = Duration::from_secs(30);
20
21enum RequestBody {
22    NotRead,
23    MissingLength,
24    TooLarge,
25    Read(Vec<u8>),
26}

28struct HttpRequest {
29    method: String,
30    path: String,
31    accept_sse: bool,
32    keep_alive: bool,
33    origin: Option<String>,
34    body: RequestBody,
35}

37enum RequestRead {
38    Closed,
39    Invalid,
40    Request(HttpRequest),
41}

43async fn read_request(
44    reader: &mut (impl AsyncBufRead + Unpin),
45    allowed_origins: Option<&str>,
46) -> Result<RequestRead> {
47    let mut request_line = String::new();
48    if reader.read_line(&mut request_line).await? == 0 {
49        return Ok(RequestRead::Closed);
50    }
51    let request_line = request_line.trim();
52    if request_line.is_empty() {
53        return Ok(RequestRead::Closed);
54    }
55
56    let parts: Vec<&str> = request_line.splitn(3, ' ').collect();
57    if parts.len() < 3 {
58        return Ok(RequestRead::Invalid);
59    }
60    let method = parts[0].to_string();
61    let path = parts[1].to_string();
62
63    let mut content_length: Option<usize> = None;
64    let mut accept_sse = false;
65    let mut keep_alive = false;
66    let mut origin: Option<String> = None;
67
68    loop {
69        let mut line = String::new();
70        reader.read_line(&mut line).await?;
71        let trimmed = line.trim_end_matches("\r\n").trim_end_matches('\n');
72        if trimmed.is_empty() {
73            break;
74        }
75        let lower = trimmed.to_lowercase();
76        if let Some(v) = lower.strip_prefix("content-length: ") {
77            content_length = v.trim().parse().ok();
78        }
79        if lower.starts_with("origin:") {
80            if let Some(idx) = trimmed.find(':') {
81                origin = Some(trimmed[idx + 1..].trim().to_string());
82            }
83        }
84        if lower.contains("text/event-stream") {
85            accept_sse = true;
86        }
87        if lower.starts_with("connection: ") && lower.contains("keep-alive") {
88            keep_alive = true;
89        }
90    }
91
92    // Only consume a body for a POST that can reach the MCP route. Invalid
93    // paths and forbidden origins retain the existing early-response behavior.
94    let body = if method == "POST"
95        && path == "/mcp"
96        && origin_allowed(origin.as_deref(), allowed_origins)
97    {
98        match content_length {
99            None => RequestBody::MissingLength,
100            // Reject the client-supplied size before allocating the body.
101            Some(len) if len > MAX_BODY_BYTES => RequestBody::TooLarge,
102            Some(len) => {
103                let mut body = vec![0u8; len];
104                reader.read_exact(&mut body).await?;
105                RequestBody::Read(body)
106            }
107        }
108    } else {
109        RequestBody::NotRead
110    };
111
112    Ok(RequestRead::Request(HttpRequest {
113        method,
114        path,
115        accept_sse,
116        keep_alive,
117        origin,
118        body,
119    }))
120}

122async fn read_request_with_timeout(
123    reader: &mut (impl AsyncBufRead + Unpin),
124    allowed_origins: Option<&str>,
125    timeout: Duration,
126) -> Result<RequestRead> {
127    tokio::time::timeout(timeout, read_request(reader, allowed_origins))
128        .await
129        .map_err(|_| anyhow::anyhow!("request read timed out"))?
130}

132/// Origin allowlist for browser callers, read from `TELEMACO_MCP_ALLOWED_ORIGINS`
133/// (comma-separated). Unset/empty → permissive (unchanged `*`) so hosted
134/// dashboards keep working (issue #175).
135fn allowed_origins_env() -> Option<String> {
136    std::env::var("TELEMACO_MCP_ALLOWED_ORIGINS")
137        .ok()
138        .filter(|s| !s.trim().is_empty())
139}
140
141/// Whether a request's `Origin` is permitted. A request with no `Origin`
142/// (native, non-browser MCP clients) is always allowed — the same-origin
143/// policy only constrains browser callers. When an allowlist is configured, a
144/// browser `Origin` must match one of its entries (case-insensitive); this
145/// stops a malicious local web page from driving the loopback MCP port.
146fn origin_allowed(origin: Option<&str>, allowlist: Option<&str>) -> bool {
147    match allowlist {
148        None => true,
149        Some(list) => match origin {
150            None => true,
151            Some(o) => {
152                let o = o.trim();
153                list.split(',')
154                    .map(str::trim)
155                    .any(|a| !a.is_empty() && a.eq_ignore_ascii_case(o))
156            }
157        },
158    }
159}

161/// CORS `Access-Control-Allow-Origin` value for a response. With no allowlist
162/// we keep the permissive `*` (issue #175). With an allowlist the request's
163/// origin has already passed `origin_allowed`, so echo it back plus `Vary:
164/// Origin` instead of advertising `*`; a native client with no `Origin` needs
165/// no CORS header at all.
166fn cors_header(origin: Option<&str>, allowlist: Option<&str>) -> String {
167    match allowlist {
168        None => "Access-Control-Allow-Origin: *\r\n".to_string(),
169        Some(_) => match origin {
170            Some(o) => format!("Access-Control-Allow-Origin: {o}\r\nVary: Origin\r\n"),
171            None => String::new(),
172        },
173    }
174}

176/// MCP Streamable HTTP transport (POST /mcp → JSON response).
177///
178/// Connections are handled sequentially on the current thread — the browser
179/// session (including the V8 runtime) is single-threaded and `!Send`, so we
180/// never need to move state across threads.
181pub async fn run(
182    host: String,
183    port: u16,
184    proxy: Option<String>,
185    user_agent: Option<String>,
186    stealth: bool,
187    limits: crate::config::ExtractionLimits,
188    agent_directives: bool,
189) -> Result<()> {
190    let addr: std::net::SocketAddr = format!("{}:{}", host, port).parse()?;
191    let listener = TcpListener::bind(&addr).await?;
192    tracing::info!("MCP HTTP server on http://{}:{}/mcp", host, port);
193
194    let mut state =
195        BrowserState::new(proxy, user_agent, stealth, limits).with_agent_directives(agent_directives);
196    let allowed_origins = allowed_origins_env();
197
198    loop {
199        let (stream, peer) = listener.accept().await?;
200        tracing::debug!("MCP HTTP connection from {}", peer);
201        if let Err(e) = handle_connection(stream, &mut state, allowed_origins.as_deref()).await {
202            tracing::debug!("connection closed: {}", e);
203        }
204    }
205}

207async fn handle_connection(
208    stream: tokio::net::TcpStream,
209    state: &mut BrowserState,
210    allowed_origins: Option<&str>,
211) -> Result<()> {
212    let (reader, mut writer) = stream.into_split();
213    let mut reader = BufReader::new(reader);
214
215    loop {
216        let request = match read_request_with_timeout(
217            &mut reader,
218            allowed_origins,
219            REQUEST_READ_TIMEOUT,
220        )
221        .await?
222        {
223            RequestRead::Closed | RequestRead::Invalid => break,
224            RequestRead::Request(request) => request,
225        };
226        let HttpRequest {
227            method,
228            path,
229            accept_sse,
230            keep_alive,
231            origin,
232            body,
233        } = request;
234
235        // ── routing ──────────────────────────────────────────────────────────
236        if path != "/mcp" {
237            respond(&mut writer, 404, b"{\"error\":\"not found\"}").await?;
238            break;
239        }
240
241        // Origin gate: when TELEMACO_MCP_ALLOWED_ORIGINS is configured, a browser
242        // request from a non-listed origin is refused before it can drive the
243        // browser session (mitigates a malicious local web page issuing
244        // cross-origin POSTs to the loopback MCP port). The permissive default
245        // and no-Origin native clients are unaffected.
246        if !origin_allowed(origin.as_deref(), allowed_origins) {
247            respond(&mut writer, 403, b"{\"error\":\"origin not allowed\"}").await?;
248            break;
249        }
250        let cors = cors_header(origin.as_deref(), allowed_origins);
251
252        match method.as_str() {
253            "OPTIONS" => {
254                // mcp-protocol-version is part of the MCP spec, Authorization /
255                // X-API-Key are common for hosted deployments. Without these
256                // listed the browser preflight check fails and blocks the actual
257                // request.
258                let hdr = format!(
259                    "HTTP/1.1 204 No Content\r\n\
260                    {cors}\
261                    Access-Control-Allow-Methods: GET, POST, OPTIONS\r\n\
262                    Access-Control-Allow-Headers: Content-Type, Authorization, X-API-Key, mcp-protocol-version\r\n\
263                    Access-Control-Max-Age: 86400\r\n\
264                    \r\n"
265                );
266                writer.write_all(hdr.as_bytes()).await?;
267            }
268
269            "GET" if accept_sse => {
270                // SSE stream: hold open and send periodic keep-alive comments.
271                // Connections are served sequentially (the browser session is
272                // `!Send`), so holding this infinite keep-alive loop inline would
273                // never return to the accept loop and would wedge every later
274                // request. The keep-alive touches no browser state and the write
275                // half is `Send + 'static`, so detach the ping loop onto its own
276                // task and return — the accept loop stays free while the stream
277                // lives on independently.
278                let hdr = format!(
279                    "HTTP/1.1 200 OK\r\n\
280                    Content-Type: text/event-stream\r\n\
281                    Cache-Control: no-cache\r\n\
282                    Connection: keep-alive\r\n\
283                    {cors}\
284                    \r\n"
285                );
286                writer.write_all(hdr.as_bytes()).await?;
287                tokio::spawn(async move {
288                    loop {
289                        tokio::time::sleep(tokio::time::Duration::from_secs(15)).await;
290                        if writer.write_all(b": ping\n\n").await.is_err() {
291                            break;
292                        }
293                        let _ = writer.flush().await;
294                    }
295                });
296                return Ok(());
297            }
298
299            "POST" => {
300                let body = match body {
301                    RequestBody::MissingLength => {
302                        respond(&mut writer, 400, b"{\"error\":\"missing Content-Length\"}").await?;
303                        break;
304                    }
305                    RequestBody::TooLarge => {
306                        respond(&mut writer, 413, b"{\"error\":\"payload too large\"}").await?;
307                        break;
308                    }
309                    RequestBody::Read(body) => body,
310                    RequestBody::NotRead => unreachable!("valid MCP POST body was not read"),
311                };
312
313                let response = process_body(&body, state).await;
314                let bytes = serde_json::to_vec(&response)?;
315                respond_json(&mut writer, &bytes, &cors).await?;
316
317                if !keep_alive {
318                    break;
319                }
320            }
321
322            _ => {
323                respond(&mut writer, 405, b"{\"error\":\"method not allowed\"}").await?;
324                break;
325            }
326        }
327    }
328
329    Ok(())
330}

332async fn process_body(body: &[u8], state: &mut BrowserState) -> Value {
333    let msg: Value = match serde_json::from_slice(body) {
334        Ok(v) => v,
335        Err(_) => return json!({"jsonrpc":"2.0","id":null,"error":{"code":-32700,"message":"Parse error"}}),
336    };
337
338    if let Some(batch) = msg.as_array() {
339        let mut results = Vec::new();
340        for item in batch {
341            if let Some(r) = process_one(item, state).await {
342                results.push(r);
343            }
344        }
345        return Value::Array(results);
346    }
347
348    process_one(&msg, state).await
349        .unwrap_or_else(|| json!({"jsonrpc":"2.0","id":null,"error":{"code":-32600,"message":"Invalid Request"}}))
350}
351
352async fn process_one(msg: &Value, state: &mut BrowserState) -> Option<Value> {
353    let id = msg.get("id").cloned()?; // notifications have no id — return None
354    let method = msg.get("method").and_then(Value::as_str).unwrap_or("");
355    let params = msg.get("params").unwrap_or(&Value::Null);
356    let resp = dispatch(method, id, params, state).await;
357    Some(serde_json::to_value(resp).unwrap())
358}

360async fn respond_json(writer: &mut (impl AsyncWriteExt + Unpin), body: &[u8], cors: &str) -> Result<()> {
361    let hdr = format!(
362        "HTTP/1.1 200 OK\r\n\
363         Content-Type: application/json\r\n\
364         Content-Length: {}\r\n\
365         {cors}\
366         Connection: keep-alive\r\n\
367         \r\n",
368        body.len()
369    );
370    writer.write_all(hdr.as_bytes()).await?;
371    writer.write_all(body).await?;
372    writer.flush().await?;
373    Ok(())
374}

376async fn respond(writer: &mut (impl AsyncWriteExt + Unpin), status: u16, body: &[u8]) -> Result<()> {
377    let status_text = match status {
378        400 => "Bad Request",
379        403 => "Forbidden",
380        404 => "Not Found",
381        405 => "Method Not Allowed",
382        413 => "Payload Too Large",
383        _ => "OK",
384    };
385    let hdr = format!(
386        "HTTP/1.1 {status} {status_text}\r\n\
387         Content-Type: application/json\r\n\
388         Content-Length: {}\r\n\
389         \r\n",
390        body.len()
391    );
392    writer.write_all(hdr.as_bytes()).await?;
393    writer.write_all(body).await?;
394    writer.flush().await?;
395    Ok(())
396}

417        assert!(origin_allowed(Some("https://APP.example.com"), list));
418        assert!(!origin_allowed(Some("https://evil.example"), list));
419        // A native client (no Origin header) is always allowed.
420        assert!(origin_allowed(None, list));
