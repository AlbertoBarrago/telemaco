1//! Configurable extraction limits for the MCP server.
2//!
3//! Every cap an MCP tool applies to its own output is resolved here, from four
4//! layers, in this order of precedence:
5//!
6//! ```text
7//! tool call argument > CLI flag > environment variable > config file > default
8//! ```
9//!
10//! The tool-call argument is applied per call by each tool. Everything below it
11//! is resolved once at startup into an [`ExtractionLimits`] carried on
12//! `BrowserState`, which every tool already receives by `&mut`.
13//!
14//! A limit of `0` means unlimited.
15//!
16//! The caps exist because MCP output lands directly in an agent's context
17//! window: an uncapped dump of a large page can burn a whole window in one
18//! call. They are a context budget, not an arbitrary restriction, which is why
19//! raising them is a deliberate act of configuration.
20
21use std::path::{Path, PathBuf};
22
23use anyhow::{Context, Result};
24use serde::Deserialize;

26pub const DEFAULT_MAX_CHARS: usize = 4_000;
27pub const DEFAULT_MAX_LINKS: usize = 100;
28pub const DEFAULT_MAX_INTERACTIVE: usize = 100;
29pub const DEFAULT_MAX_SEARCH_RESULTS: usize = 10;
30pub const DEFAULT_SEARCH_CONTEXT_CHARS: usize = 80;
31pub const DEFAULT_MAX_NETWORK_REQUESTS: usize = 500;
32pub const DEFAULT_MAX_CONSOLE_MESSAGES: usize = 500;
33pub const DEFAULT_MAX_FORMS: usize = 100;
34
35/// Resolved output caps for every MCP tool that can emit unbounded text.
36///
37/// `deny_unknown_fields` is deliberate: a misspelled key must fail loudly at
38/// startup rather than be silently dropped, leaving the operator to wonder why
39/// their setting had no effect.
40#[derive(Debug, Clone, PartialEq, Eq, Deserialize)]
41#[serde(default, deny_unknown_fields)]
42pub struct ExtractionLimits {
43    /// Characters of page text returned by `browser_markdown` / `browser_snapshot`.
44    pub max_chars: usize,
45    /// Anchors returned by `browser_links`.
46    pub max_links: usize,
47    /// Elements returned by `browser_interactive_elements`.
48    pub max_interactive: usize,
49    /// Matches returned by `browser_search`.
50    pub max_search_results: usize,
51    /// Characters of surrounding context per `browser_search` match.
52    pub search_context_chars: usize,
53    /// Entries returned by `browser_network_requests`.
54    pub max_network_requests: usize,
55    /// Entries returned by `browser_console_messages`.
56    pub max_console_messages: usize,
57    /// Forms returned by `browser_detect_forms`.
58    pub max_forms: usize,
59}
60
61impl Default for ExtractionLimits {
62    fn default() -> Self {
63        ExtractionLimits {
64            max_chars: DEFAULT_MAX_CHARS,
65            max_links: DEFAULT_MAX_LINKS,
66            max_interactive: DEFAULT_MAX_INTERACTIVE,
67            max_search_results: DEFAULT_MAX_SEARCH_RESULTS,
68            search_context_chars: DEFAULT_SEARCH_CONTEXT_CHARS,
69            max_network_requests: DEFAULT_MAX_NETWORK_REQUESTS,
70            max_console_messages: DEFAULT_MAX_CONSOLE_MESSAGES,
71            max_forms: DEFAULT_MAX_FORMS,
72        }
73    }
74}
75
76impl ExtractionLimits {
77    /// Turn a configured limit into a comparable ceiling. `0` means unlimited,
78    /// so it maps to `usize::MAX` and every `take`/`>` at a call site keeps
79    /// working without a special case.
80    pub fn cap(value: usize) -> usize {
81        if value == 0 {
82            usize::MAX
83        } else {
84            value
85        }
86    }
87
88    /// Per-call override: a tool argument wins over the resolved config, and
89    /// `0` from a caller means unlimited just as it does in the file.
90    pub fn override_with(configured: usize, arg: Option<u64>) -> usize {
91        Self::cap(arg.map(|n| n as usize).unwrap_or(configured))
92    }
93}

95/// Root of the config file. Only `[mcp.limits]` is understood today; unknown
96/// sections are rejected for the same reason unknown keys are.
97#[derive(Debug, Clone, Default, Deserialize)]
98#[serde(default, deny_unknown_fields)]
99struct ConfigFile {
100    mcp: McpSection,
101}
102
103#[derive(Debug, Clone, Default, Deserialize)]
104#[serde(default, deny_unknown_fields)]
105struct McpSection {
106    limits: ExtractionLimits,
107}

109/// Config file locations searched when `--config` is not given, in order.
110const CWD_CONFIG: &str = "telemaco.toml";
111const HOME_CONFIG: &str = ".config/telemaco/config.toml";
112
113/// Every environment override, paired with the field it sets.
114const ENV_KEYS: &[(&str, fn(&mut ExtractionLimits, usize))] = &[
115    ("TELEMACO_MCP_MAX_CHARS", |l, v| l.max_chars = v),
116    ("TELEMACO_MCP_MAX_LINKS", |l, v| l.max_links = v),
117    ("TELEMACO_MCP_MAX_INTERACTIVE", |l, v| l.max_interactive = v),
118    ("TELEMACO_MCP_MAX_SEARCH_RESULTS", |l, v| l.max_search_results = v),
119    ("TELEMACO_MCP_SEARCH_CONTEXT_CHARS", |l, v| l.search_context_chars = v),
120    ("TELEMACO_MCP_MAX_NETWORK_REQUESTS", |l, v| l.max_network_requests = v),
121    ("TELEMACO_MCP_MAX_CONSOLE_MESSAGES", |l, v| l.max_console_messages = v),
122    ("TELEMACO_MCP_MAX_FORMS", |l, v| l.max_forms = v),
123];
124
125/// Apply environment overrides on top of `base`.
126///
127/// The lookup is injected rather than read from the process environment so the
128/// precedence rules can be unit-tested without mutating global state, following
129/// `navigation_timeout_from_env_value` in `telemaco-browser`.
130///
131/// An unparsable value is ignored and leaves the lower layer in place: a typo in
132/// a shell export must not silently clamp output to zero.
133pub fn apply_env_overrides<F>(base: ExtractionLimits, lookup: F) -> ExtractionLimits
134where
135    F: Fn(&str) -> Option<String>,
136{
137    let mut limits = base;
138    for (key, set) in ENV_KEYS {
139        if let Some(value) = lookup(key).and_then(|v| v.trim().parse::<usize>().ok()) {
140            set(&mut limits, value);
141        }
142    }
143    limits
144}

146/// Parse config file contents. Separated from IO so malformed-input behavior is
147/// directly testable.
148pub fn parse_config(contents: &str) -> Result<ExtractionLimits> {
149    let parsed: ConfigFile = toml::from_str(contents)?;
150    Ok(parsed.mcp.limits)
151}
152
153/// The config file to read, or `None` for defaults.
154///
155/// An explicit `--config` path that does not exist is an error: the operator
156/// named a file, so silently ignoring it would hide the mistake. A discovered
157/// path that does not exist simply means "no config".
158fn resolve_path(explicit: Option<&Path>) -> Result<Option<PathBuf>> {
159    if let Some(path) = explicit {
160        if !path.is_file() {
161            anyhow::bail!("config file not found: {}", path.display());
162        }
163        return Ok(Some(path.to_path_buf()));
164    }
165
166    let cwd = PathBuf::from(CWD_CONFIG);
167    if cwd.is_file() {
168        return Ok(Some(cwd));
169    }
170
171    if let Some(home) = std::env::var_os("HOME") {
172        let home_path = PathBuf::from(home).join(HOME_CONFIG);
173        if home_path.is_file() {
174            return Ok(Some(home_path));
175        }
176    }
177
178    Ok(None)
179}

181/// Resolve the limits for this process: defaults, then the config file, then
182/// the environment. CLI flags are layered on by the caller, and tool arguments
183/// per call, so this is everything below those two.
184///
185/// A malformed or unreadable config is a hard error at startup rather than a
186/// fallback to defaults: a broken config that silently does nothing is worse
187/// than a server that refuses to start.
188pub fn load(explicit: Option<&Path>) -> Result<ExtractionLimits> {
189    let from_file = match resolve_path(explicit)? {
190        Some(path) => {
191            let contents = std::fs::read_to_string(&path)
192                .with_context(|| format!("reading config file {}", path.display()))?;
193            parse_config(&contents)
194                .with_context(|| format!("parsing config file {}", path.display()))?
195        }
196        None => ExtractionLimits::default(),
197    };
198
199    Ok(apply_env_overrides(from_file, |key| std::env::var(key).ok()))
200}

210    #[test]
211    fn defaults_match_the_documented_values() {
212        let limits = ExtractionLimits::default();
213        assert_eq!(limits.max_chars, 4_000);
214        assert_eq!(limits.max_links, 100);
215        assert_eq!(limits.max_interactive, 100);
216        assert_eq!(limits.max_search_results, 10);
217        assert_eq!(limits.search_context_chars, 80);
218    }

220    #[test]
221    fn empty_config_yields_defaults() {
222        assert_eq!(parse_config("").unwrap(), ExtractionLimits::default());
223    }

225    #[test]
226    fn partial_config_leaves_other_fields_at_default() {
227        let limits = parse_config("[mcp.limits]\nmax_chars = 20000\n").unwrap();
228        assert_eq!(limits.max_chars, 20_000);
229        assert_eq!(limits.max_links, DEFAULT_MAX_LINKS);
230    }

251    #[test]
252    fn env_overrides_the_file() {
253        let from_file = parse_config("[mcp.limits]\nmax_chars = 20000\n").unwrap();
254        let resolved = apply_env_overrides(from_file, |key| {
255            (key == "TELEMACO_MCP_MAX_CHARS").then(|| "50000".to_string())
256        });
257        assert_eq!(resolved.max_chars, 50_000);
258    }

260    #[test]

262        let resolved = apply_env_overrides(ExtractionLimits::default(), |key| {
263            (key == "TELEMACO_MCP_MAX_LINKS").then(|| "7".to_string())
264        });
265        assert_eq!(resolved.max_links, 7);
266        assert_eq!(resolved.max_chars, DEFAULT_MAX_CHARS);
267    }
