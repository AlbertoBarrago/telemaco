3// A unit of pending serialization work. Held on an explicit heap stack instead
4// of the call stack so a deeply nested tree cannot overflow the thread stack and
5// abort the process (a stack overflow is a hard abort that `op_dom`'s
6// catch_unwind cannot recover). `descendants()` is iterative + capped for the
7// same reason; the serializer must be too.
8enum SerializeWork {
9    // Serialize this node; `include_self` mirrors the old recursive param.
10    Node(NodeId, bool),
11    // Emit a previously-opened element's closing tag, after its children.
12    CloseTag(String),
13}
14
15impl DomTree {
16    pub fn outer_html(&self, node_id: NodeId) -> String {
17        let mut buf = String::new();
18        self.serialize_worklist(vec![SerializeWork::Node(node_id, true)], &mut buf);
19        buf
20    }
21
22    pub fn inner_html(&self, node_id: NodeId) -> String {
23        let mut buf = String::new();
24        self.serialize_worklist(self.child_work(self.content_source(node_id)), &mut buf);
25        buf
26    }
27
28    // The node whose children represent `node_id`'s markup. For a <template>
29    // that is its contents document, since the parser puts template children
30    // there rather than under the element (issue #463); for everything else it
31    // is the node itself.
32    fn content_source(&self, node_id: NodeId) -> NodeId {
33        self.with_node(node_id, |n| match &n.data {
34            NodeData::Element { template_contents: Some(contents), .. } => Some(*contents),
35            _ => None,
36        })
37        .flatten()
38        .unwrap_or(node_id)
39    }
40
41    // Children as work items in document order (top of the LIFO stack first).
42    fn child_work(&self, node_id: NodeId) -> Vec<SerializeWork> {
43        self.children(node_id)
44            .into_iter()
45            .rev()
46            .map(|c| SerializeWork::Node(c, true))
47            .collect()
48    }
49
50    fn serialize_worklist(&self, mut stack: Vec<SerializeWork>, buf: &mut String) {
51        // Defense in depth, mirroring descendants(): a well-formed subtree emits
52        // at most one Node plus one CloseTag per node, so 2*nodes.len() work
53        // items bound a valid walk. Exceeding it means the graph is cyclic (the
54        // append_child / insert_before guards prevent that); stop rather than
55        // spin forever. On a valid tree this bound is never reached.
56        let max_steps = self
57            .node_slot_count()
58            .saturating_mul(2)
59            .saturating_add(16);
60        let mut steps = 0usize;
61
62        while let Some(work) = stack.pop() {
63            steps += 1;
64            if steps > max_steps {
65                eprintln!("telemaco: serialize worklist cap hit - tree has a cycle");
66                break;
67            }
68
69            let (node_id, include_self) = match work {
70                SerializeWork::CloseTag(tag) => {
71                    buf.push_str("</");
72                    buf.push_str(&tag);
73                    buf.push('>');
74                    continue;
75                }
76                SerializeWork::Node(node_id, include_self) => (node_id, include_self),
77            };
78
79            let node = match self.get_node(node_id) {
80                Some(n) => n,
81                None => continue,
82            };
83
84            match &node.data {
85                NodeData::Document => {
86                    for w in self.child_work(node_id) {
87                        stack.push(w);
88                    }
89                }
90                NodeData::Doctype { name, .. } => {
91                    buf.push_str("<!DOCTYPE ");
92                    buf.push_str(name);
93                    buf.push('>');
94                }
95                NodeData::Element { name, attrs, template_contents, .. } => {
96                    let tag = name.local.as_ref();
97                    if include_self {
98                        buf.push('<');
99                        buf.push_str(tag);
100                        for attr in attrs {
101                            buf.push(' ');
102                            if let Some(prefix) = &attr.name.prefix {
103                                buf.push_str(prefix);
104                                buf.push(':');
105                            }
106                            buf.push_str(attr.name.local.as_ref());
107                            buf.push_str("=\"");
108                            escape_attr(&attr.value, buf);
109                            buf.push('"');
110                        }
111                        buf.push('>');
112                    }
113
114                    if !is_void_element(tag) {
115                        // Push the closing tag first so it pops after all the
116                        // children we push next.
117                        if include_self {
118                            stack.push(SerializeWork::CloseTag(tag.to_string()));
119                        }
120                        // A <template> serializes its contents document, not its
121                        // own (always empty) children (issue #463). template_contents
122                        // is already in hand from this node, so read it directly
123                        // rather than re-fetching and cloning the node.
124                        let source = template_contents.unwrap_or(node_id);
125                        for w in self.child_work(source) {
126                            stack.push(w);
127                        }
128                    }
129                }
130                NodeData::Text { contents } => {
131                    let parent_is_raw = node.parent
132                        .and_then(|pid| {
133                            self.with_node(pid, |p| {
134                                p.as_element()
135                                    .map(|name| is_raw_text_element(name.local.as_ref()))
136                                    .unwrap_or(false)
137                            })
138                        })
139                        .unwrap_or(false);
140
141                    if parent_is_raw {
142                        buf.push_str(contents);
143                    } else {
144                        escape_text(contents, buf);
145                    }
146                }
147                NodeData::Comment { contents } => {
148                    buf.push_str("<!--");
149                    // The HTML parser can never produce a comment that closes early,
150                    // but script can via document.createComment(...). The tokenizer
151                    // ends a comment on ANY of four sequences: "-->", "--!>", a
152                    // leading ">", or a leading "->" (comment-start / -start-dash
153                    // abrupt-close). Every one requires a ">". Emitting it verbatim
154                    // would close the comment early and let the trailing text parse
155                    // as live markup (mXSS). Entities are not decoded inside
156                    // comments, so escaping every ">" to "&gt;" neutralizes all four
157                    // forms at once and keeps the data as a single comment.
158                    // (Supersedes the earlier "-->"-only guard, which left the
159                    // leading-">", leading-"->", and "--!>" forms exploitable.)
160                    if contents.contains('>') {
161                        buf.push_str(&contents.replace('>', "&gt;"));
162                    } else {
163                        buf.push_str(contents);
164                    }
165                    buf.push_str("-->");
166                }
167                NodeData::ProcessingInstruction { target, data } => {
168                    buf.push_str("<?");
169                    buf.push_str(target);
170                    buf.push(' ');
171                    buf.push_str(data);
172                    buf.push('>');
173                }
174            }
175        }
176    }
177}

179fn escape_text(s: &str, buf: &mut String) {
180    for c in s.chars() {
181        match c {
182            '&' => buf.push_str("&amp;"),
183            '<' => buf.push_str("&lt;"),
184            '>' => buf.push_str("&gt;"),
185            _ => buf.push(c),
186        }
187    }
188}
189
190fn escape_attr(s: &str, buf: &mut String) {
191    for c in s.chars() {
192        match c {
193            '&' => buf.push_str("&amp;"),
194            '"' => buf.push_str("&quot;"),
195            _ => buf.push(c),
196        }
197    }
198}

200fn is_void_element(tag: &str) -> bool {
201    matches!(
202        tag,
203        "area" | "base" | "br" | "col" | "embed" | "hr" | "img" | "input" | "link" | "meta"
204            | "param" | "source" | "track" | "wbr"
205    )
206}
207
208fn is_raw_text_element(tag: &str) -> bool {
209    matches!(tag, "script" | "style" | "textarea" | "title")
210}

266    #[test]
267    fn comment_serialization_neutralizes_all_terminator_forms() {
268        use crate::tree::NodeData;
269
270        // A comment ends on any of "-->", "--!>", a leading ">", or a leading
271        // "->". `document.createComment(...)` accepts arbitrary strings, so a
272        // scripted comment can carry each closing form followed by a real tag.
273        // Serializing must keep the payload inside a single comment; if it
274        // closes early, the trailing "<img>" becomes live markup (mXSS).
275        let payloads = [
276            "><img src=x onerror=alert(1)>", // leading ">" abrupt-closes empty comment
277            "-><img src=x>",                 // leading "->" abrupt-closes
278            "a--!><img src=x>",              // internal "--!>" closes (incorrectly-closed)
279            "a--><img src=x>",               // internal "-->" closes (the previously-fixed form)
280        ];
281
282        for payload in payloads {
283            let tree = parse_html(r#"<div id="host"></div>"#);
284            let host = tree.get_element_by_id("host").unwrap();
285            let comment = tree.new_node(NodeData::Comment { contents: payload.to_string() });
286            tree.append_child(host, comment);
287
288            let serialized = tree.outer_html(host);
289
290            // Re-parsing the serialized markup must not surface an <img>: the
291            // payload has to stay inside the comment.
292            let reparsed = parse_html(&serialized);
293            assert!(
294                reparsed.query_selector("img").unwrap().is_none(),
295                "payload {payload:?} escaped the comment; serialized = {serialized}"
296            );
297            // And the serialized comment data must carry no raw ">".
298            let inner = &serialized[serialized.find("<!--").unwrap() + 4..];
299            let inner = &inner[..inner.find("-->").unwrap()];
300            assert!(
301                !inner.contains('>'),
302                "comment data still contains a raw '>': {serialized}"
303            );
304        }
305    }

308    fn host_serialization_excludes_its_native_shadow_tree() {
309        let tree = parse_html(r#"<x-card id="host"><span id="light">light</span></x-card>"#);
310        let host = tree.get_element_by_id("host").unwrap();
311        let root = tree.attach_shadow_root(host, ShadowRootMode::Open).unwrap();
312        let shadow = tree.new_node(NodeData::Element {
313            name: QualName::new(None, ns!(html), LocalName::from("strong")),
314            attrs: Vec::new(),
315            template_contents: None,
316            mathml_annotation_xml_integration_point: false,
317        });
318        let shadow_text = tree.new_node(NodeData::Text {
319            contents: "shadow".to_string(),
320        });
321        tree.append_child(root, shadow);
322        tree.append_child(shadow, shadow_text);
323
324        assert_eq!(tree.inner_html(host), r#"<span id="light">light</span>"#);
325        assert_eq!(
326            tree.outer_html(host),
327            r#"<x-card id="host"><span id="light">light</span></x-card>"#
328        );
329        assert_eq!(tree.inner_html(root), "<strong>shadow</strong>");
330    }
