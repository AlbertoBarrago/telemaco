1//! Charset detection and decoding for HTTP response bodies.
2//!
3//! Issue #113: telemaco used to call `String::from_utf8_lossy` on every
4//! response body, which silently corrupts every non-UTF-8 page (GBK, Big5,
5//! Shift-JIS, Windows-125x, EUC-KR, ISO-8859-x). Picking the right decoder
6//! is required for scraping non-Latin sites at all.
7//!
8//! Detection order, mirroring real browsers (HTML5 spec § 8.2.2.4):
9//!   1. `Content-Type: text/html; charset=...` from the HTTP response header.
10//!   2. `<meta charset="...">` or `<meta http-equiv="Content-Type" content="text/html; charset=...">`
11//!      sniffed from the first 1024 bytes of the body.
12//!   3. Default UTF-8.
13//!
14//! For non-HTML resources (JS, CSS, JSON), only steps 1 and 3 apply.
15
16use encoding_rs::{DecoderResult, EncoderResult, Encoding, UTF_8};

18/// WHATWG canonical (lowercased) name for an encoding label, or None if the
19/// label is not a known encoding. Backs `TextDecoder`'s label validation and
20/// its `.encoding` property.
21pub fn label_name(label: &str) -> Option<String> {
22    Encoding::for_label(label.as_bytes()).map(|e| e.name().to_ascii_lowercase())
23}

25/// Decode `bytes` with an explicit encoding label, with TextDecoder semantics.
26/// Returns None when the label is unknown, or (when `fatal`) when the input is
27/// not valid in that encoding. Non-fatal decoding replaces errors with U+FFFD.
28pub fn decode_with_label(label: &str, bytes: &[u8], fatal: bool, ignore_bom: bool) -> Option<String> {
29    let enc = Encoding::for_label(label.as_bytes())?;
30    let mut dec = if ignore_bom {
31        enc.new_decoder_without_bom_handling()
32    } else {
33        enc.new_decoder()
34    };
35    if fatal {
36        let mut out = String::with_capacity(bytes.len() + 1);
37        let (res, _) = dec.decode_to_string_without_replacement(bytes, &mut out, true);
38        match res {
39            DecoderResult::InputEmpty => Some(out),
40            _ => None,
41        }
42    } else {
43        let mut out = String::with_capacity(bytes.len() * 2 + 1);
44        let _ = dec.decode_to_string(bytes, &mut out, true);
45        Some(out)
46    }
47}

49/// Decode an HTTP response body. `content_type_header` is the raw header
50/// value if present (e.g. `text/html; charset=gbk`). For HTML resources,
51/// the parser also sniffs `<meta charset>` in the first 1KB.
52pub fn decode_response(bytes: &[u8], content_type_header: Option<&str>) -> String {
53    let (encoding, _) = detect_encoding(bytes, content_type_header);
54    let (cow, _, _) = encoding.decode(bytes);
55    cow.into_owned()
56}

58/// Like `decode_response`, but also returns the WHATWG canonical name of the
59/// encoding that was used (e.g. "EUC-JP", "Shift_JIS", "UTF-8"). Callers use
60/// the name to expose `document.characterSet` and to do document-encoding-aware
61/// URL query serialization (the WHATWG "encoding override").
62pub fn decode_response_with_name(
63    bytes: &[u8],
64    content_type_header: Option<&str>,
65) -> (String, &'static str) {
66    let (encoding, _) = detect_encoding(bytes, content_type_header);
67    let (cow, _, _) = encoding.decode(bytes);
68    (cow.into_owned(), encoding.name())
69}

71const PCT_HEX: &[u8; 16] = b"0123456789ABCDEF";
72
73fn push_pct(out: &mut String, b: u8) {
74    out.push('%');
75    out.push(PCT_HEX[(b >> 4) as usize] as char);
76    out.push(PCT_HEX[(b & 0x0F) as usize] as char);
77}

79/// Append an ASCII `byte` to a URL query string, percent-encoding it when it is
80/// in the WHATWG query percent-encode set (C0 controls, space, `"`, `#`, `<`,
81/// `>`, 0x7F), plus `'` for special schemes (the special-query set). ASCII
82/// delimiters like `=` and `&` are left literal, so structured queries survive.
83fn push_query_ascii(out: &mut String, b: u8, special: bool) {
84    let must_encode = b <= 0x20
85        || b == 0x7F
86        || matches!(b, 0x22 | 0x23 | 0x3C | 0x3E)
87        || (special && b == 0x27);
88    if must_encode {
89        push_pct(out, b);
90    } else {
91        out.push(b as char);
92    }
93}

95/// Encode a run of non-ASCII code points to the target charset and percent-
96/// encode EVERY resulting byte. The bytes serialize a non-ASCII character, so
97/// all of them are escaped (this is what the WPT legacy-mb encode-href tests
98/// expect, e.g. Big5 `一` -> `%A4%40` even though the 0x40 trail byte is ASCII).
99/// Unmappable code points become the literal `%26%23<decimal>%3B` sequence (a
100/// percent-encoded `&#NNN;` numeric character reference), per the URL spec.
101fn encode_run_pct(out: &mut String, run: &str, enc: &'static Encoding) {
102    let mut encoder = enc.new_encoder();
103    let mut input = run;
104    let mut buf = [0u8; 256];
105    loop {
106        let (result, read, written) =
107            encoder.encode_from_utf8_without_replacement(input, &mut buf, true);
108        for &b in &buf[..written] {
109            push_pct(out, b);
110        }
111        input = &input[read..];
112        match result {
113            EncoderResult::InputEmpty => break,
114            EncoderResult::OutputFull => continue,
115            EncoderResult::Unmappable(c) => {
116                out.push_str("%26%23");
117                out.push_str(&(c as u32).to_string());
118                out.push_str("%3B");
119            }
120        }
121    }
122}

124/// WHATWG URL "percent-encode after encoding" for the query component, using a
125/// non-UTF-8 document encoding override (`label`). `query` is the already
126/// UTF-8-percent-decoded query string. ASCII code points use the (special-)
127/// query percent-encode set so real query delimiters (`=`, `&`) stay literal;
128/// runs of non-ASCII code points are encoded to the target charset with every
129/// byte percent-encoded. Returns None when the label is unknown.
130pub fn url_encode_query(query: &str, label: &str, special: bool) -> Option<String> {
131    let enc = Encoding::for_label(label.as_bytes())?;
132    let mut out = String::with_capacity(query.len() * 3);
133    let mut run_start: Option<usize> = None;
134    for (idx, c) in query.char_indices() {
135        if c.is_ascii() {
136            if let Some(s) = run_start.take() {
137                encode_run_pct(&mut out, &query[s..idx], enc);
138            }
139            push_query_ascii(&mut out, c as u8, special);
140        } else if run_start.is_none() {
141            run_start = Some(idx);
142        }
143    }
144    if let Some(s) = run_start {
145        encode_run_pct(&mut out, &query[s..], enc);
146    }
147    Some(out)
148}

150/// Same as `decode_response` but skips the `<meta charset>` sniff. Use for
151/// non-HTML resources where embedded HTML meta tags are not authoritative
152/// (script and style bodies, JSON, plain text).
153pub fn decode_non_html(bytes: &[u8], content_type_header: Option<&str>) -> String {
154    let encoding = content_type_header
155        .and_then(charset_from_content_type)
156        .and_then(|name| Encoding::for_label(name.as_bytes()))
157        .unwrap_or(UTF_8);
158    let (cow, _, _) = encoding.decode(bytes);
159    cow.into_owned()
160}

162/// Resolve the encoding to use for an HTML response, mirroring the HTML5
163/// detection order. Returns the encoding and a tag describing where it was
164/// picked from (for logging / tests).
165pub fn detect_encoding<'a>(
166    bytes: &'a [u8],
167    content_type_header: Option<&str>,
168) -> (&'static Encoding, &'static str) {
169    if let Some(charset) = content_type_header.and_then(charset_from_content_type) {
170        if let Some(enc) = Encoding::for_label(charset.as_bytes()) {
171            return (enc, "content-type");
172        }
173    }
174    if let Some(enc) = sniff_meta_charset(bytes) {
175        return (enc, "meta-charset");
176    }
177    (UTF_8, "default-utf8")
178}

180/// Pull the `charset=` parameter out of a Content-Type header value.
181fn charset_from_content_type(header: &str) -> Option<String> {
182    for part in header.split(';') {
183        let trimmed = part.trim();
184        if let Some(rest) = trimmed.strip_prefix("charset=").or_else(|| trimmed.strip_prefix("Charset=")) {
185            // Strip surrounding quotes if present.
186            let value = rest.trim_matches(|c: char| c == '"' || c == '\'').trim();
187            if !value.is_empty() {
188                return Some(value.to_ascii_lowercase());
189            }
190        }
191        // Some servers send `Content-Type: text/html; CHARSET = gbk`.
192        let lower = trimmed.to_ascii_lowercase();
193        if let Some(rest) = lower.strip_prefix("charset") {
194            let rest = rest.trim_start();
195            if let Some(rest) = rest.strip_prefix('=') {
196                let value = rest.trim().trim_matches(|c: char| c == '"' || c == '\'');
197                if !value.is_empty() {
198                    return Some(value.to_ascii_lowercase());
199                }
200            }
201        }
202    }
203    None
204}

206/// Return an exact attribute value from a lowercased `<meta ...>` tag.
207///
208/// This deliberately tokenizes attribute names instead of searching for a
209/// substring: `data-charset` and a description containing `charset=...` are
210/// not character encoding declarations.
211fn meta_attribute<'a>(tag: &'a str, target: &str) -> Option<&'a str> {
212    let mut rest = tag.strip_prefix("<meta")?;
213    if rest
214        .chars()
215        .next()
216        .is_some_and(|c| !c.is_ascii_whitespace() && !matches!(c, '>' | '/'))
217    {
218        return None;
219    }
220
221    while !rest.is_empty() {
222        rest = rest.trim_start_matches(|c: char| c.is_ascii_whitespace() || c == '/');
223        if rest.is_empty() || rest.starts_with('>') {
224            break;
225        }
226
227        let name_end = rest
228            .find(|c: char| c.is_ascii_whitespace() || matches!(c, '=' | '>' | '/'))
229            .unwrap_or(rest.len());
230        if name_end == 0 {
231            rest = &rest[rest.chars().next()?.len_utf8()..];
232            continue;
233        }
234
235        let name = &rest[..name_end];
236        rest = rest[name_end..].trim_start();
237
238        let mut value = "";
239        if let Some(after_equals) = rest.strip_prefix('=') {
240            let after_equals = after_equals.trim_start();
241            if let Some(quote) = after_equals
242                .chars()
243                .next()
244                .filter(|c| matches!(c, '"' | '\''))
245            {
246                let quoted = &after_equals[quote.len_utf8()..];
247                if let Some(end) = quoted.find(quote) {
248                    value = &quoted[..end];
249                    rest = &quoted[end + quote.len_utf8()..];
250                } else {
251                    value = quoted;
252                    rest = "";
253                }
254            } else {
255                let end = after_equals
256                    .find(|c: char| c.is_ascii_whitespace() || matches!(c, '>' | '/'))
257                    .unwrap_or(after_equals.len());
258                value = &after_equals[..end];
259                rest = &after_equals[end..];
260            }
261        }
262
263        if name.eq_ignore_ascii_case(target) {
264            return Some(value);
265        }
266    }
267
268    None
269}

271/// Scan the first 1024 bytes for a `<meta charset="...">` or
272/// `<meta http-equiv="Content-Type" content="...; charset=...">` declaration.
273/// We only look at ASCII bytes; valid meta-charset declarations are always
274/// ASCII regardless of the page's actual encoding.
275fn sniff_meta_charset(bytes: &[u8]) -> Option<&'static Encoding> {
276    let prefix_len = bytes.len().min(1024);
277    let prefix = &bytes[..prefix_len];
278    // Lossy is fine: any meta charset attribute is ASCII, even on a non-UTF-8 page.
279    let s = String::from_utf8_lossy(prefix).to_ascii_lowercase();
280
281    // Look for any `<meta ... charset=...>` pattern in the first 1KB. We
282    // intentionally accept both the modern shorthand (`<meta charset=gbk>`)
283    // and the legacy http-equiv form (`<meta http-equiv="content-type" content="text/html; charset=gbk">`).
284    let mut pos = 0;
285    while let Some(meta_start) = s[pos..].find("<meta") {
286        let abs = pos + meta_start;
287        // Find the closing `>` for this meta tag.
288        let end = s[abs..].find('>').map(|e| abs + e).unwrap_or(s.len());
289        let tag = &s[abs..end];
290
291        if let Some(enc) = meta_attribute(tag, "charset")
292            .filter(|value| !value.is_empty())
293            .and_then(|value| Encoding::for_label(value.as_bytes()))
294        {
295            return Some(enc);
296        }
297
298        let is_legacy_declaration = meta_attribute(tag, "http-equiv")
299            .is_some_and(|value| value.eq_ignore_ascii_case("content-type"));
300        if is_legacy_declaration {
301            if let Some(enc) = meta_attribute(tag, "content")
302                .and_then(charset_from_content_type)
303                .and_then(|value| Encoding::for_label(value.as_bytes()))
304            {
305                return Some(enc);
306            }
307        }
308
309        pos = end + 1;
310        if pos >= s.len() {
311            break;
312        }
313    }
314
315    None
316}

324        let bytes = b"<html><head><meta charset=\"utf-8\"></head><body></body></html>";
