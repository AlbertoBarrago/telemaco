6pub struct BrowserContext {
7    pub id: String,
8    pub cookie_jar: Arc<CookieJar>,
9    pub http_client: Arc<TelemacoHttpClient>,
10    pub user_agent: String,
11    pub platform: String,
12    pub ua_platform: String,
13    pub ua_platform_version: String,
14    pub proxy_url: Option<String>,
15    pub robots_cache: Arc<RobotsCache>,
16    pub obey_robots: bool,
17    pub stealth: bool,
18    /// When true, CDP-driven navigation to file:// URLs is permitted.
19    /// Default is false: a remote CDP client cannot point the browser
20    /// at /etc/shadow even if Telemaco is running as a privileged user.
21    /// Flip on via `telemaco serve --allow-file-access` for legitimate
22    /// local-HTML testing workflows. The CLI's own `telemaco fetch
23    /// file://...` path is unaffected because it does not go through
24    /// the CDP server.
25    pub allow_file_access: bool,
26    pub storage_dir: Option<PathBuf>,
27    /// When true, the http client allows fetching localhost / RFC1918 /
28    /// link-local addresses. Set via `--allow-private-network` (issue #33).
29    /// Independent of `allow_file_access` because they cover different threat
30    /// models: file:// is a local file-system read, while private-network is
31    /// the broader SSRF gate from issue #4.
32    pub allow_private_network: bool,
33}

66        proxy_url: Option<String>,
67        stealth: bool,
68        user_agent: Option<String>,
69        storage_dir: Option<PathBuf>,
70        allow_private_network: bool,

74
75    fn _new_inner(
76        id: String,
77        proxy_url: Option<String>,
78        stealth: bool,
79        user_agent: Option<String>,
80        storage_dir: Option<PathBuf>,
81        allow_private_network: bool,
82    ) -> Self {
83        let cookie_jar = Arc::new(CookieJar::new());
84
85        // Restore cookies from disk if storage_dir is configured
86        if let Some(ref dir) = storage_dir {
87            let cookie_path = dir.join("cookies.json");
88            if cookie_path.exists() {
89                match cookie_jar.load_from_file(&cookie_path) {
90                    Ok(n) if n > 0 => {
91                        tracing::info!("Loaded {} cookies from {}", n, cookie_path.display());
92                    }
93                    Ok(_) => {}
94                    Err(e) => {
95                        tracing::warn!("Failed to load cookies from {}: {}", cookie_path.display(), e);
96                    }
97                }
98            }
99        }
100
101        let mut client = TelemacoHttpClient::with_full_options(
102            cookie_jar.clone(),
103            proxy_url.as_deref(),
104            allow_private_network,
105        );
106        if stealth {
107            client.block_trackers = true;
108        }
109        let profile = crate::profiles::select_profile();
110        let resolved_ua = user_agent.unwrap_or_else(|| profile.user_agent.to_string());
111        let platform = profile.platform.to_string();
112        let ua_platform = profile.ua_platform.to_string();
113        let ua_platform_version = profile.ua_platform_version.to_string();
114        // Sync the http client's UA at construction so navigation requests pick it
115        // up before any async setup runs. The lock has no other holders here, so
116        // try_write always succeeds; we fall back silently if it ever fails.
117        if let Ok(mut guard) = client.user_agent.try_write() {
118            *guard = resolved_ua.clone();
119        }
120        let http_client = Arc::new(client);
121        BrowserContext {
122            id,
123            cookie_jar,
124            http_client,
125            user_agent: resolved_ua,
126            platform,
127            ua_platform,
128            ua_platform_version,
129            proxy_url,
130            robots_cache: Arc::new(RobotsCache::new()),
131            obey_robots: false,
132            stealth,
133            allow_file_access: false,
134            storage_dir,
135            allow_private_network,
136        }
137    }

156    /// Create a context with the same browser configuration but independent
157    /// mutable network state. Persistent copies start with the template's
158    /// current cookies; incognito copies start empty and never write to the
159    /// template's storage directory.
160    pub fn isolated_copy(&self, id: String, persistent: bool) -> Self {
161        let cookie_jar = Arc::new(CookieJar::new());
162        if persistent {
163            cookie_jar.set_cookies_from_cdp(self.cookie_jar.get_all_cookies());
164        }
165
166        let mut client = TelemacoHttpClient::with_full_options(
167            cookie_jar.clone(),
168            self.proxy_url.as_deref(),
169            self.allow_private_network,
170        );
171        if self.stealth {
172            client.block_trackers = true;
173        }
174        if let Ok(mut guard) = client.user_agent.try_write() {
175            *guard = self.user_agent.clone();
176        }
177
178        BrowserContext {
179            id,
180            cookie_jar,
181            http_client: Arc::new(client),
182            user_agent: self.user_agent.clone(),
183            platform: self.platform.clone(),
184            ua_platform: self.ua_platform.clone(),
185            ua_platform_version: self.ua_platform_version.clone(),
186            proxy_url: self.proxy_url.clone(),
187            robots_cache: Arc::new(RobotsCache::new()),
188            obey_robots: self.obey_robots,
189            stealth: self.stealth,
190            allow_file_access: self.allow_file_access,
191            storage_dir: persistent.then(|| self.storage_dir.clone()).flatten(),
192            allow_private_network: self.allow_private_network,
193        }
194    }
195
196    /// Persist cookies to disk if storage_dir is configured.
197    /// Called during graceful shutdown.
198    pub fn save_cookies(&self) {
199        if let Some(ref dir) = self.storage_dir {
200            let _ = std::fs::create_dir_all(dir);
201            let cookie_path = dir.join("cookies.json");
202            if let Err(e) = self.cookie_jar.save_to_file(&cookie_path) {
203                tracing::warn!("Failed to save cookies to {}: {}", cookie_path.display(), e);
204            } else {
205                tracing::info!("Saved cookies to {}", cookie_path.display());
206            }
207        }
208    }

215    #[tokio::test(flavor = "current_thread")]
216    async fn with_full_options_propagates_user_agent_to_http_client() {
217        let ctx = BrowserContext::with_full_options(
218            "test".to_string(),
219            None,
220            false,
221            Some("Custom-UA/1.0".to_string()),
222        );
223        assert_eq!(ctx.user_agent, "Custom-UA/1.0");
224        let client_ua = ctx.http_client.user_agent.read().await.clone();
225        assert_eq!(client_ua, "Custom-UA/1.0");
226    }
227
228    #[tokio::test(flavor = "current_thread")]
229    async fn with_full_options_falls_back_to_chrome_default() {
230        let ctx = BrowserContext::with_full_options(
231            "test".to_string(),
232            None,
233            false,
234            None,
235        );
236        assert!(ctx.user_agent.contains("Chrome"));
237        let client_ua = ctx.http_client.user_agent.read().await.clone();
238        assert!(client_ua.contains("Chrome"));
239        assert_eq!(ctx.user_agent, client_ua);
240    }

242    #[tokio::test(flavor = "current_thread")]
243    async fn with_options_keeps_default_user_agent() {
244        let ctx = BrowserContext::with_options("test".to_string(), None, false);
245        assert!(ctx.user_agent.contains("Chrome"));
246    }
247
248    #[tokio::test(flavor = "current_thread")]
249    async fn isolated_copy_does_not_share_mutable_network_state() {
250        let source = BrowserContext::with_full_options(
251            "source".to_string(),
252            None,
253            false,
254            Some("Template-UA/1.0".to_string()),
255        );
256        source.cookie_jar.set_cookie("sid=source", &url::Url::parse("https://example.com").unwrap());
257
258        let persistent = source.isolated_copy("persistent".to_string(), true);
259        let incognito = source.isolated_copy("incognito".to_string(), false);
260
261        assert_eq!(persistent.cookie_jar.get_all_cookies().len(), 1);
262        assert!(incognito.cookie_jar.get_all_cookies().is_empty());
263        persistent.cookie_jar.clear();
264        persistent.http_client.set_user_agent("Changed-UA/2.0").await;
265
266        assert_eq!(source.cookie_jar.get_all_cookies().len(), 1);
267        assert_eq!(source.http_client.user_agent.read().await.as_str(), "Template-UA/1.0");
268    }
