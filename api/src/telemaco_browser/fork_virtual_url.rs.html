1//! Fork-only: adopt a URL the page routed to itself.
2//!
3//! A single page app answers a click by calling `history.pushState` and
4//! rendering the next view in place. `bootstrap.js` tracks that in
5//! `__virtualUrl` so `location.href` reads correctly, but nothing on the Rust
6//! side ever looked at it, so the page had moved on while `page.url()` still
7//! reported the old document. To a CDP client that is a click that did nothing,
8//! and `tools/ab/clicklocal.mjs` reports it as `spa route: FAILED`.
9//!
10//! Ported from fork commit `d7dca7a`. Kept out of `page.rs` so an upstream
11//! rewrite of that file does not touch it; Rust allows an inherent impl in any
12//! module of the defining crate, so the call site still reads
13//! `self.sync_virtual_url()`.
14
15use url::Url;
16
17use crate::page::Page;
18
19impl Page {
20    /// Adopt a URL the page routed to itself, without fetching anything.
21    ///
22    /// Returns whether the URL changed.
23    pub fn sync_virtual_url(&mut self) -> bool {
24        let Some(js) = self.js.as_mut() else {
25            return false;
26        };
27        let Ok(virtual_url) = js.evaluate("globalThis.__virtualUrl || ''") else {
28            return false;
29        };
30        let Some(virtual_url) = virtual_url.as_str().filter(|url| !url.is_empty()) else {
31            return false;
32        };
33        let Ok(parsed) = Url::parse(virtual_url) else {
34            return false;
35        };
36        if self.url.as_ref() == Some(&parsed) {
37            return false;
38        }
39        self.url = Some(parsed);
40        self.push_history(self.url_string());
41        true
42    }
43}
