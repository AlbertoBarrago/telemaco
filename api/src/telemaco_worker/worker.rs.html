2use std::sync::Arc;
3
4use telemaco_browser::{BrowserContext, Page};
5use serde::{Deserialize, Serialize};
6use tokio::io::{AsyncBufReadExt, AsyncWriteExt, BufReader};
7
8#[derive(Debug, Deserialize)]
9#[serde(tag = "cmd")]
10enum WorkerCommand {
11    #[serde(rename = "navigate")]
12    Navigate { url: String },
13    #[serde(rename = "evaluate")]
14    Evaluate { expression: String },
15    #[serde(rename = "title")]
16    Title,
17    #[serde(rename = "dump_html")]
18    DumpHtml,
19    #[serde(rename = "dump_text")]
20    DumpText,
21    #[serde(rename = "shutdown")]
22    Shutdown,
23}
24
25#[derive(Debug, Serialize)]
26struct WorkerResponse {
27    ok: bool,
28    #[serde(skip_serializing_if = "Option::is_none")]
29    result: Option<serde_json::Value>,
30    #[serde(skip_serializing_if = "Option::is_none")]
31    error: Option<String>,
32}
33
34impl WorkerResponse {
35    fn success(result: serde_json::Value) -> Self {
36        WorkerResponse { ok: true, result: Some(result), error: None }
37    }
38    fn error(msg: String) -> Self {
39        WorkerResponse { ok: false, result: None, error: Some(msg) }
40    }
41}

43#[tokio::main(flavor = "current_thread")]
44async fn main() {
45    tracing_subscriber::fmt()
46        .with_env_filter("warn")
47        .with_writer(std::io::stderr)
48        .init();
49
50    let proxy = std::env::var("TELEMACO_PROXY")
51        .ok()
52        .map(|value| value.trim().to_string())
53        .filter(|value| !value.is_empty());
54    let stealth = std::env::var("TELEMACO_STEALTH")
55        .map(|v| matches!(v.trim(), "1" | "true" | "yes" | "on"))
56        .unwrap_or(false);
57    let obey_robots = std::env::var("TELEMACO_OBEY_ROBOTS")
58        .map(|v| matches!(v.trim(), "1" | "true" | "yes" | "on"))
59        .unwrap_or(false);
60    let mut context = BrowserContext::with_options("worker".to_string(), proxy, stealth);
61    context.obey_robots = obey_robots;
62    let context = Arc::new(context);
63    let mut page = Page::new("page-1".to_string(), context);
64
65    let stdin = tokio::io::stdin();
66    let mut stdout = tokio::io::stdout();
67    let mut reader = BufReader::new(stdin);
68    let mut line = String::new();
69
70    loop {
71        line.clear();
72        match reader.read_line(&mut line).await {
73            Ok(0) => break,
74            Ok(_) => {}
75            Err(e) => {
76                eprintln!("Worker stdin error: {}", e);
77                break;
78            }
79        }
80
81        let trimmed = line.trim();
82        if trimmed.is_empty() {
83            continue;
84        }
85
86        let cmd: WorkerCommand = match serde_json::from_str(trimmed) {
87            Ok(c) => c,
88            Err(e) => {
89                let resp = WorkerResponse::error(format!("Invalid command: {}", e));
90                let mut out = serde_json::to_string(&resp).unwrap();
91                out.push('\n');
92                let _ = stdout.write_all(out.as_bytes()).await;
93                let _ = stdout.flush().await;
94                continue;
95            }
96        };
97
98        let resp = match cmd {
99            WorkerCommand::Navigate { url } => {
100                match page.navigate(&url).await {
101                    Ok(()) => WorkerResponse::success(serde_json::json!({
102                        "title": page.title,
103                        "url": page.url_string(),
104                    })),
105                    Err(e) => WorkerResponse::error(e.to_string()),
106                }
107            }
108            WorkerCommand::Evaluate { expression } => {
109                // Await promise-returning expressions so async IIFEs resolve
110                // before serialization. Previously the sync path serialized an
111                // unresolved Promise as `{}`, making single-invocation flows
112                // that call async app APIs impossible (issue #693). A 30s cap
113                // matches the CDP await timeout so a never-settling promise
114                // cannot hang the worker.
115                let result = match page
116                    .evaluate_for_cdp_with_timeout(&expression, true, true, 30_000)
117                    .await
118                {
119                    Ok(info) => match info.value {
120                        Some(v) => v,
121                        None => serde_json::Value::String(info.description),
122                    },
123                    Err(_) => serde_json::Value::Null,
124                };
125                WorkerResponse::success(result)
126            }
127            WorkerCommand::Title => {
128                WorkerResponse::success(serde_json::json!(page.title))
129            }
130            WorkerCommand::DumpHtml => {
131                let html = page.with_dom(|dom| {
132                    if let Ok(Some(html_node)) = dom.query_selector("html") {
133                        dom.outer_html(html_node)
134                    } else {
135                        dom.inner_html(dom.document())
136                    }
137                }).unwrap_or_default();
138                WorkerResponse::success(serde_json::json!(html))
139            }
140            WorkerCommand::DumpText => {
141                let text = page.with_dom(|dom| {
142                    if let Ok(Some(body)) = dom.query_selector("body") {
143                        dom.text_content(body)
144                    } else {
145                        String::new()
146                    }
147                }).unwrap_or_default();
148                WorkerResponse::success(serde_json::json!(text))
149            }
150            WorkerCommand::Shutdown => {
151                let resp = WorkerResponse::success(serde_json::json!("bye"));
152                let mut out = serde_json::to_string(&resp).unwrap();
153                out.push('\n');
154                let _ = stdout.write_all(out.as_bytes()).await;
155                let _ = stdout.flush().await;
156                break;
157            }
158        };
159
160        let mut out = serde_json::to_string(&resp).unwrap();
161        out.push('\n');
162        let _ = stdout.write_all(out.as_bytes()).await;
163        let _ = stdout.flush().await;
164    }
165}
