17// Resolve the cookie jar for a Network request: prefer the session's page jar,
18// fall back to the default browser context. Puppeteer and Playwright both call
19// Network.setCookie/getCookies/deleteCookies BEFORE attaching to a target —
20// requiring a session would break those flows (Storage.* already mirrors this).
21fn cookie_jar_for<'a>(ctx: &'a CdpContext, session_id: &Option<String>) -> &'a Arc<CookieJar> {
22    ctx.get_session_page(session_id)
23        .map(|p| &p.context.cookie_jar)
24        .unwrap_or(&ctx.default_context.cookie_jar)
25}

27pub async fn handle(
28    method: &str,
29    params: &Value,
30    ctx: &mut CdpContext,
31    session_id: &Option<String>,
32) -> Result<Value, String> {
33    match method {
34        "enable" => Ok(json!({})),
35        "disable" => {
36            if let Some(page) = ctx.get_session_page_mut(session_id) {
37                page.clear_response_bodies();
38            } else {
39                for page in &mut ctx.pages {
40                    page.clear_response_bodies();
41                }
42            }
43            Ok(json!({}))
44        }
45        "setExtraHTTPHeaders" => {
46            let headers = params.get("headers").and_then(|v| v.as_object());
47            if let Some(page) = ctx.get_session_page(session_id) {
48                if let Some(headers) = headers {
49                    let header_map: std::collections::HashMap<String, String> = headers
50                        .iter()
51                        .map(|(k, v)| (k.clone(), v.as_str().unwrap_or("").to_string()))
52                        .collect();
53                    page.http_client.set_extra_headers(header_map).await;
54                }
55            }
56            Ok(json!({}))
57        }
58        "setUserAgentOverride" => {
59            let ua = params.get("userAgent").and_then(|v| v.as_str()).unwrap_or("");
60            if let Some(page) = ctx.get_session_page(session_id) {
61                page.http_client.set_user_agent(ua).await;
62            }
63            Ok(json!({}))
64        }
65        "getCookies" | "getAllCookies" => {
66            let cookies = cookie_jar_for(ctx, session_id).get_all_cookies();
67            let cdp_cookies: Vec<Value> = cookies.iter().map(cookie_info_to_cdp_json).collect();
68            Ok(json!({ "cookies": cdp_cookies }))
69        }
70        "setCookie" => {
71            let cookie = parse_cdp_cookie(params)
72                .ok_or("setCookie: missing required name/domain (or url)")?;
73            cookie_jar_for(ctx, session_id).set_cookies_from_cdp(vec![cookie]);
74            Ok(json!({ "success": true }))
75        }
76        "setCookies" => {
77            if let Some(cookies) = params.get("cookies").and_then(|v| v.as_array()) {
78                let parsed: Vec<_> = cookies.iter().filter_map(parse_cdp_cookie).collect();
79                cookie_jar_for(ctx, session_id).set_cookies_from_cdp(parsed);
80            }
81            Ok(json!({}))
82        }
83        "deleteCookies" => {
84            if let Some(filter) = parse_delete_cookies_params(params) {
85                cookie_jar_for(ctx, session_id).delete_cookies_filtered(
86                    &filter.name,
87                    &filter.domain,
88                    filter.path.as_deref(),
89                );
90            }
91            Ok(json!({}))
92        }
93        "clearBrowserCookies" => {
94            cookie_jar_for(ctx, session_id).clear();
95            Ok(json!({}))
96        }
97        "setCacheDisabled" => Ok(json!({})),
98        "setRequestInterception" => Ok(json!({})),
99        "setBlockedURLs" => {
100            let patterns = params
101                .get("urls")
102                .and_then(|value| value.as_array())
103                .map(|values| {
104                    values
105                        .iter()
106                        .filter_map(|value| value.as_str().map(ToString::to_string))
107                        .collect::<Vec<_>>()
108                })
109                .unwrap_or_default();
110
111            if let Some(page) = ctx.get_session_page_mut(session_id) {
112                page.set_blocked_urls(patterns);
113            } else {
114                for page in &mut ctx.pages {
115                    page.set_blocked_urls(patterns.clone());
116                }
117            }
118            Ok(json!({}))
119        }
120        "getResponseBody" => {
121            let request_id = params
122                .get("requestId")
123                .and_then(|v| v.as_str())
124                .ok_or("Network.getResponseBody requires requestId")?;
125
126            let body = if let Some(page) = ctx.get_session_page(session_id) {
127                page.get_response_body(request_id)
128            } else {
129                ctx.pages.iter().find_map(|page| page.get_response_body(request_id))
130            };
131
132            match body {
133                Some(body) => Ok(json!({
134                    "body": body.body,
135                    "base64Encoded": body.base64_encoded,
136                })),
137                None => Err(format!("No response body found for requestId {}", request_id)),
138            }
139        }
140        _ => Err(format!("Unknown Network method: {}", method)),
141    }
142}

406pub(crate) fn cookie_info_to_cdp_json(c: &telemaco_net::CookieInfo) -> Value {
407    let expires = c.expires.unwrap_or(SESSION_COOKIE_EXPIRES);
408    let session = c.expires.is_none();
409    let same_site = if c.same_site.is_empty() {
410        DEFAULT_SAME_SITE
411    } else {
412        c.same_site.as_str()
413    };
414    json!({
415        "name": c.name,
416        "value": c.value,
417        "domain": c.domain,
418        "path": c.path,
419        "expires": expires,
420        "size": c.name.len() + c.value.len(),
421        "httpOnly": c.http_only,
422        "secure": c.secure,
423        "session": session,
424        "sameSite": same_site,
425        "sameParty": false,
426        "sourceScheme": if c.secure { SOURCE_SCHEME_SECURE } else { SOURCE_SCHEME_NONSECURE },
427        "sourcePort": if c.secure { DEFAULT_SECURE_PORT } else { DEFAULT_INSECURE_PORT },
428        "priority": "Medium",
429    })
