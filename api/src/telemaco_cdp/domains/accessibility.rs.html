26pub async fn handle(
27    method: &str,
28    _params: &Value,
29    ctx: &mut CdpContext,
30    session_id: &Option<String>,
31) -> Result<Value, String> {
32    match method {
33        "enable" => Ok(json!({})),
34        "getFullAXTree" => {
35            let page = ctx
36                .get_session_page(session_id)
37                .ok_or("No page")?;
38            let nodes = page
39                .with_dom(|dom| build_ax_nodes(dom))
40                .unwrap_or_default();
41            Ok(json!({ "nodes": nodes }))
42        }
43        _ => Ok(json!({})),
44    }
45}

47/// Walk the full DOM tree and produce CDP Accessibility AXNode array.
48fn build_ax_nodes(dom: &DomTree) -> Vec<Value> {
49    let mut nodes: Vec<Value> = Vec::new();
50    let mut id_counter: u32 = 0;
51    // Map DOM NodeId → AX string id, populated only for nodes actually in the AX tree
52    let mut dom_to_ax: std::collections::HashMap<u32, String> = std::collections::HashMap::new();
53
54    let document = dom.document();
55
56    // Collect all DOM nodes in tree order (root + descendants)
57    let mut all_dom_ids: Vec<NodeId> = vec![document];
58    all_dom_ids.extend(dom.descendants(document));
59
60    // First pass: assign AX IDs only to nodes that will produce an AX node
61    let mut eligible: Vec<NodeId> = Vec::new();
62    for dom_id in &all_dom_ids {
63        // Quick check without full build_ax_node (just role check to avoid borrow issues)
64        if let Some(node) = dom.get_node(*dom_id) {
65            let role = map_role(&node.data);
66            if !role.is_empty() {
67                id_counter += 1;
68                dom_to_ax.insert(dom_id.raw(), id_counter.to_string());
69                eligible.push(*dom_id);
70            }
71        }
72    }
73
74    // Second pass: build AXNode for eligible nodes
75    for dom_id in &eligible {
76        if let Some(ax) = build_ax_node(dom, *dom_id, &dom_to_ax) {
77            nodes.push(ax);
78        }
79    }
80
81    nodes
82}

84fn build_ax_node(
85    dom: &DomTree,
86    node_id: NodeId,
87    dom_to_ax: &std::collections::HashMap<u32, String>,
88) -> Option<Value> {
89    let node = dom.get_node(node_id)?;
90    let ax_id = dom_to_ax.get(&node_id.raw())?.clone();
91
92    let role = map_role(&node.data);
93    // Skip non-relevant nodes (Document, Doctype, Comment, PI)
94    if role.is_empty() {
95        return None;
96    }
97
98    let name = compute_name(dom, &node);
99    let value = compute_value(dom, &node);
100    let properties = compute_properties(dom, &node);
101
102    let child_ids: Vec<String> = dom
103        .children(node_id)
104        .iter()
105        .filter_map(|child_id| dom_to_ax.get(&child_id.raw()).cloned())
106        .collect();
107
108    // Resolve parentId — walk DOM ancestors until we find one in the AX tree
109    let parent_id: Option<String> = {
110        let mut current = node_id;
111        let mut result = None;
112        loop {
113            let next_parent = dom.with_node(current, |n| n.parent).flatten();
114            match next_parent {
115                Some(pid) => {
116                    if let Some(ax_pid) = dom_to_ax.get(&pid.raw()) {
117                        result = Some(ax_pid.clone());
118                        break;
119                    }
120                    current = pid;
121                }
122                None => break,
123            }
124        }
125        result
126    };
127
128    // Build node with only non-empty optional fields (per CDP spec, optional fields should be omitted when empty)
129    let mut ax_node = json!({
130        "nodeId": ax_id,
131        "ignored": false,
132        "role": ax_value_role(role),
133    });
134
135    if let Some(ref pid) = parent_id {
136        ax_node.as_object_mut().unwrap().insert("parentId".into(), json!(pid));
137    }
138    if let Some(ref n) = name {
139        ax_node.as_object_mut().unwrap().insert("name".into(), json!(ax_value_string(n)));
140    }
141    if let Some(ref v) = value {
142        ax_node.as_object_mut().unwrap().insert("value".into(), json!(ax_value_string(v)));
143    }
144    if !properties.is_empty() {
145        ax_node.as_object_mut().unwrap().insert("properties".into(), json!(properties));
146    }
147    if !child_ids.is_empty() {
148        ax_node.as_object_mut().unwrap().insert("childIds".into(), json!(child_ids));
149    }
150    ax_node.as_object_mut().unwrap().insert("backendDOMNodeId".into(), json!(node_id.raw()));
151
152    Some(ax_node)
153}

155/// Map HTML element tag to ARIA role value.
156fn map_role(data: &NodeData) -> &'static str {
157    match data {
158        NodeData::Document => "RootWebArea",
159        NodeData::Element { name, attrs, .. } => {
160            let tag = name.local.as_ref();
161
162            // Check explicit role attribute first
163            if let Some(role_attr) = attrs.iter().find(|a| a.name.local.as_ref() == "role") {
164                return match role_attr.value.as_str() {
165                    "button" => "button",
166                    "link" => "link",
167                    "heading" => "heading",
168                    "textbox" | "searchbox" => "textbox",
169                    "checkbox" => "checkbox",
170                    "radio" => "radio",
171                    "listbox" => "listbox",
172                    "combobox" => "combobox",
173                    "list" => "list",
174                    "listitem" => "listitem",
175                    "navigation" => "navigation",
176                    "banner" => "banner",
177                    "main" => "main",
178                    "complementary" => "complementary",
179                    "contentinfo" => "contentinfo",
180                    "form" => "form",
181                    "table" => "table",
182                    "row" => "row",
183                    "cell" | "gridcell" => "cell",
184                    "img" => "image",
185                    "dialog" => "dialog",
186                    "alert" => "alert",
187                    "tab" => "tab",
188                    "tablist" => "tablist",
189                    "tabpanel" => "tabpanel",
190                    "menu" => "menu",
191                    "menuitem" => "menuitem",
192                    "toolbar" => "toolbar",
193                    "separator" => "separator",
194                    "presentation" | "none" => {
195                        // presentation/none roles get the role but content is still in tree
196                        "presentation"
197                    }
198                    _ => "generic",
199                };
200            }
201
202            match tag {
203                "a" => {
204                    if attrs.iter().any(|a| a.name.local.as_ref() == "href") {
205                        "link"
206                    } else {
207                        "generic"
208                    }
209                }
210                "button" | "summary" => "button",
211                "input" => {
212                    let type_attr = attrs
213                        .iter()
214                        .find(|a| a.name.local.as_ref() == "type")
215                        .map(|a| a.value.as_str())
216                        .unwrap_or("text");
217                    match type_attr {
218                        "submit" | "reset" | "button" | "image" => "button",
219                        "checkbox" => "checkbox",
220                        "radio" => "radio",
221                        "range" => "slider",
222                        "number" => "spinbutton",
223                        "search" => "searchbox",
224                        _ => "textbox",
225                    }
226                }
227                "textarea" => "textbox",
228                "select" => {
229                    if attrs.iter().any(|a| {
230                        a.name.local.as_ref() == "multiple"
231                            || a.name.local.as_ref() == "size"
232                    }) {
233                        "listbox"
234                    } else {
235                        "combobox"
236                    }
237                }
238                "h1" | "h2" | "h3" | "h4" | "h5" | "h6" => "heading",
239                "img" | "svg" => "image",
240                "ul" | "ol" | "menu" => "list",
241                "li" => "listitem",
242                "table" => "table",
243                "tr" => "row",
244                "td" | "th" => "cell",
245                "nav" => "navigation",
246                "header" => "banner",
247                "main" => "main",
248                "footer" => "contentinfo",
249                "form" => "form",
250                "dialog" => "dialog",
251                "hr" => "separator",
252                "label" => "LabelText",
253                "article" => "article",
254                "aside" => "complementary",
255                "section" => "region",
256                "figure" => "figure",
257                "figcaption" => "StaticText",
258                "p" | "div" | "span" | "pre" | "blockquote" | "code"
259                | "em" | "strong" | "b" | "i" | "u" | "s" | "small"
260                | "sub" | "sup" | "mark" | "del" | "ins" => "generic",
261                "iframe" => "Iframe",
262                _ => "generic",
263            }
264        }
265        NodeData::Text { .. } => "StaticText",
266        NodeData::Doctype { .. } | NodeData::Comment { .. } | NodeData::ProcessingInstruction { .. } => {
267            ""
268        }
269    }
270}

272/// Compute the accessible name for a node.
273fn compute_name(dom: &DomTree, node: &telemaco_dom::Node) -> Option<String> {
274    if let NodeData::Element { attrs, .. } = &node.data {
275        // aria-label takes highest priority
276        if let Some(label) = attrs
277            .iter()
278            .find(|a| a.name.local.as_ref() == "aria-label")
279            .map(|a| a.value.clone())
280        {
281            return Some(label);
282        }
283
284        // aria-labelledby
285        if let Some(labelledby) = attrs
286            .iter()
287            .find(|a| a.name.local.as_ref() == "aria-labelledby")
288        {
289            let ids: Vec<&str> = labelledby.value.split_whitespace().collect();
290            let mut name = String::new();
291            for id_str in ids {
292                if let Some(ref_id) = dom.get_element_by_id(id_str) {
293                    name.push_str(&dom.text_content(ref_id));
294                    name.push(' ');
295                }
296            }
297            let trimmed = name.trim().to_string();
298            if !trimmed.is_empty() {
299                return Some(trimmed);
300            }
301        }
302
303        // alt attribute for images
304        if let Some(alt) = attrs
305            .iter()
306            .find(|a| a.name.local.as_ref() == "alt")
307            .map(|a| a.value.clone())
308        {
309            if !alt.is_empty() {
310                return Some(alt);
311            }
312        }
313
314        // title attribute
315        if let Some(title) = attrs
316            .iter()
317            .find(|a| a.name.local.as_ref() == "title")
318            .map(|a| a.value.clone())
319        {
320            if !title.is_empty() {
321                return Some(title);
322            }
323        }
324
325        // placeholder
326        if let Some(placeholder) = attrs
327            .iter()
328            .find(|a| a.name.local.as_ref() == "placeholder")
329            .map(|a| a.value.clone())
330        {
331            if !placeholder.is_empty() {
332                return Some(placeholder);
333            }
334        }
335    }
336
337    // For text nodes, the name is the text content
338    if let NodeData::Text { contents } = &node.data {
339        let trimmed = contents.trim().to_string();
340        if !trimmed.is_empty() {
341            return Some(trimmed);
342        }
343    }
344
345    None
346}

348/// Compute the accessible value for a node (e.g., current input value).
349fn compute_value(dom: &DomTree, node: &telemaco_dom::Node) -> Option<String> {
350    if let NodeData::Element { name, attrs, .. } = &node.data {
351        let tag = name.local.as_ref();
352        // For native form controls, return the value attribute.
353        if tag == "input" || tag == "textarea" || tag == "select" {
354            return attrs
355                .iter()
356                .find(|a| a.name.local.as_ref() == "value")
357                .map(|a| a.value.clone());
358        }
359
360        if is_content_editing_host(dom, node) {
361            return Some(dom.text_content(node.id));
362        }
363    }
364    None
365}

367fn contenteditable_keyword(node: &telemaco_dom::Node) -> Option<bool> {
368    let NodeData::Element { attrs, .. } = &node.data else {
369        return None;
370    };
371    let value = &attrs
372        .iter()
373        .find(|attr| attr.name.local.as_ref() == "contenteditable")?
374        .value;
375
376    if value.is_empty()
377        || value.eq_ignore_ascii_case("true")
378        || value.eq_ignore_ascii_case("plaintext-only")
379    {
380        Some(true)
381    } else if value.eq_ignore_ascii_case("false") {
382        Some(false)
383    } else {
384        None
385    }
386}

388fn is_effectively_contenteditable(dom: &DomTree, node_id: NodeId) -> bool {
389    let mut current = Some(node_id);
390    while let Some(id) = current {
391        let Some(node) = dom.get_node(id) else {
392            return false;
393        };
394        if let Some(editable) = contenteditable_keyword(&node) {
395            return editable;
396        }
397        current = node.parent;
398    }
399    false
400}
401
402fn is_content_editing_host(dom: &DomTree, node: &telemaco_dom::Node) -> bool {
403    is_effectively_contenteditable(dom, node.id)
404        && node
405            .parent
406            .is_none_or(|parent| !is_effectively_contenteditable(dom, parent))
407}

409/// Compute accessibility properties for a node.
410fn compute_properties(_dom: &DomTree, node: &telemaco_dom::Node) -> Vec<Value> {
411    if let NodeData::Element { name, attrs, .. } = &node.data {
412        let tag = name.local.as_ref();
413        let mut props = Vec::new();
414
415        // focusable
416        let focusable = matches!(
417            tag,
418            "a" | "button" | "input" | "select" | "textarea" | "details" | "summary"
419        ) || attrs.iter().any(|a| {
420            let an = a.name.local.as_ref();
421            an == "tabindex" || an == "contenteditable"
422        });
423        if focusable {
424            props.push(json!({"name": "focusable", "value": ax_value_boolean(true)}));
425        }
426
427        // editable
428        if tag == "input" || tag == "textarea"
429            || attrs
430                .iter()
431                .any(|a| a.name.local.as_ref() == "contenteditable" && a.value != "false")
432        {
433            props.push(json!({"name": "editable", "value": ax_value_boolean(true)}));
434        }
435
436        // checked for checkboxes/radios
437        if attrs
438            .iter()
439            .any(|a| a.name.local.as_ref() == "checked")
440        {
441            props.push(json!({"name": "checked", "value": ax_value_boolean(true)}));
442        }
443
444        // disabled
445        if attrs
446            .iter()
447            .any(|a| a.name.local.as_ref() == "disabled")
448        {
449            props.push(json!({"name": "disabled", "value": ax_value_boolean(true)}));
450        }
451
452        // level for headings
453        if let Some(level) = tag.strip_prefix('h').and_then(|s| s.parse::<u32>().ok()) {
454            if level >= 1 && level <= 6 {
455                props.push(json!({"name": "level", "value": ax_value_integer(level)}));
456            }
457        }
458
459        // required
460        if attrs
461            .iter()
462            .any(|a| a.name.local.as_ref() == "required" || a.name.local.as_ref() == "aria-required")
463        {
464            props.push(json!({"name": "required", "value": ax_value_boolean(true)}));
465        }
466
467        // multiline for textarea
468        if tag == "textarea" {
469            props.push(json!({"name": "multiline", "value": ax_value_boolean(true)}));
470        }
471
472        props
473    } else {
474        Vec::new()
475    }
476}

483    fn ax_node_for<'a>(nodes: &'a [Value], backend_node_id: NodeId) -> &'a Value {
484        nodes
485            .iter()
486            .find(|node| node["backendDOMNodeId"] == backend_node_id.raw())
487            .expect("element is present in the AX tree")
488    }

490    fn assert_ax_value(nodes: &[Value], dom: &DomTree, id: &str, expected: Option<&str>) {
491        let node = ax_node_for(nodes, dom.get_element_by_id(id).unwrap());
492        assert_eq!(
493            node.get("value").and_then(|value| value["value"].as_str()),
494            expected,
495            "unexpected AX value for #{id}"
496        );
