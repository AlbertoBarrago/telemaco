5// Insert `text` at the caret, replacing any non-collapsed selection the way a
6// real browser does when you type over selected text (for example after a
7// triple-click select-all). selectionStart is null during ordinary typing, so
8// the legacy append path is kept when no selection is tracked.
9//
10// The text is embedded as a JSON string literal rather than escaped by hand
11// into single quotes. JSON string syntax is a subset of JavaScript's, so this
12// covers the quote and the backslash of issue #433 and the control characters
13// they left out: a newline inside a single-quoted literal is a syntax error,
14// so the whole snippet was dropped and nothing was inserted. telemaco-mcp
15// already builds its typing snippet this way.
16fn insert_text_js(text: &str) -> String {
17    let literal = serde_json::to_string(text).unwrap_or_else(|_| "\"\"".to_string());
18    format!(
19        "(function() {{\
20            var t = document.activeElement;\
21            if (!t || (t.localName !== 'input' && t.localName !== 'textarea')) return;\
22            var ins = {text};\
23            var v = t.value || '';\
24            var s = t.selectionStart, e = t.selectionEnd;\
25            if (s == null) {{\
26                globalThis.__telemaco_setFieldValue(t, 'value', v + ins);\
27            }} else {{\
28                s = Math.max(0, Math.min(s, v.length));\
29                e = (e == null) ? s : Math.max(0, Math.min(e, v.length));\
30                var lo = Math.min(s, e), hi = Math.max(s, e);\
31                globalThis.__telemaco_setFieldValue(t, 'value', v.slice(0, lo) + ins + v.slice(hi));\
32                var caret = lo + ins.length;\
33                t.setSelectionRange(caret, caret);\
34            }}\
35            t.dispatchEvent(globalThis.__telemaco_markTrusted(new Event('input', {{bubbles:true}})));\
36        }})()",
37        text = literal,
38    )
39}

41// Backspace deletes the selected range when there is one, so the common
42// "triple-click to select-all, then Backspace to clear" pattern works. With a
43// collapsed caret it removes the character before the caret, and with no
44// selection tracked it falls back to trimming the last character (legacy).
45const BACKSPACE_JS: &str = "(function() {\
46    var t = document.activeElement;\
47    if (!t || (t.localName !== 'input' && t.localName !== 'textarea')) return;\
48    var v = t.value || '';\
49    var s = t.selectionStart, e = t.selectionEnd;\
50    if (s == null) {\
51        globalThis.__telemaco_setFieldValue(t, 'value', v.slice(0, -1));\
52    } else {\
53        s = Math.max(0, Math.min(s, v.length));\
54        e = (e == null) ? s : Math.max(0, Math.min(e, v.length));\
55        if (s !== e) {\
56            var lo = Math.min(s, e), hi = Math.max(s, e);\
57            globalThis.__telemaco_setFieldValue(t, 'value', v.slice(0, lo) + v.slice(hi));\
58            t.setSelectionRange(lo, lo);\
59        } else if (s > 0) {\
60            globalThis.__telemaco_setFieldValue(t, 'value', v.slice(0, s - 1) + v.slice(s));\
61            t.setSelectionRange(s - 1, s - 1);\
62        }\
63    }\
64    t.dispatchEvent(globalThis.__telemaco_markTrusted(new Event('input', {bubbles:true})));\
65})()";

67fn mouse_button_code(button: &str) -> u8 {
68    match button {
69        "middle" => 1,
70        "right" => 2,
71        "back" => 3,
72        "forward" => 4,
73        _ => 0,
74    }

77fn mouse_button_mask(button: &str) -> u64 {
78    match button {
79        "right" => 2,
80        "middle" => 4,
81        "back" => 8,
82        "forward" => 16,
83        "none" => 0,
84        _ => 1,
85    }

88fn modifier_flags(modifiers: u64) -> (bool, bool, bool, bool) {
89    // CDP Input.Modifier: Alt=1, Ctrl=2, Meta=4, Shift=8.
90    (
91        modifiers & 1 != 0,
92        modifiers & 2 != 0,
93        modifiers & 4 != 0,
94        modifiers & 8 != 0,
95    )
96}

98pub async fn handle(
99    method: &str,
100    params: &Value,
101    ctx: &mut CdpContext,
102    session_id: &Option<String>,
103) -> Result<Value, String> {
104    match method {
105        "dispatchMouseEvent" => {
106            let event_type = params.get("type").and_then(|v| v.as_str()).unwrap_or("");
107            let x = params.get("x").and_then(|v| v.as_f64()).unwrap_or(0.0);
108            let y = params.get("y").and_then(|v| v.as_f64()).unwrap_or(0.0);
109            let button = params.get("button").and_then(|v| v.as_str()).unwrap_or("left");
110            let button_code = mouse_button_code(button);
111            let buttons = params
112                .get("buttons")
113                .and_then(|v| v.as_u64())
114                .unwrap_or_else(|| mouse_button_mask(button));
115            let click_count = params.get("clickCount").and_then(|v| v.as_u64()).unwrap_or(1);
116            let modifiers = params.get("modifiers").and_then(|v| v.as_u64()).unwrap_or(0);
117            let (alt_key, ctrl_key, meta_key, shift_key) = modifier_flags(modifiers);
118
119            if event_type == "mousePressed" {
120                if let Some(page) = ctx.get_session_page_mut(session_id) {
121                    let code = format!(
122                        "(function() {{\
123                            var target = (document.elementFromPoint && document.elementFromPoint({x},{y})) || globalThis.__telemaco_click_target || document.activeElement || document.body;\
124                            if (!target) return;\
125                            globalThis.__telemaco_click_target = target;\
126                            globalThis.__telemaco_mouse_down = {{target:target,button:{button_code},clickCount:{click_count}}};\
127                            var evt = globalThis.__telemaco_markTrusted(new MouseEvent('mousedown', {{bubbles:true,cancelable:true,view:globalThis,clientX:{x},clientY:{y},button:{button_code},buttons:{buttons},detail:{click_count},altKey:{alt_key},ctrlKey:{ctrl_key},metaKey:{meta_key},shiftKey:{shift_key}}}));\
128                            target.dispatchEvent(evt);\
129                            if (!evt.defaultPrevented) {{\
130                                var ft = target;\
131                                while (ft && !globalThis.__telemaco_isFocusable(ft)) ft = ft.parentElement;\
132                                if (ft && typeof ft.focus === 'function') ft.focus();\
133                            }}\
134                        }})()",
135                        x = x,
136                        y = y,
137                        button_code = button_code,
138                        buttons = buttons,
139                        click_count = click_count,
140                        alt_key = alt_key,
141                        ctrl_key = ctrl_key,
142                        meta_key = meta_key,
143                        shift_key = shift_key,
144                    );
145                    page.evaluate(&code);
146                }
147            } else if event_type == "mouseReleased" {
148                let moved_frame = if let Some(page) = ctx.get_session_page_mut(session_id) {
149                    let code = format!(
150                        "(function() {{\
151                            var target = (document.elementFromPoint && document.elementFromPoint({x},{y})) || globalThis.__telemaco_click_target || document.activeElement || document.body;\
152                            if (!target) return;\
153                            var down = globalThis.__telemaco_mouse_down;\
154                            globalThis.__telemaco_mouse_down = null;\
155                            var evt = globalThis.__telemaco_markTrusted(new MouseEvent('mouseup', {{bubbles:true,cancelable:true,view:globalThis,clientX:{x},clientY:{y},button:{button_code},buttons:0,detail:{click_count},altKey:{alt_key},ctrlKey:{ctrl_key},metaKey:{meta_key},shiftKey:{shift_key}}}));\
156                            target.dispatchEvent(evt);\
157                            if (!down || down.button !== {button_code} || {button_code} !== 0) return;\
158                            var clickTarget = down.target;\
159                            while (clickTarget && clickTarget !== target && !(clickTarget.contains && clickTarget.contains(target))) {{\
160                                clickTarget = clickTarget.parentElement;\
161                            }}\
162                            if (!clickTarget) return;\
163                            var tag = clickTarget.tagName;\
164                            var type = (clickTarget.getAttribute && clickTarget.getAttribute('type') || '').toLowerCase();\
165                            if (globalThis.__telemaco_isDisabled(clickTarget)) return;\
166                            var checkable = tag === 'INPUT' && (type === 'checkbox' || type === 'radio');\
167                            var oldChecked = checkable ? !!clickTarget.checked : false;\
168                            var oldIndeterminate = checkable ? !!clickTarget.indeterminate : false;\
169                            var radioStates = null;\
170                            if (checkable && type === 'radio') {{\
171                                var radioName = clickTarget.getAttribute('name') || '';\
172                                if (radioName) {{\
173                                    var candidates = document.querySelectorAll('input');\
174                                    radioStates = [];\
175                                    for (var ri = 0; ri < candidates.length; ri++) {{\
176                                        var radio = candidates[ri];\
177                                        if ((radio.getAttribute('type') || '').toLowerCase() !== 'radio' || (radio.getAttribute('name') || '') !== radioName || radio.form !== clickTarget.form) continue;\
178                                        radioStates.push([radio, !!radio.checked]);\
179                                        if (radio !== clickTarget) radio.checked = false;\
180                                    }}\
181                                }}\
182                                clickTarget.checked = true;\
183                            }} else if (checkable) {{\
184                                clickTarget.checked = !oldChecked;\
185                                clickTarget.indeterminate = false;\
186                            }}\
187                            var click = globalThis.__telemaco_markTrusted(new MouseEvent('click', {{bubbles:true,cancelable:true,view:globalThis,clientX:{x},clientY:{y},button:0,buttons:0,detail:{click_count},altKey:{alt_key},ctrlKey:{ctrl_key},metaKey:{meta_key},shiftKey:{shift_key}}}));\
188                            var cancelled = !clickTarget.dispatchEvent(click);\
189                            if (cancelled) {{\
190                                if (radioStates) {{\
191                                    for (var rr = 0; rr < radioStates.length; rr++) radioStates[rr][0].checked = radioStates[rr][1];\
192                                }} else if (checkable) {{ clickTarget.checked = oldChecked; clickTarget.indeterminate = oldIndeterminate; }}\
193                                return;\
194                            }}\
195                            if (checkable && clickTarget.checked !== oldChecked) {{\
196                                try {{ clickTarget.dispatchEvent(globalThis.__telemaco_markTrusted(new Event('input', {{bubbles:true}}))); }} catch(e) {{}}\
197                                try {{ clickTarget.dispatchEvent(globalThis.__telemaco_markTrusted(new Event('change', {{bubbles:true}}))); }} catch(e) {{}}\
198                                return;\
199                            }}\
200                            var labelHost = tag === 'LABEL' ? clickTarget : (clickTarget.closest ? clickTarget.closest('label') : null);\
201                            var interactiveHost = globalThis.__telemaco_interactiveHost(clickTarget);\
202                            if (labelHost && !(interactiveHost && labelHost.contains(interactiveHost))) {{\
203                                var ctl = globalThis.__telemaco_labeledControl(labelHost);\
204                                if (ctl && globalThis.__telemaco_isFocusable(ctl) && typeof ctl.focus === 'function') ctl.focus();\
205                                if (ctl && ctl !== clickTarget && globalThis.__telemaco_activateLabel(labelHost, ctl, true)) {{ return; }}\
206                            }}\
207                            var link = clickTarget.closest ? clickTarget.closest('a[href]') : null;\
208                            if (!link && tag === 'A' && clickTarget.getAttribute('href')) link = clickTarget;\
209                            if (link) {{\
210                                var href = link.getAttribute('href');\
211                                if (href && !href.startsWith('#') && !href.startsWith('javascript:')) location.assign(href);\
212                            }} else if (tag === 'BUTTON' && type !== 'button' && type !== 'reset') {{\
213                                var form = clickTarget.closest ? clickTarget.closest('form') : null;\
214                                if (form) {{ try {{ if (typeof form.requestSubmit === 'function') {{ form.requestSubmit(clickTarget); }} else {{ form.submit(clickTarget); }} }} catch(e) {{}} }}\
215                            }} else if (tag === 'INPUT' && (type === 'submit' || type === 'image')) {{\
216                                var form2 = clickTarget.closest ? clickTarget.closest('form') : null;\
217                                if (form2) {{ try {{ if (typeof form2.requestSubmit === 'function') {{ form2.requestSubmit(clickTarget); }} else {{ form2.submit(clickTarget); }} }} catch(e) {{}} }}\
218                            }} else if ({click_count} >= 3 && (tag === 'INPUT' || tag === 'TEXTAREA')) {{\
219                                var len = clickTarget.value ? clickTarget.value.length : 0;\
220                                if (clickTarget.setSelectionRange) clickTarget.setSelectionRange(0, len);\
221                                else {{ clickTarget.selectionStart = 0; clickTarget.selectionEnd = len; }}\
222                            }}\
223                        }})()",
224                        x = x,
225                        y = y,
226                        button_code = button_code,
227                        click_count = click_count,
228                        alt_key = alt_key,
229                        ctrl_key = ctrl_key,
230                        meta_key = meta_key,
231                        shift_key = shift_key,
232                    );
233                    page.evaluate(&code);
234                    let moved = page
235                        .process_pending_navigation()
236                        .await
237                        .map_err(|e| e.to_string())?;
238                    // Fork: a single page app answers a click by routing itself,
239                    // with no document fetch. The client still has to be told the
240                    // frame moved, or the click looks like it did nothing.
241                    if moved {
242                        let url = page.url_string();
243                        let frame_id = page.frame_id.clone();
244                        Some((page.id.clone(), frame_id, url))
245                    } else {
246                        None
247                    }
248                } else {
249                    None
250                };
251                if let Some((page_id, frame_id, url)) = moved_frame {
252                    let loader_id = ctx
253                        .current_loader_ids
254                        .get(&page_id)
255                        .cloned()
256                        .unwrap_or_else(|| format!("loader-blank-{page_id}"));
257                    ctx.pending_events.push(crate::types::CdpEvent {
258                        method: "Page.frameNavigated".into(),
259                        params: json!({
260                            "frame": crate::domains::page::frame_value(
261                                &frame_id,
262                                None,
263                                &loader_id,
264                                &url,
265                                "text/html",
266                            ),
267                            "type": "Navigation",
268                        }),
269                        session_id: Some(session_id.clone().unwrap_or_default()),
270                    });
271                }
272            } else if event_type == "mouseWheel" {
273                let delta_x = params.get("deltaX").and_then(|v| v.as_f64()).unwrap_or(0.0);
274                let delta_y = params.get("deltaY").and_then(|v| v.as_f64()).unwrap_or(0.0);
275                if let Some(page) = ctx.get_session_page_mut(session_id) {
276                    let code = format!(
277                        "(function() {{\
278                            var target = (document.elementFromPoint && document.elementFromPoint({x},{y})) || document.body || document.documentElement;\
279                            if (!target) return;\
280                            var wheel = globalThis.__telemaco_markTrusted(new WheelEvent('wheel', {{bubbles:true,cancelable:true,view:globalThis,clientX:{x},clientY:{y},deltaX:{delta_x},deltaY:{delta_y},deltaMode:0,altKey:{alt_key},ctrlKey:{ctrl_key},metaKey:{meta_key},shiftKey:{shift_key}}}));\
281                            if (!target.dispatchEvent(wheel)) return;\
282                            var dx = {delta_x}, dy = {delta_y};\
283                            var root = document.scrollingElement || document.documentElement || document.body;\
284                            var scrollTarget = null;\
285                            var el = target;\
286                            while (el && el.nodeType === 1 && el !== root && el !== document.body && el !== document.documentElement) {{\
287                                var maxX = Math.max(0, (el.scrollWidth || 0) - (el.clientWidth || 0));\
288                                var maxY = Math.max(0, (el.scrollHeight || 0) - (el.clientHeight || 0));\
289                                var style = null;\
290                                try {{ style = getComputedStyle(el); }} catch (_e) {{}}\
291                                var ox = style ? (style.overflowX || style.overflow || '') : '';\
292                                var oy = style ? (style.overflowY || style.overflow || '') : '';\
293                                var allowX = ox === 'auto' || ox === 'scroll' || ox === 'overlay';\
294                                var allowY = oy === 'auto' || oy === 'scroll' || oy === 'overlay';\
295                                var consumesX = allowX && ((dx > 0 && el.scrollLeft < maxX) || (dx < 0 && el.scrollLeft > 0));\
296                                var consumesY = allowY && ((dy > 0 && el.scrollTop < maxY) || (dy < 0 && el.scrollTop > 0));\
297                                if (consumesX || consumesY) {{ scrollTarget = el; break; }}\
298                                el = el.parentElement;\
299                            }}\
300                            if (!scrollTarget) scrollTarget = root;\
301                            if (scrollTarget === root && root && typeof root.scrollBy === 'function') {{\
302                                var beforeX = root.scrollLeft, beforeY = root.scrollTop;\
303                                root.scrollBy(dx, dy);\
304                                if (root.scrollLeft !== beforeX || root.scrollTop !== beforeY) setTimeout(function() {{\
305                                    try {{ document.dispatchEvent(new Event('scroll', {{bubbles:false}})); }} catch (_e) {{}}\
306                                    try {{ globalThis.dispatchEvent(new Event('scroll', {{bubbles:false}})); }} catch (_e) {{}}\
307                                }}, 0);\
308                            }} else if (scrollTarget && typeof scrollTarget.scrollBy === 'function') scrollTarget.scrollBy(dx, dy);\
309                        }})()",
310                        x = x,
311                        y = y,
312                        delta_x = delta_x,
313                        delta_y = delta_y,
314                        alt_key = alt_key,
315                        ctrl_key = ctrl_key,
316                        meta_key = meta_key,
317                        shift_key = shift_key,
318                    );
319                    page.evaluate(&code);
320                }
321            }
322
323            Ok(json!({}))
324        }
325        "dispatchKeyEvent" => {
326            let event_type = params.get("type").and_then(|v| v.as_str()).unwrap_or("");
327            let key = params.get("key").and_then(|v| v.as_str()).unwrap_or("");
328            let code = params.get("code").and_then(|v| v.as_str()).unwrap_or("");
329            let text = params.get("text").and_then(|v| v.as_str()).unwrap_or("");
330
331            if let Some(page) = ctx.get_session_page_mut(session_id) {
332                match event_type {
333                    "keyDown" | "rawKeyDown" => {
334                        let js = format!(
335                            "(function() {{\
336                                var target = document.activeElement || document.body;\
337                                var evt = globalThis.__telemaco_markTrusted(new KeyboardEvent('keydown', {{bubbles:true,cancelable:true,key:'{key}',code:'{code}'}}));\
338                                return target.dispatchEvent(evt);\
339                            }})()",
340                            // Escape backslash BEFORE single-quote (as the text
341                            // path below does) so a key like "\" — Chrome's
342                            // backslash key — doesn't escape the closing quote
343                            // and produce a syntax error that drops the event.
344                            key = key.replace('\\', "\\\\").replace('\'', "\\'"),
345                            code = code.replace('\\', "\\\\").replace('\'', "\\'"),
346                        );
347                        // dispatchEvent returns false when a listener cancels
348                        // the keydown; a real browser then neither inserts the
349                        // key's text nor fires keypress, and Puppeteer typing
350                        // must observe the same rule.
351                        let allowed = page.evaluate(&js).as_bool().unwrap_or(true);
352
353                        if !text.is_empty() && text != "\r" && text != "\n" && allowed {
354                            page.evaluate(&insert_text_js(text));
355                        }
356
357                        if key == "Enter" {
358                            // In a textarea Enter inserts a newline; in input fields
359                            // it submits the containing form. Real Chrome distinguishes
360                            // these two and we should too: previously every Enter tried
361                            // to submit the nearest form even from a textarea.
362                            let js = "(function() {\
363                                var target = document.activeElement;\
364                                if (!target) return;\
365                                target.dispatchEvent(globalThis.__telemaco_markTrusted(new KeyboardEvent('keypress', {bubbles:true,key:'Enter',code:'Enter'})));\
366                                if (target.localName === 'textarea') {\
367                                    globalThis.__telemaco_setFieldValue(target, 'value', (target.value || '') + '\\n');\
368                                    target.dispatchEvent(globalThis.__telemaco_markTrusted(new Event('input', {bubbles:true})));\
369                                } else {\
370                                    var form = target.form || (target.closest && target.closest('form'));\
371                                    if (form) {{ try {{ if (typeof form.requestSubmit === 'function') {{ form.requestSubmit(); }} else {{ form.submit(); }} }} catch(e) {{}} }}\
372                                }\
373                            })()";
374                            page.evaluate(js);
375                        }
376
377                        if key == "Backspace" {
378                            page.evaluate(BACKSPACE_JS);
379                        }
380                    }
381                    "keyUp" => {
382                        let js = format!(
383                            "(function() {{\
384                                var target = document.activeElement || document.body;\
385                                var evt = globalThis.__telemaco_markTrusted(new KeyboardEvent('keyup', {{bubbles:true,key:'{key}',code:'{code}'}}));\
386                                target.dispatchEvent(evt);\
387                            }})()",
388                            key = key.replace('\\', "\\\\").replace('\'', "\\'"),
389                            code = code.replace('\\', "\\\\").replace('\'', "\\'"),
390                        );
391                        page.evaluate(&js);
392                    }
393                    "char" => {
394                        if !text.is_empty() {
395                            // A printable character still surfaces as keydown,
396                            // keypress and keyup the way a physical key would:
397                            // shells that render typed text from document-level
398                            // keydown handlers (terminal widgets with no input
399                            // element) only ever see the page through those
400                            // events. A preventDefault on keydown suppresses the
401                            // insertion and the keypress, exactly as in Chrome;
402                            // keyup fires regardless.
403                            let chars: Vec<char> = text.chars().collect();
404                            if chars.len() == 1 && !chars[0].is_control() {
405                                let key = chars[0].to_string();
406                                let code = match chars[0] {
407                                    'a'..='z' => format!("Key{}", chars[0].to_ascii_uppercase()),
408                                    'A'..='Z' => format!("Key{}", chars[0]),
409                                    '0'..='9' => format!("Digit{}", chars[0]),
410                                    ' ' => "Space".to_string(),
411                                    _ => String::new(),
412                                };
413                                // JSON string literals embed cleanly (the same
414                                // trick insert_text_js uses), so printable
415                                // punctuation like a quote can't break out.
416                                let key_lit =
417                                    serde_json::to_string(&key).unwrap_or_else(|_| "\"\"".to_string());
418                                let code_lit =
419                                    serde_json::to_string(&code).unwrap_or_else(|_| "\"\"".to_string());
420                                let js = format!(
421                                    "(function() {{\
422                                        var target = document.activeElement || document.body;\
423                                        var evt = globalThis.__telemaco_markTrusted(new KeyboardEvent('keydown', {{bubbles:true,cancelable:true,key:{key},code:{code}}}));\
424                                        return target.dispatchEvent(evt);\
425                                    }})()",
426                                    key = key_lit,
427                                    code = code_lit,
428                                );
429                                let allowed = page.evaluate(&js).as_bool().unwrap_or(true);
430                                if allowed {
431                                    page.evaluate(&insert_text_js(text));
432                                    let js = format!(
433                                        "(function() {{\
434                                            var target = document.activeElement || document.body;\
435                                            target.dispatchEvent(globalThis.__telemaco_markTrusted(new KeyboardEvent('keypress', {{bubbles:true,cancelable:true,key:{key},code:{code}}})));\
436                                            target.dispatchEvent(globalThis.__telemaco_markTrusted(new KeyboardEvent('keyup', {{bubbles:true,key:{key},code:{code}}})));\
437                                        }})()",
438                                        key = key_lit,
439                                        code = code_lit,
440                                    );
441                                    page.evaluate(&js);
442                                } else {
443                                    let js = format!(
444                                        "(function() {{\
445                                            var target = document.activeElement || document.body;\
446                                            target.dispatchEvent(globalThis.__telemaco_markTrusted(new KeyboardEvent('keyup', {{bubbles:true,key:{key},code:{code}}})));\
447                                        }})()",
448                                        key = key_lit,
449                                        code = code_lit,
450                                    );
451                                    page.evaluate(&js);
452                                }
453                            } else {
454                                page.evaluate(&insert_text_js(text));
455                            }
456                            // Pump event loop so Angular change detection picks up the input
457                            page.settle(50).await;
458                        }
459                    }
460                    _ => {}
461                }
462            }
463
464            Ok(json!({}))
465        }
466        "dispatchTouchEvent" => Ok(json!({})),
467        "setIgnoreInputEvents" => Ok(json!({})),
468        _ => Err(format!("Unknown Input method: {}", method)),
469    }
470}
