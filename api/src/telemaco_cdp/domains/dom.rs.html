7/// Resolve a DOM `nodeId` from CDP params. Honors `nodeId`, `backendNodeId`,
8/// and `objectId` in that order. Playwright commonly passes only `objectId`
9/// (returned by a prior `DOM.resolveNode`); without this fallback those
10/// requests silently default to node 0 and click the wrong element.
11fn resolve_node_id(page: &mut Page, params: &Value) -> Result<u64, String> {
12    if let Some(nid) = params.get("nodeId").and_then(|v| v.as_u64()) {
13        return Ok(nid);
14    }
15    if let Some(nid) = params.get("backendNodeId").and_then(|v| v.as_u64()) {
16        return Ok(nid);
17    }
18    if let Some(oid) = params.get("objectId").and_then(|v| v.as_str()) {
19        let code = format!(
20            "(function() {{ var o = globalThis.__telemaco_objects && globalThis.__telemaco_objects[{}]; \
21             return (o && typeof o._nid === 'number') ? o._nid : -1; }})()",
22            crate::util::object_id_literal(oid)
23        );
24        let result = page.evaluate(&code);
25        let nid = result.as_f64().map(|n| n as i64).unwrap_or(-1);
26        if nid < 0 {
27            return Err(format!("objectId {oid} could not be resolved to a node"));
28        }
29        return Ok(nid as u64);
30    }
31    Err("nodeId, backendNodeId, or objectId required".to_string())
32}

34/// Standard base64 (with padding). Used to ferry file bytes to the JS layer for
35/// DOM.setFileInputFiles without pulling in a dependency.
36fn encode_base64(input: &[u8]) -> String {
37    const T: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
38    let mut out = String::with_capacity((input.len() + 2) / 3 * 4);
39    for chunk in input.chunks(3) {
40        let b0 = chunk[0] as u32;
41        let b1 = *chunk.get(1).unwrap_or(&0) as u32;
42        let b2 = *chunk.get(2).unwrap_or(&0) as u32;
43        let n = (b0 << 16) | (b1 << 8) | b2;
44        out.push(T[((n >> 18) & 63) as usize] as char);
45        out.push(T[((n >> 12) & 63) as usize] as char);
46        out.push(if chunk.len() > 1 { T[((n >> 6) & 63) as usize] as char } else { '=' });
47        out.push(if chunk.len() > 2 { T[(n & 63) as usize] as char } else { '=' });
48    }
49    out
50}

52/// Best-effort MIME type from a file extension, for the File objects created by
53/// DOM.setFileInputFiles. Defaults to application/octet-stream.
54fn guess_mime(path: &str) -> &'static str {
55    let ext = std::path::Path::new(path)
56        .extension()
57        .and_then(|e| e.to_str())
58        .unwrap_or("")
59        .to_ascii_lowercase();
60    match ext.as_str() {
61        "jpg" | "jpeg" => "image/jpeg",
62        "png" => "image/png",
63        "gif" => "image/gif",
64        "webp" => "image/webp",
65        "svg" => "image/svg+xml",
66        "bmp" => "image/bmp",
67        "ico" => "image/x-icon",
68        "pdf" => "application/pdf",
69        "txt" => "text/plain",
70        "html" | "htm" => "text/html",
71        "css" => "text/css",
72        "js" | "mjs" => "text/javascript",
73        "json" => "application/json",
74        "xml" => "application/xml",
75        "csv" => "text/csv",
76        "zip" => "application/zip",
77        "gz" => "application/gzip",
78        "mp4" => "video/mp4",
79        "webm" => "video/webm",
80        "mp3" => "audio/mpeg",
81        "wav" => "audio/wav",
82        "doc" => "application/msword",
83        "docx" => "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
84        "xls" => "application/vnd.ms-excel",
85        "xlsx" => "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
86        _ => "application/octet-stream",
87    }
88}

90pub async fn handle(
91    method: &str,
92    params: &Value,
93    ctx: &mut CdpContext,
94    session_id: &Option<String>,
95) -> Result<Value, String> {
96    match method {
97        "enable" => Ok(json!({})),
98        "getDocument" => {
99            let page = ctx.get_session_page(session_id).ok_or("No page")?;
100            let depth = params.get("depth").and_then(|v| v.as_i64()).unwrap_or(2);
101            page.with_dom(|dom| {
102                let node = serialize_node(dom, dom.document(), depth as u32, 0);
103                json!({ "root": node })
104            }).ok_or_else(|| "No DOM loaded".to_string())
105        }
106        "querySelector" => {
107            let page = ctx.get_session_page(session_id).ok_or("No page")?;
108            let selector = params.get("selector").and_then(|v| v.as_str()).ok_or("selector required")?;
109            let result = page.with_dom(|dom| {
110                dom.query_selector(selector).ok().flatten().map(|id| id.index()).unwrap_or(0)
111            }).unwrap_or(0);
112            Ok(json!({ "nodeId": result }))
113        }
114        "querySelectorAll" => {
115            let page = ctx.get_session_page(session_id).ok_or("No page")?;
116            let selector = params.get("selector").and_then(|v| v.as_str()).ok_or("selector required")?;
117            let ids = page.with_dom(|dom| {
118                dom.query_selector_all(selector).ok()
119                    .map(|ids| ids.iter().map(|id| id.index() as u64).collect::<Vec<_>>())
120                    .unwrap_or_default()
121            }).unwrap_or_default();
122            Ok(json!({ "nodeIds": ids }))
123        }
124        "getOuterHTML" => {
125            let page = ctx.get_session_page(session_id).ok_or("No page")?;
126            let node_id = params.get("nodeId").and_then(|v| v.as_u64())
127                .or_else(|| params.get("backendNodeId").and_then(|v| v.as_u64()))
128                .ok_or("nodeId required")?;
129            let html = page.with_dom(|dom| {
130                dom.outer_html(NodeId::new(node_id as u32))
131            }).unwrap_or_default();
132            Ok(json!({ "outerHTML": html }))
133        }
134        "describeNode" => {
135            let page = ctx.get_session_page_mut(session_id).ok_or("No page")?;
136            let depth = params.get("depth").and_then(|v| v.as_i64()).unwrap_or(0);
137
138            let node_id = if let Some(nid) = params.get("nodeId").and_then(|v| v.as_u64())
139                .or_else(|| params.get("backendNodeId").and_then(|v| v.as_u64()))
140            {
141                nid
142            } else if let Some(oid) = params.get("objectId").and_then(|v| v.as_str()) {
143                let code = format!(
144                    "(function() {{ var o = globalThis.__telemaco_objects[{}]; if (!o) return -1; return (typeof o._nid === 'number') ? o._nid : -1; }})()",
145                    crate::util::object_id_literal(oid)
146                );
147                let result = page.evaluate(&code);
148                result.as_f64().map(|n| n as u64).unwrap_or(0)
149            } else {
150                return Err("nodeId or objectId required".to_string());
151            };
152
153            let node = page.with_dom(|dom| {
154                serialize_node(dom, NodeId::new(node_id as u32), depth as u32, 0)
155            }).unwrap_or(json!(null));
156            Ok(json!({ "node": node }))
157        }
158        "resolveNode" => {
159            let page = ctx.get_session_page_mut(session_id).ok_or("No page")?;
160            let node_id = if let Some(nid) = params.get("nodeId").and_then(|v| v.as_u64())
161                .or_else(|| params.get("backendNodeId").and_then(|v| v.as_u64()))
162            {
163                nid
164            } else if let Some(oid) = params.get("objectId").and_then(|v| v.as_str()) {
165                let code = format!(
166                    "(function() {{ var o = globalThis.__telemaco_objects[{}]; return (o && typeof o._nid === 'number') ? o._nid : -1; }})()",
167                    crate::util::object_id_literal(oid)
168                );
169                let result = page.evaluate(&code);
170                result.as_f64().map(|n| n as u64).unwrap_or(0)
171            } else {
172                return Err("nodeId or objectId required".to_string());
173            };
174
175            let js_code = format!(
176                "(function() {{\
177                    var nid = {};\
178                    var node = null;\
179                    if (globalThis._cache && globalThis._cache.has(nid)) {{\
180                        node = globalThis._cache.get(nid);\
181                    }} else {{\
182                        var t = +Deno.core.ops.op_dom('node_type', String(nid), '', globalThis.__telemaco_frameId >>> 0);\
183                        if (t === 1) node = new Element(nid);\
184                        else if (t === 9) node = globalThis.document;\
185                        else node = new Node(nid);\
186                        if (globalThis._cache) globalThis._cache.set(nid, node);\
187                    }}\
188                    return node;\
189                }})()",
190                node_id,
191            );
192
193            let info = if let Some(js) = &mut page.js {
194                match js.store_object_with_meta(&js_code) {
195                    Ok(info) => info,
196                    Err(_) => {
197                        return Ok(json!({
198                            "object": {
199                                "type": "object",
200                                "subtype": "node",
201                                "className": "HTMLElement",
202                                "objectId": format!("node-{}", node_id),
203                            }
204                        }));
205                    }
206                }
207            } else {
208                return Err("No JS runtime".to_string());
209            };
210
211            Ok(json!({
212                "object": {
213                    "type": "object",
214                    "subtype": "node",
215                    "className": if info.class_name.is_empty() { "HTMLElement".to_string() } else { info.class_name },
216                    "description": info.description,
217                    "objectId": info.object_id.unwrap_or_else(|| format!("node-{}", node_id)),
218                }
219            }))
220        }
221        "setAttributeValue" => Ok(json!({})),
222        "removeNode" => Ok(json!({})),
223        "focus" => {
224            // No layout engine, but telemaco's JS focus() sets document.activeElement,
225            // which Input.dispatchKeyEvent targets. CDP clients (browser-use) focus an
226            // input via DOM.focus before typing; without this their keystrokes land on
227            // nothing and the field stays empty.
228            let page = ctx.get_session_page_mut(session_id).ok_or("No page")?;
229            let node_id = resolve_node_id(page, params)?;
230            let code = format!(
231                "(function() {{ var el = globalThis._wrap && globalThis._wrap({0}); \
232                 if (el && typeof el.focus === 'function') {{ el.focus(); return true; }} return false; }})()",
233                node_id
234            );
235            let _ = page.evaluate(&code);
236            Ok(json!({}))
237        }
238        "scrollIntoViewIfNeeded" => {
239            let page = ctx.get_session_page_mut(session_id).ok_or("No page")?;
240            let node_id = resolve_node_id(page, params)?;
241            // Telemaco has no layout viewport to move, but the JS shim records
242            // this element for the hit testing used by subsequent input events.
243            let code = format!(
244                "(function() {{ var el = globalThis._wrap && globalThis._wrap({0}); \
245                 if (!el || typeof el.scrollIntoView !== 'function') return false; \
246                 el.scrollIntoView(); return true; }})()",
247                node_id
248            );
249            let did_scroll = page.evaluate(&code).as_bool().unwrap_or(false);
250            if !did_scroll {
251                return Err(format!(
252                    "node {node_id} could not be resolved to a scrollable element"
253                ));
254            }
255            Ok(json!({}))
256        }
257        "setFileInputFiles" => {
258            // Puppeteer's ElementHandle.uploadFile / Playwright's setInputFiles
259            // drive an <input type=file> through this CDP call (issue #359). Read
260            // each local file, then hand its bytes (base64) to the JS layer, which
261            // builds real File objects and fires input+change like a real
262            // selection so page code can read/upload them.
263            let page = ctx.get_session_page_mut(session_id).ok_or("No page")?;
264            // setFileInputFiles reads local files and hands their bytes to page
265            // JS. Anyone who can reach the CDP port (default localhost, but
266            // Docker images bind 0.0.0.0) could otherwise read any file the
267            // process can read — the same threat as Page.navigate to file://, so
268            // it honours the same opt-in and is off by default.
269            if !page.context.allow_file_access {
270                return Err("DOM.setFileInputFiles is disabled. Restart with `telemaco serve --allow-file-access` to enable local file uploads.".to_string());
271            }
272            let node_id = resolve_node_id(page, params)?;
273            let paths: Vec<String> = params
274                .get("files")
275                .and_then(|v| v.as_array())
276                .map(|a| a.iter().filter_map(|v| v.as_str().map(String::from)).collect())
277                .unwrap_or_default();
278
279            let mut specs = Vec::with_capacity(paths.len());
280            for p in &paths {
281                let bytes = std::fs::read(p)
282                    .map_err(|e| format!("setFileInputFiles: cannot read '{}': {}", p, e))?;
283                let name = std::path::Path::new(p)
284                    .file_name()
285                    .and_then(|n| n.to_str())
286                    .unwrap_or("file")
287                    .to_string();
288                specs.push(json!({ "name": name, "type": guess_mime(p), "b64": encode_base64(&bytes) }));
289            }
290
291            let specs_json = serde_json::to_string(&specs).unwrap_or_else(|_| "[]".to_string());
292            let code = format!(
293                "(function() {{ var el = globalThis._wrap && globalThis._wrap({0}); \
294                 if (el && globalThis.__telemaco_setInputFiles) {{ globalThis.__telemaco_setInputFiles(el, {1}); return true; }} return false; }})()",
295                node_id, specs_json
296            );
297            let _ = page.evaluate(&code);
298            Ok(json!({}))
299        }
300        "getBoxModel" => {
301            let page = ctx.get_session_page_mut(session_id).ok_or("No page")?;
302            let node_id = match resolve_node_id(page, params) {
303                Ok(nid) => nid,
304                Err(_) => return Ok(json!(null)),
305            };
306            let code = format!(
307                "(function() {{\
308                    var el = globalThis._wrap && globalThis._wrap({0});\
309                    if (!el || typeof el.getBoundingClientRect !== 'function') return null;\
310                    var r = el.getBoundingClientRect();\
311                    return [r.left, r.top, r.right, r.top, r.right, r.bottom, r.left, r.bottom,\
312                            r.width, r.height];\
313                }})()",
314                node_id
315            );
316            let val = page.evaluate(&code);
317            let (quad, w, h) = if let Some(arr) = val.as_array() {
318                let nums: Vec<f64> = arr.iter().filter_map(|v| v.as_f64()).collect();
319                if nums.len() >= 10 {
320                    let q: Vec<Value> = nums[..8].iter().map(|n| coord_value(*n)).collect();
321                    (q, nums[8], nums[9])
322                } else {
323                    (vec![json!(8),json!(8),json!(108),json!(8),json!(108),json!(28),json!(8),json!(28)], 100.0, 20.0)
324                }
325            } else {
326                (vec![json!(8),json!(8),json!(108),json!(8),json!(108),json!(28),json!(8),json!(28)], 100.0, 20.0)
327            };
328            Ok(json!({
329                "model": {
330                    "content": quad.clone(),
331                    "padding": quad.clone(),
332                    "border": quad.clone(),
333                    "margin": quad,
334                    "width": coord_value(w), "height": coord_value(h),
335                }
336            }))
337        }
338        "getContentQuads" => {
339            let page = ctx.get_session_page_mut(session_id).ok_or("No page")?;
340            let node_id = match resolve_node_id(page, params) {
341                Ok(nid) => nid,
342                Err(_) => return Ok(json!(null)),
343            };
344            let code = format!(
345                "(function() {{\
346                    var el = globalThis._wrap && globalThis._wrap({0});\
347                    if (!el || typeof el.getBoundingClientRect !== 'function') return null;\
348                    var r = el.getBoundingClientRect();\
349                    return [r.left, r.top, r.right, r.top, r.right, r.bottom, r.left, r.bottom];\
350                }})()",
351                node_id
352            );
353            let val = page.evaluate(&code);
354            let quad = if let Some(arr) = val.as_array() {
355                let nums: Vec<f64> = arr.iter().filter_map(|v| v.as_f64()).collect();
356                if nums.len() == 8 { nums.iter().map(|n| coord_value(*n)).collect::<Vec<_>>() }
357                else { vec![json!(8),json!(8),json!(108),json!(8),json!(108),json!(28),json!(8),json!(28)] }
358            } else {
359                vec![json!(8),json!(8),json!(108),json!(8),json!(108),json!(28),json!(8),json!(28)]
360            };
361            Ok(json!({ "quads": [quad] }))
362        }
363        _ => Err(format!("Unknown DOM method: {}", method)),
364    }

367/// Serialize a CDP box-model coordinate the way Chrome does: an integral double
368/// (e.g. `256.0`) becomes a JSON integer (`256`), while a genuinely fractional
369/// value (`206.0390625`) stays a float. serde_json always writes an `f64` with a
370/// decimal point, so `json!(256.0)` yields `256.0` — which strict CDP clients
371/// (Hermes Agent) deserialize as `i64` and reject, breaking every click-by-
372/// coordinate flow. Chrome only widens fractional coordinates, so mirroring that
373/// keeps those clients working. (issue #576)
374fn coord_value(n: f64) -> Value {
375    // Collapse to an integer only when the value is exactly integral and fits an
376    // i64 losslessly; NaN/inf and out-of-range doubles fall through unchanged.
377    if n.is_finite() && n.fract() == 0.0 && n >= i64::MIN as f64 && n <= i64::MAX as f64 {
378        json!(n as i64)
379    } else {
380        json!(n)
381    }
382}

384// Hard cap on how deep a getDocument/describeNode response may nest, independent
385// of the requested `depth`. DOM.getDocument{depth:-1} arrives here as u32::MAX,
386// which on a pathologically deep DOM (trivially scriptable, unbounded by
387// html5ever) produces a Value nested that far. Even built without recursion,
388// serde_json's own serialization and the Value's Drop recurse over that nesting
389// and overflow the stack — on tokio's ~2 MiB worker stacks especially. Bounding
390// the depth keeps the response safe to serialize and drop. Real DOMs are shallow
391// (deep React trees are a few hundred), so this only truncates pathological
392// nesting, which beats crashing the worker (AGENTS.md: "one page must never
393// crash a worker"). Mirrors DOMSnapshot's MAX_NODES guard (issue #341).
394//
395// The number is sized for the ~2 MiB stack of a tokio worker thread (where the
396// CDP processor runs, and where `#[test]` threads also run): each DOM level
397// becomes two nested JSON containers (object -> "children" array -> object ...),
398// so serde_json's recursive serialization and the Value's recursive Drop descend
399// ~2x this depth. 256 keeps that a few hundred frames deep with wide margin,
400// while still far exceeding any real page. Clients needing a deeper subtree
401// re-request it with DOM.requestChildNodes / describeNode on a specific node.
402const MAX_SERIALIZE_DEPTH: u32 = 256;
403
404/// Build the CDP Node object for a single node (without its `children` array),
405/// returning it together with that node's child ids. `None` for a missing node.
406fn node_value(dom: &DomTree, node_id: NodeId) -> Option<(Value, Vec<NodeId>)> {
407    let node = dom.get_node(node_id)?;
408    let children_ids = dom.children(node_id);
409    let child_count = children_ids.len();
410    let mut result = json!({ "nodeId": node_id.index(), "backendNodeId": node_id.index(), "childNodeCount": child_count });
411
412    match &node.data {
413        NodeData::Document => {
414            result["nodeType"] = json!(9); result["nodeName"] = json!("#document");
415            result["localName"] = json!(""); result["nodeValue"] = json!("");
416            result["documentURL"] = json!(""); result["baseURL"] = json!(""); result["xmlVersion"] = json!("");
417        }
418        NodeData::Doctype { name, public_id, system_id } => {
419            result["nodeType"] = json!(10); result["nodeName"] = json!(name);
420            result["localName"] = json!(""); result["nodeValue"] = json!("");
421            result["publicId"] = json!(public_id); result["systemId"] = json!(system_id);
422        }
423        NodeData::Element { name, attrs, .. } => {
424            result["nodeType"] = json!(1);
425            result["nodeName"] = json!(name.local.as_ref().to_ascii_uppercase());
426            result["localName"] = json!(name.local.as_ref());
427            result["nodeValue"] = json!("");
428            let cdp_attrs: Vec<String> = attrs.iter()
429                .flat_map(|a| vec![a.name.local.to_string(), a.value.clone()]).collect();
430            result["attributes"] = json!(cdp_attrs);
431        }
432        NodeData::Text { contents } => {
433            result["nodeType"] = json!(3); result["nodeName"] = json!("#text");
434            result["localName"] = json!(""); result["nodeValue"] = json!(contents);
435        }
436        NodeData::Comment { contents } => {
437            result["nodeType"] = json!(8); result["nodeName"] = json!("#comment");
438            result["localName"] = json!(""); result["nodeValue"] = json!(contents);
439        }
440        NodeData::ProcessingInstruction { target, data } => {
441            result["nodeType"] = json!(7); result["nodeName"] = json!(target);
442            result["localName"] = json!(""); result["nodeValue"] = json!(data);
443        }
444    }
445
446    Some((result, children_ids))
447}

449/// Serialize a node and its descendants into the CDP Node tree, iteratively.
450/// The requested `max_depth` is clamped to `current_depth + MAX_SERIALIZE_DEPTH`
451/// so a `depth:-1` (u32::MAX) request on a very deep DOM cannot produce a Value
452/// that overflows the stack when serde_json later serializes or drops it. An
453/// explicit heap worklist keeps the builder itself off the call stack.
454fn serialize_node(dom: &DomTree, node_id: NodeId, max_depth: u32, current_depth: u32) -> Value {
455    let max_depth = max_depth.min(current_depth.saturating_add(MAX_SERIALIZE_DEPTH));
456
457    struct Frame {
458        value: Value,
459        children: Vec<NodeId>,
460        next: usize,
461        built: Vec<Value>,
462        depth: u32,
463        expand: bool,
464    }
465
466    let (root_value, root_children) = match node_value(dom, node_id) {
467        Some(v) => v,
468        None => return json!(null),
469    };
470    let root_expand = current_depth < max_depth && !root_children.is_empty();
471    let mut stack = vec![Frame {
472        value: root_value,
473        children: root_children,
474        next: 0,
475        built: Vec::new(),
476        depth: current_depth,
477        expand: root_expand,
478    }];
479
480    loop {
481        // Decide the next step without holding a borrow across a push.
482        let next_child = {
483            let top = stack.last_mut().expect("stack is non-empty in loop");
484            if top.expand && top.next < top.children.len() {
485                let cid = top.children[top.next];
486                top.next += 1;
487                Some(cid)
488            } else {
489                None
490            }
491        };
492
493        match next_child {
494            Some(cid) => {
495                let child_depth = stack.last().unwrap().depth + 1;
496                match node_value(dom, cid) {
497                    Some((cval, cchildren)) => {
498                        let cexpand = child_depth < max_depth && !cchildren.is_empty();
499                        stack.push(Frame {
500                            value: cval,
501                            children: cchildren,
502                            next: 0,
503                            built: Vec::new(),
504                            depth: child_depth,
505                            expand: cexpand,
506                        });
507                    }
508                    // Missing child: match the old recursive behavior of emitting null.
509                    None => stack.last_mut().unwrap().built.push(json!(null)),
510                }
511            }
512            None => {
513                // This node's children are all built; finalize and fold into parent.
514                let mut frame = stack.pop().unwrap();
515                if !frame.built.is_empty() {
516                    frame.value["children"] = json!(frame.built);
517                }
518                match stack.last_mut() {
519                    Some(parent) => parent.built.push(frame.value),
520                    None => return frame.value,
521                }
522            }
523        }
524    }
525}

673    // A hostile or heavy page can nest nodes tens of thousands deep (trivially
674    // scriptable, and html5ever puts no cap on generic nesting). DOM.getDocument
675    // with the standard depth:-1 (which becomes u32::MAX here) must handle such a
676    // tree without crashing the worker (AGENTS.md: "one page must never crash a
677    // worker"). Two failure modes: (1) a recursive serialize_node overflows the
678    // stack building the Value, and (2) even an iterative builder would emit a
679    // Value so deeply nested that serde_json's own recursive serialize/Drop
680    // overflow. The fix derecurses the builder AND bounds the depth, so the
681    // response is always safe to serialize and drop. With the old recursive
682    // serialize_node this test aborts with SIGABRT.
683    #[test]
684    fn get_document_deep_tree_does_not_overflow() {
685        use telemaco_dom::{DomTree, NodeData};
686
687        // Build the deep chain directly (no parser) so setup is O(n) and fast.
688        let dom = DomTree::new();
689        let mut parent = dom.document();
690        let depth = 50_000usize;
691        for _ in 0..depth {
692            let n = dom.new_node(NodeData::Text { contents: String::new() });
693            dom.append_child(parent, n);
694            parent = n;
695        }
696
697        // Mirror getDocument {"depth": -1}: as_i64() == -1, then `depth as u32`.
698        let node = serialize_node(&dom, dom.document(), (-1i64) as u32, 0);
699
700        // serde_json's own serialization recurses over the Value nesting; this
701        // must not overflow either. That is why the fix bounds depth, not just
702        // the builder.
703        let s = serde_json::to_string(&node).expect("serialize");
704        assert!(!s.is_empty());
705
706        // Output nesting is bounded well below the tree's true depth (truncated,
707        // not crashed) yet still serializes a meaningful prefix.
708        let mut cur = &node;
709        let mut levels = 0usize;
710        while let Some(children) = cur.get("children").and_then(|c| c.as_array()) {
711            let Some(first) = children.first() else { break };
712            cur = first;
713            levels += 1;
714        }
715        assert!(levels >= 100, "should serialize a deep prefix, got {levels}");
716        assert!(
717            levels < depth,
718            "nesting must be bounded below the tree's true depth, got {levels}"
719        );
720    }
