1//! `DOMSnapshot.captureSnapshot` for layout-free engines.
2//!
3//! browser-use (and other CDP DOM-agent frameworks) build their interactive
4//! element index from `DOMSnapshot.captureSnapshot`: the per-node bounds,
5//! computed styles, and `isClickable` flag, correlated with `DOM.getDocument`
6//! by `backendNodeId`. Without this domain their DOM build aborts and the agent
7//! sees zero elements.
8//!
9//! Telemaco has no layout/paint engine, so there is no real geometry to report.
10//! We synthesize it: every node gets a distinct, on-screen, non-icon-sized box
11//! (a simple vertical stack) plus plausible computed styles (visible, opaque,
12//! pointer cursor on interactive tags). That is enough for the element
13//! detection path, which keys off tag name / ARIA / accessibility role and does
14//! not need true geometry. Clicking still falls back to JS `.click()` since the
15//! coordinates are synthetic. `backendNodeId == nid`, matching `DOM.getDocument`.
16

18use serde_json::{json, Value};

22/// Computed-style names browser-use requests, in the exact order it expects to
23/// read them back out of each layout node's `styles` index array.
24const REQUIRED_STYLES: &[&str] = &[
25    "display",
26    "visibility",
27    "opacity",
28    "overflow",
29    "overflow-x",
30    "overflow-y",
31    "cursor",
32    "pointer-events",
33    "position",
34    "background-color",
35];
36
37/// Cap the synthesized snapshot so a pathologically large DOM cannot produce a
38/// runaway payload. Matches the spirit of the descendants() length cap.
39const MAX_NODES: usize = 20_000;
40
41pub async fn handle(
42    method: &str,
43    _params: &Value,
44    ctx: &mut CdpContext,
45    session_id: &Option<String>,
46) -> Result<Value, String> {
47    match method {
48        "enable" | "disable" => Ok(json!({})),
49        "captureSnapshot" => {
50            let page = ctx.get_session_page(session_id).ok_or("No page")?;
51            let url = page.url_string();
52            let title = page.title.clone();
53            page.with_dom(|dom| build_capture_snapshot(dom, &url, &title))
54                .ok_or_else(|| "No DOM loaded".to_string())
55        }
56        // Permissive no-op for the rest of the domain (e.g. getSnapshot) so a
57        // client that probes it does not abort on an Unknown-method error.
58        _ => Ok(json!({})),
59    }
60}

62/// String table with de-duplication. Every string in a DOMSnapshot response is
63/// referenced by its index into the top-level `strings` array.
64struct Interner {
65    map: std::collections::HashMap<String, i64>,
66    list: Vec<String>,
67}
68
69impl Interner {
70    fn new() -> Self {
71        let mut s = Interner { map: std::collections::HashMap::new(), list: Vec::new() };
72        s.intern(""); // index 0 is the empty string by convention
73        s
74    }
75    fn intern(&mut self, s: &str) -> i64 {
76        if let Some(&i) = self.map.get(s) {
77            return i;
78        }
79        let i = self.list.len() as i64;
80        self.list.push(s.to_string());
81        self.map.insert(s.to_string(), i);
82        i
83    }
84}

86/// Pre-order DFS from the document, recording each node and its parent's index.
87/// Iterative with an explicit stack: the MAX_NODES cap bounds the node count,
88/// but recursion depth is a separate axis. A deeply nested linear chain (script
89/// can build thousands of nested elements) would recurse that many frames deep
90/// and could overflow the stack before the count guard triggers. An explicit
91/// stack keeps the depth on the heap (issue #341).
92fn walk(
93    dom: &DomTree,
94    root: NodeId,
95    root_parent: i64,
96    order: &mut Vec<NodeId>,
97    parent_idx: &mut Vec<i64>,
98) {
99    // (node, parent_index). Children are pushed in reverse so they pop in
100    // document order, preserving the original pre-order traversal.
101    let mut stack: Vec<(NodeId, i64)> = vec![(root, root_parent)];
102    while let Some((id, parent)) = stack.pop() {
103        if order.len() >= MAX_NODES {
104            break;
105        }
106        let my = order.len() as i64;
107        order.push(id);
108        parent_idx.push(parent);
109        let children: Vec<NodeId> = dom.children(id);
110        for child in children.into_iter().rev() {
111            stack.push((child, my));
112        }
113    }
114}

116fn build_capture_snapshot(dom: &DomTree, url: &str, title: &str) -> Value {
117    let mut order: Vec<NodeId> = Vec::new();
118    let mut parent_idx: Vec<i64> = Vec::new();
119    walk(dom, dom.document(), -1, &mut order, &mut parent_idx);
120
121    let mut strings = Interner::new();
122    let doc_url_idx = strings.intern(url);
123    let title_idx = strings.intern(title);
124
125    let n = order.len();
126    let mut node_type: Vec<i64> = Vec::with_capacity(n);
127    let mut node_name: Vec<i64> = Vec::with_capacity(n);
128    let mut node_value: Vec<i64> = Vec::with_capacity(n);
129    let mut backend_ids: Vec<i64> = Vec::with_capacity(n);
130    let mut attributes: Vec<Value> = Vec::with_capacity(n);
131    let mut clickable: Vec<i64> = Vec::new();
132
133    // Layout arrays are 1:1 with nodes (nodeIndex[i] == i).
134    let mut layout_node_index: Vec<i64> = Vec::with_capacity(n);
135    let mut bounds: Vec<Value> = Vec::with_capacity(n);
136    let mut styles: Vec<Value> = Vec::with_capacity(n);
137    let mut paint_orders: Vec<i64> = Vec::with_capacity(n);
138    let mut client_rects: Vec<Value> = Vec::with_capacity(n);
139    let mut layout_text: Vec<i64> = Vec::with_capacity(n);
140
141    for (i, &nid) in order.iter().enumerate() {
142        let node = match dom.get_node(nid) {
143            Some(node) => node,
144            None => {
145                // Keep arrays aligned even for a vanished node.
146                node_type.push(0);
147                node_name.push(0);
148                node_value.push(0);
149                backend_ids.push(nid.index() as i64);
150                attributes.push(json!([]));
151                layout_node_index.push(i as i64);
152                bounds.push(json!([0.0, 0.0, 0.0, 0.0]));
153                styles.push(json!([]));
154                paint_orders.push(i as i64);
155                client_rects.push(json!([0.0, 0.0, 0.0, 0.0]));
156                layout_text.push(-1);
157                continue;
158            }
159        };
160
161        let (ntype, nname, nval, attrs, tag): (i64, String, String, Vec<(String, String)>, String) =
162            match &node.data {
163                NodeData::Document => (9, "#document".into(), String::new(), vec![], String::new()),
164                NodeData::Doctype { name, .. } => {
165                    (10, name.to_string(), String::new(), vec![], String::new())
166                }
167                NodeData::Element { name, attrs, .. } => {
168                    let tag = name.local.as_ref().to_string();
169                    let flat = attrs
170                        .iter()
171                        .map(|a| (a.name.local.to_string(), a.value.clone()))
172                        .collect();
173                    (1, tag.to_ascii_uppercase(), String::new(), flat, tag.to_ascii_lowercase())
174                }
175                NodeData::Text { contents } => {
176                    (3, "#text".into(), contents.clone(), vec![], String::new())
177                }
178                NodeData::Comment { contents } => {
179                    (8, "#comment".into(), contents.clone(), vec![], String::new())
180                }
181                NodeData::ProcessingInstruction { target, data } => {
182                    (7, target.to_string(), data.clone(), vec![], String::new())
183                }
184            };
185
186        node_type.push(ntype);
187        node_name.push(strings.intern(&nname));
188        node_value.push(strings.intern(&nval));
189        backend_ids.push(nid.index() as i64);
190
191        let mut attr_idx: Vec<Value> = Vec::with_capacity(attrs.len() * 2);
192        for (k, v) in &attrs {
193            attr_idx.push(json!(strings.intern(k)));
194            attr_idx.push(json!(strings.intern(v)));
195        }
196        attributes.push(json!(attr_idx));
197
198        let interactive = matches!(
199            tag.as_str(),
200            "a" | "button" | "input" | "select" | "textarea" | "summary" | "details" | "option" | "label"
201        );
202        let has_onclick = attrs.iter().any(|(k, _)| k.eq_ignore_ascii_case("onclick"));
203        if interactive || has_onclick {
204            clickable.push(i as i64);
205        }
206
207        // Tags that never render box content; report display:none so the agent
208        // does not treat them as visible.
209        let hidden = matches!(
210            tag.as_str(),
211            "head" | "meta" | "title" | "script" | "style" | "link" | "noscript" | "base"
212        );
213        let display = if ntype == 1 && hidden { "none" } else { "block" };
214        let cursor = if interactive { "pointer" } else { "auto" };
215        let style_vals = [
216            display,
217            "visible",
218            "1",
219            "visible",
220            "visible",
221            "visible",
222            cursor,
223            "auto",
224            "static",
225            "rgba(0, 0, 0, 0)",
226        ];
227        debug_assert_eq!(style_vals.len(), REQUIRED_STYLES.len());
228        let style_idx: Vec<Value> = style_vals.iter().map(|s| json!(strings.intern(s))).collect();
229        styles.push(json!(style_idx));
230
231        // Synthetic geometry: a vertical stack, full-width, 18px tall. Distinct
232        // and non-icon-sized so visibility/size heuristics include the element;
233        // the coordinates are not real (no layout engine).
234        let y = (i as f64) * 18.0;
235        bounds.push(json!([0.0, y, 1280.0, 18.0]));
236        client_rects.push(json!([0.0, y, 1280.0, 18.0]));
237        paint_orders.push(i as i64);
238        layout_node_index.push(i as i64);
239        layout_text.push(-1);
240    }
241
242    let content_height = (n as i64) * 18;
243    json!({
244        "documents": [{
245            "documentURL": doc_url_idx,
246            "title": title_idx,
247            "baseURL": doc_url_idx,
248            "contentLanguage": 0,
249            "encodingName": 0,
250            "publicId": 0,
251            "systemId": 0,
252            "frameId": 0,
253            "nodes": {
254                "parentIndex": parent_idx,
255                "nodeType": node_type,
256                "nodeName": node_name,
257                "nodeValue": node_value,
258                "backendNodeId": backend_ids,
259                "attributes": attributes,
260                "isClickable": { "index": clickable },
261            },
262            "layout": {
263                "nodeIndex": layout_node_index,
264                "styles": styles,
265                "bounds": bounds,
266                "text": layout_text,
267                "paintOrders": paint_orders,
268                "clientRects": client_rects,
269            },
270            "textBoxes": { "layoutIndex": [], "bounds": [], "start": [], "length": [] },
271            "scrollOffsetX": 0.0,
272            "scrollOffsetY": 0.0,
273            "contentWidth": 1280,
274            "contentHeight": content_height,
275        }],
276        "strings": strings.list,
277    })
278}

285    fn collect_backend_ids(node: &Value, out: &mut Vec<i64>) {
286        if let Some(id) = node.get("backendNodeId").and_then(|v| v.as_i64()) {
287            out.push(id);
288        }
289        if let Some(children) = node.get("children").and_then(|v| v.as_array()) {
290            for c in children {
291                collect_backend_ids(c, out);
292            }
293        }
294    }
295
296    fn find_backend_id_by_name(node: &Value, name: &str) -> Option<i64> {
297        if node.get("nodeName").and_then(|v| v.as_str()) == Some(name) {
298            return node.get("backendNodeId").and_then(|v| v.as_i64());
299        }
300        node.get("children")
301            .and_then(|v| v.as_array())
302            .and_then(|children| children.iter().find_map(|c| find_backend_id_by_name(c, name)))
303    }

305    async fn navigate(ctx: &mut CdpContext, body: &str) -> String {
306        let page_id = ctx.create_page();
307        let session_id = format!("{}-session", page_id);
308        ctx.sessions.insert(session_id.clone(), page_id.clone());
309        let url = format!("data:text/html,{body}");
310        crate::domains::page::handle(
311            "navigate",
312            &json!({ "url": url, "waitUntil": "load" }),
313            ctx,
314            &Some(session_id.clone()),
315        )
316        .await
317        .expect("navigate should succeed");
318        session_id
319    }

321    #[tokio::test]
322    async fn capture_snapshot_matches_getdocument_and_flags_clickable() {
323        let mut ctx = CdpContext::new();
324        let session = navigate(&mut ctx, "<button id=go>Go</button><a href=/x>L</a>").await;
325
326        // DOM.getDocument is the structural source; the snapshot must use the
327        // same backendNodeId scheme so a client can correlate the two.
328        let doc = crate::domains::dom::handle(
329            "getDocument",
330            &json!({ "depth": -1 }),
331            &mut ctx,
332            &Some(session.clone()),
333        )
334        .await
335        .expect("getDocument should succeed");
336        let mut doc_ids = Vec::new();
337        collect_backend_ids(&doc["root"], &mut doc_ids);
338        assert!(!doc_ids.is_empty(), "getDocument returned no nodes");
339
340        let snap = handle("captureSnapshot", &json!({}), &mut ctx, &Some(session.clone()))
341            .await
342            .expect("captureSnapshot should succeed");
343
344        let documents = snap["documents"].as_array().expect("documents array");
345        assert_eq!(documents.len(), 1);
346        let nodes = &documents[0]["nodes"];
347        let layout = &documents[0]["layout"];
348
349        let snap_ids: Vec<i64> = nodes["backendNodeId"]
350            .as_array()
351            .unwrap()
352            .iter()
353            .map(|v| v.as_i64().unwrap())
354            .collect();
355
356        // Every node from getDocument must appear in the snapshot under the
357        // identical backendNodeId.
358        for id in &doc_ids {
359            assert!(
360                snap_ids.contains(id),
361                "snapshot is missing backendNodeId {id} present in getDocument"
362            );
363        }
364
365        // String table is populated (everything is referenced by index).
366        assert!(
367            snap["strings"].as_array().unwrap().len() > 1,
368            "string table should be populated"
369        );
370
371        // Layout arrays are aligned 1:1 with nodes and carry bounds + the 10
372        // computed styles browser-use reads back positionally.
373        let n = snap_ids.len();
374        assert_eq!(layout["nodeIndex"].as_array().unwrap().len(), n);
375        assert_eq!(layout["bounds"].as_array().unwrap().len(), n);
376        assert_eq!(layout["styles"].as_array().unwrap().len(), n);
377        assert_eq!(
378            layout["bounds"][0].as_array().unwrap().len(),
379            4,
380            "bounds entries are [x, y, w, h]"
381        );
382        assert_eq!(
383            layout["styles"][0].as_array().unwrap().len(),
384            REQUIRED_STYLES.len(),
385            "each layout node carries all required computed styles"
386        );
387
388        // Interactive elements are flagged isClickable (by node index).
389        let clickable: Vec<i64> = nodes["isClickable"]["index"]
390            .as_array()
391            .unwrap()
392            .iter()
393            .map(|v| v.as_i64().unwrap())
394            .collect();
395        for tag in ["BUTTON", "A"] {
396            let bid = find_backend_id_by_name(&doc["root"], tag)
397                .unwrap_or_else(|| panic!("{tag} should be in the document"));
398            let idx = snap_ids
399                .iter()
400                .position(|&id| id == bid)
401                .unwrap_or_else(|| panic!("{tag} should be in the snapshot")) as i64;
402            assert!(clickable.contains(&idx), "{tag} must be flagged isClickable");
403        }
404    }
