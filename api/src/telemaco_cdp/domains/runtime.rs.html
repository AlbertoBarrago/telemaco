7/// Whether a binding name is a plain JS identifier and therefore safe to
8/// interpolate into the generated shim / teardown scripts. Chromium bindings
9/// are identifiers; anything else (quotes, brackets, spaces, operators) could
10/// break out of the surrounding string literal and inject arbitrary JS into the
11/// page. `Runtime.addBinding` always enforced this, but `Runtime.removeBinding`
12/// did not, so a crafted name escaped `delete globalThis['{name}']` and ran in
13/// the page context. Both handlers now share this guard.
14fn is_valid_binding_name(name: &str) -> bool {
15    !name.is_empty()
16        && name.chars().all(|c| c.is_alphanumeric() || c == '_' || c == '$')
17        && !name.chars().next().unwrap_or('0').is_ascii_digit()
18}

20/// Drain pending JS-initiated navigation (form.submit, location.assign, etc),
21/// then emit the same CDP nav-event sequence Page.navigate emits so
22/// Puppeteer's waitForNavigation / Playwright's wait_for_url resolves.
23/// Without this, in-page navigations look like Runtime.evaluate finishing
24/// to clients and they hang waiting for a frameNavigated that never fires.
25async fn emit_post_eval_nav(
26    ctx: &mut CdpContext,
27    session_id: &Option<String>,
28) -> Result<(), String> {
29    let page = ctx
30        .get_session_page_mut(session_id)
31        .ok_or("No page")?;
32    let did_navigate = page.process_pending_navigation().await.map_err(|e| e.to_string())?;
33    if !did_navigate {
34        return Ok(());
35    }
36    let (frame_id, page_url, page_id, network_events, reached_idle) = {
37        let p = ctx.get_session_page_mut(session_id).ok_or("No page")?;
38        (
39            p.frame_id.clone(),
40            p.url_string(),
41            p.id.clone(),
42            p.network_events.drain(..).collect::<Vec<_>>(),
43            p.lifecycle.is_network_idle(),
44        )
45    };
46    let loader_id = format!("loader-{}", uuid::Uuid::new_v4());
47    super::page::emit_navigation_events(
48        ctx,
49        session_id,
50        &frame_id,
51        &loader_id,
52        &page_url,
53        &page_id,
54        &network_events,
55        WaitUntil::Load,
56        reached_idle,
57    );
58    Ok(())
59}

61pub async fn handle(
62    method: &str,
63    params: &Value,
64    ctx: &mut CdpContext,
65    session_id: &Option<String>,
66) -> Result<Value, String> {
67    match method {
68        "enable" => {
69            // puppeteer-extra's FrameManager.initialize calls Runtime.enable on
70            // the browser-level connection BEFORE any page target exists. Real
71            // Chrome replies with `{}` and emits executionContextCreated when
72            // a context appears. Returning "No page" here breaks the standard
73            // puppeteer connect/newPage flow. If there's no session, succeed
74            // silently — the next Target.attachToTarget will set things up.
75            match ctx.get_session_page(session_id) {
76                Some(page) => {
77                    let origin = page.url_string();
78                    let page_id = page.id.clone();
79                    let frame_id = page.frame_id.clone();
80                    if let Some(session_id) = session_id {
81                        ctx.runtime_enabled_sessions.insert(session_id.clone());
82                    }
83                    ctx.refresh_runtime_event_collection(&page_id);
84                    let event = crate::types::CdpEvent {
85                        method: "Runtime.executionContextCreated".to_string(),
86                        params: json!({
87                            "context": {
88                                "id": 1,
89                                "origin": origin,
90                                "name": "",
91                                "uniqueId": format!("ctx-{}", page_id),
92                                "auxData": {
93                                    "isDefault": true,
94                                    "type": "default",
95                                    "frameId": frame_id,
96                                }
97                            }
98                        }),
99                        session_id: session_id.clone(),
100                    };
101                    ctx.pending_events.push(event);
102                }
103                None => {
104                    // No session attached yet — that's fine. Just ack.
105                }
106            }
107            Ok(json!({}))
108        }
109        "disable" => {
110            if let Some(session_id) = session_id {
111                let page_id = ctx.sessions.get(session_id).cloned();
112                ctx.runtime_enabled_sessions.remove(session_id);
113                if let Some(page_id) = page_id {
114                    ctx.refresh_runtime_event_collection(&page_id);
115                }
116            }
117            Ok(json!({}))
118        }
119        "evaluate" => {
120            let expression = params
121                .get("expression")
122                .and_then(|v| v.as_str())
123                .ok_or("expression required")?;
124            let return_by_value = params
125                .get("returnByValue")
126                .and_then(|v| v.as_bool())
127                .unwrap_or(false);
128
129            validate_context_id(params, "contextId", ctx, "evaluate")?;
130
131            let await_promise = params
132                .get("awaitPromise")
133                .and_then(|v| v.as_bool())
134                .unwrap_or(false);
135
136            // CDP `timeout` field (milliseconds). Default to Chrome's
137            // protocolTimeout (30s) so long evaluations don't pin the V8 lock
138            // indefinitely and starve every other CDP command on the same
139            // session.
140            let timeout_ms = params
141                .get("timeout")
142                .and_then(|v| v.as_u64())
143                .unwrap_or(30_000);
144
145            let page = ctx
146                .get_session_page_mut(session_id)
147                .ok_or("No page")?;
148            let info = match tokio::time::timeout(
149                std::time::Duration::from_millis(timeout_ms),
150                page.evaluate_for_cdp_with_timeout(
151                    expression,
152                    return_by_value,
153                    await_promise,
154                    timeout_ms,
155                ),
156            )
157            .await
158            {
159                Ok(Ok(info)) => info,
160                Ok(Err(error)) => return Err(error),
161                Err(_) => {
162                    return Err(format!(
163                        "Runtime.evaluate exceeded {timeout_ms}ms timeout"
164                    ));
165                }
166            };
167            emit_post_eval_nav(ctx, session_id).await?;
168
169            Ok(json!({ "result": remote_object_from_info(&info) }))
170        }
171        "callFunctionOn" => {
172            let function_declaration = params
173                .get("functionDeclaration")
174                .and_then(|v| v.as_str())
175                .unwrap_or("() => undefined");
176            let return_by_value = params
177                .get("returnByValue")
178                .and_then(|v| v.as_bool())
179                .unwrap_or(false);
180            let await_promise = params
181                .get("awaitPromise")
182                .and_then(|v| v.as_bool())
183                .unwrap_or(false);
184            let object_id = params.get("objectId").and_then(|v| v.as_str());
185            let arguments = params
186                .get("arguments")
187                .and_then(|v| v.as_array())
188                .map(|a| a.to_vec())
189                .unwrap_or_default();
190
191            // #51: validate executionContextId the same way Runtime.evaluate
192            // does. CDP names this field `executionContextId` on
193            // callFunctionOn (not `contextId`); a request may omit it when
194            // `objectId` is supplied — in that case validate_context_id is a
195            // no-op and the default context is used.
196            validate_context_id(params, "executionContextId", ctx, "callFunctionOn")?;
197
198            // Keep awaitPromise alive for the same command budget as evaluate.
199            // Playwright implements waits with callFunctionOn on some utility
200            // paths, so a shorter hidden cap makes the client return before the
201            // requested browser timer fires.
202            let timeout_ms = params
203                .get("timeout")
204                .and_then(|v| v.as_u64())
205                .unwrap_or(30_000);
206
207            let page = ctx
208                .get_session_page_mut(session_id)
209                .ok_or("No page")?;
210            let info = match tokio::time::timeout(
211                std::time::Duration::from_millis(timeout_ms),
212                page.call_function_on_for_cdp_with_timeout(
213                    function_declaration,
214                    object_id,
215                    &arguments,
216                    return_by_value,
217                    await_promise,
218                    timeout_ms,
219                ),
220            )
221            .await
222            {
223                Ok(Ok(info)) => info,
224                Ok(Err(error)) => return Err(error),
225                Err(_) => {
226                    return Err(format!(
227                        "Runtime.callFunctionOn exceeded {timeout_ms}ms timeout"
228                    ));
229                }
230            };
231            emit_post_eval_nav(ctx, session_id).await?;
232
233            Ok(json!({ "result": remote_object_from_info(&info) }))
234        }
235        "getProperties" => {
236            // Puppeteer's $$() flow:
237            //   1. evaluate querySelectorAll → handle for the NodeList
238            //   2. getProperties on that handle → indexed items
239            //   3. For each item, JSHandle.asElement() checks subtype === 'node';
240            //      if true, wraps as ElementHandle (with click/type/etc).
241            //
242            // Older impl returned the raw value via JSON, dropping the node
243            // identity. Items came back as `{type:'object'}` with no objectId
244            // and no subtype, so asElement returned null and the caller got
245            // plain JSHandles back from page.$$ -- breaking checkboxes[0].click().
246            //
247            // We now:
248            //   1. Walk the underlying object in JS, allocating a stable child
249            //      oid per (parent_oid + index) and stashing each value in
250            //      __telemaco_objects so later callFunctionOn can resolve it.
251            //   2. Annotate each item with subtype:'node' + className when the
252            //      value has a numeric nodeType, so Puppeteer wraps it as
253            //      ElementHandle.
254            let object_id = params.get("objectId").and_then(|v| v.as_str());
255            if let Some(oid) = object_id {
256                let page = ctx
257                    .get_session_page_mut(session_id)
258                    .ok_or("No page")?;
259                // The child ids minted below are `<parent>::<key>` and the key
260                // is a property name off a page object, so the page decides
261                // what ends up inside this literal. A JSON literal covers the
262                // C0 controls a manual quote/backslash pair leaves alone; see
263                // `util::object_id_literal`.
264                let oid_literal = crate::util::object_id_literal(oid);
265                let code = format!(
266                    "(function() {{\
267                        var obj = globalThis.__telemaco_objects[{oid}];\
268                        if (!obj || typeof obj !== 'object') return [];\
269                        var keys = Object.keys(obj);\
270                        return keys.map(function(k) {{\
271                            var v = obj[k];\
272                            var t = typeof v;\
273                            var item = {{ name: k, type: t }};\
274                            if (v === null) {{ item.value = null; return item; }}\
275                            if (t !== 'object' && t !== 'function') {{ item.value = v; return item; }}\
276                            var childOid = {oid} + '::' + k;\
277                            globalThis.__telemaco_objects[childOid] = v;\
278                            item.childOid = childOid;\
279                            if (typeof v.nodeType === 'number') {{\
280                                item.subtype = 'node';\
281                                item.className = v.constructor && v.constructor.name ? v.constructor.name : (v.tagName ? 'HTML' + v.tagName.charAt(0) + v.tagName.slice(1).toLowerCase() + 'Element' : 'Node');\
282                                item.description = v.tagName ? v.tagName.toLowerCase() : (v.nodeName || 'node');\
283                            }} else if (Array.isArray(v)) {{\
284                                item.subtype = 'array';\
285                                item.className = 'Array';\
286                                item.description = 'Array(' + v.length + ')';\
287                            }} else {{\
288                                item.className = (v.constructor && v.constructor.name) || 'Object';\
289                                item.description = item.className;\
290                            }}\
291                            return item;\
292                        }});\
293                    }})()",
294                    oid = oid_literal,
295                );
296                let result = page.evaluate(&code);
297                if let serde_json::Value::Array(props) = result {
298                    let descriptors: Vec<Value> = props
299                        .iter()
300                        .map(|p| {
301                            let name = p.get("name").and_then(|v| v.as_str()).unwrap_or("");
302                            let prop_type =
303                                p.get("type").and_then(|v| v.as_str()).unwrap_or("undefined");
304                            let mut remote = json!({ "type": prop_type });
305                            if let Some(child_oid) = p.get("childOid").and_then(|v| v.as_str()) {
306                                remote["type"] = json!("object");
307                                if let Some(sub) = p.get("subtype").and_then(|v| v.as_str()) {
308                                    remote["subtype"] = json!(sub);
309                                }
310                                if let Some(cls) = p.get("className").and_then(|v| v.as_str()) {
311                                    remote["className"] = json!(cls);
312                                }
313                                if let Some(desc) = p.get("description").and_then(|v| v.as_str()) {
314                                    remote["description"] = json!(desc);
315                                }
316                                remote["objectId"] = json!(child_oid);
317                            } else if let Some(val) = p.get("value") {
318                                match val {
319                                    Value::Null => {
320                                        remote["type"] = json!("object");
321                                        remote["subtype"] = json!("null");
322                                        remote["value"] = json!(null);
323                                    }
324                                    Value::String(s) => {
325                                        remote["type"] = json!("string");
326                                        remote["value"] = json!(s);
327                                    }
328                                    Value::Number(n) => {
329                                        remote["type"] = json!("number");
330                                        remote["value"] = json!(n);
331                                    }
332                                    Value::Bool(b) => {
333                                        remote["type"] = json!("boolean");
334                                        remote["value"] = json!(b);
335                                    }
336                                    _ => {
337                                        remote["value"] = val.clone();
338                                    }
339                                }
340                            }
341                            json!({
342                                "name": name,
343                                "value": remote,
344                                "configurable": true,
345                                "enumerable": true,
346                                "writable": true,
347                                "isOwn": true,
348                            })
349                        })
350                        .collect();
351                    Ok(json!({ "result": descriptors, "internalProperties": [] }))
352                } else {
353                    Ok(json!({ "result": [], "internalProperties": [] }))
354                }
355            } else {
356                Ok(json!({ "result": [], "internalProperties": [] }))
357            }
358        }
359        "releaseObject" => {
360            if let Some(oid) = params.get("objectId").and_then(|v| v.as_str()) {
361                if let Some(page) = ctx.get_session_page_mut(session_id) {
362                    page.release_object(oid);
363                }
364            }
365            Ok(json!({}))
366        }
367        "releaseObjectGroup" => {
368            if let Some(page) = ctx.get_session_page_mut(session_id) {
369                page.release_object_group();
370            }
371            Ok(json!({}))
372        }
373        "addBinding" => {
374            let name = params.get("name").and_then(|v| v.as_str()).unwrap_or("");
375            if is_valid_binding_name(name) {
376                // The shim forwards every call back to Rust through
377                // op_binding_called; the CDP dispatcher then drains the
378                // queue and emits Runtime.bindingCalled events the same
379                // way Chromium does. Chromium's V8InspectorImpl rejects
380                // calls without exactly one argument and ToString-coerces
381                // that argument before emitting it as the payload — we
382                // match the coercion (`String(arg)`) and silently drop
383                // calls with wrong arity, which is what Chrome does.
384                let shim = format!(
385                    "globalThis['{name}'] = function (arg) {{\
386                        if (arguments.length !== 1) return;\
387                        try {{\
388                            const payload = typeof arg === 'string' ? arg : String(arg);\
389                            Deno.core.ops.op_binding_called('{name}', payload);\
390                        }} catch (e) {{ /* swallow: binding must not throw into page */ }}\
391                    }};",
392                    name = name,
393                );
394                // Re-install on every navigation: globalThis is wiped on
395                // each new document, and puppeteer registers bindings
396                // once-per-page rather than once-per-document.
397                let key = format!("__telemaco_binding__{}", name);
398                ctx.preload_scripts.retain(|(k, _)| k != &key);
399                ctx.preload_scripts.push((key, shim.clone()));
400                // Remember who subscribed, so the call goes back to this
401                // session rather than to whichever session of the page a
402                // HashMap happens to yield first. A client discards an event
403                // addressed to a session it does not hold, and the session
404                // Target.createTarget leaves behind is not the one a client
405                // ends up using.
406                if let Some(session_id) = session_id {
407                    let owners = ctx.binding_sessions.entry(name.to_string()).or_default();
408                    if !owners.contains(session_id) {
409                        owners.push(session_id.clone());
410                    }
411                }
412                // Install on the current page so the binding is usable
413                // immediately, without waiting for the next navigation.
414                if let Some(page) = ctx.get_session_page_mut(session_id) {
415                    page.evaluate(&shim);
416                }
417            }
418            Ok(json!({}))
419        }
420        "removeBinding" => {
421            let name = params.get("name").and_then(|v| v.as_str()).unwrap_or("");
422            if is_valid_binding_name(name) {
423                let key = format!("__telemaco_binding__{}", name);
424                ctx.preload_scripts.retain(|(k, _)| k != &key);
425                if let Some(session_id) = session_id {
426                    if let Some(owners) = ctx.binding_sessions.get_mut(name) {
427                        owners.retain(|owner| owner != session_id);
428                        if owners.is_empty() {
429                            ctx.binding_sessions.remove(name);
430                        }
431                    }
432                }
433                if let Some(page) = ctx.get_session_page_mut(session_id) {
434                    page.evaluate(&format!("delete globalThis['{}'];", name));
435                }
436            }
437            Ok(json!({}))
438        }
439        "runIfWaitingForDebugger" => Ok(json!({})),
440        "getExceptionDetails" => Ok(json!({ "exceptionDetails": null })),
441        "discardConsoleEntries" => Ok(json!({})),
442        _ => Err(format!("Unknown Runtime method: {}", method)),
443    }
444}

446/// Reject `Runtime.{evaluate,callFunctionOn}` calls that target an execution
447/// context Telemaco has not advertised. Returns `Ok(())` when the parameter is
448/// absent (defaulting to the page's default context) or when the id matches
449/// one of `ctx.valid_context_ids`. Logs a debug trace on accept for #51.
450fn validate_context_id(
451    params: &Value,
452    field: &str,
453    ctx: &crate::dispatch::CdpContext,
454    method: &str,
455) -> Result<(), String> {
456    let Some(id) = params.get(field).and_then(|v| v.as_i64()) else {
457        return Ok(());
458    };
459    if !ctx.valid_context_ids.contains(&id) {
460        return Err(format!(
461            "Cannot find context with specified id: {}",
462            id
463        ));
464    }
465    tracing::debug!(
466        target: "telemaco_cdp::runtime",
467        "Runtime.{}: {}={} (single-isolate routing)",
468        method,
469        field,
470        id
471    );
472    Ok(())
473}

475fn remote_object_from_info(info: &RemoteObjectInfo) -> Value {
476    let mut obj = json!({ "type": info.js_type });
477
478    if let Some(ref subtype) = info.subtype {
479        obj["subtype"] = json!(subtype);
480    }
481
482    if !info.class_name.is_empty() {
483        obj["className"] = json!(info.class_name);
484    }
485
486    if !info.description.is_empty() {
487        obj["description"] = json!(info.description);
488    }
489
490    if let Some(ref oid) = info.object_id {
491        obj["objectId"] = json!(oid);
492    }
493
494    if let Some(ref value) = info.value {
495        obj["value"] = value.clone();
496    }
497
498    obj
499}

631    /// target exists. Real Chrome replies with `{}`; before the fix Telemaco
632    /// returned `{"error":{"code":-32601,"message":"No page"}}` and the
633    /// puppeteer connect flow died.
634    #[tokio::test]
635    async fn enable_succeeds_when_no_session_attached() {
636        let mut ctx = CdpContext::new();
637        let result = handle("enable", &json!({}), &mut ctx, &None)
638            .await

643    /// SEC-002 / #578 — Runtime.removeBinding must validate the binding name the
644    /// same way addBinding does. Before the fix the name was interpolated
645    /// straight into `delete globalThis['{name}']`, so a CDP client could break
646    /// out of the string delimiter and run arbitrary JS in the page. This drives
647    /// the real handler against a live page and asserts the injected statement
648    /// never executes.
649    #[tokio::test(flavor = "current_thread")]
650    async fn remove_binding_rejects_injection_in_name() {
651        let mut ctx = CdpContext::new();
652        let page_id = ctx.create_page();
653        let session = Some(format!("{page_id}-session"));
654        ctx.sessions.insert(session.clone().unwrap(), page_id);
655
656        crate::domains::page::handle(
657            "navigate",
658            &json!({ "url": "data:text/html,<p>hi</p>", "waitUntil": "load" }),
659            &mut ctx,
660            &session,
661        )
662        .await
663        .expect("navigate should succeed");
664
665        // Canary the injection would flip from 0 to 1.
666        ctx.get_session_page_mut(&session)
667            .unwrap()
668            .evaluate("globalThis.__pwned = 0");
669
670        // The generated code is `delete globalThis['{name}']`, which the runtime
671        // wraps as `return ( ... )`. A comma-expression payload stays a single
672        // valid expression through that wrapper and runs the assignment:
673        //   delete globalThis['x'] , (globalThis.__pwned = 1) , globalThis['y']
674        handle(
675            "removeBinding",
676            &json!({ "name": "x'] , (globalThis.__pwned = 1) , globalThis['y" }),
677            &mut ctx,
678            &session,
679        )
680        .await
681        .expect("removeBinding must return Ok regardless of the name");
682
683        let pwned = ctx
684            .get_session_page_mut(&session)
685            .unwrap()
686            .evaluate("globalThis.__pwned");
687        assert_ne!(
688            pwned.as_f64(),
689            Some(1.0),
690            "removeBinding must not execute JS injected via the binding name (got {pwned:?})"
691        );
692    }
