1/// Helpers shared across CDP domain handlers.
2///
3/// The file-scheme detector here is reused by every CDP entrypoint that can
4/// trigger a navigation, so we don't end up with one domain enforcing the
5/// `--allow-file-access` gate and another silently letting `file://` through
6/// (see GHSA-q55h-vfv9-qcr5 and its incomplete-fix variant in
7/// `Target.createTarget`).
8
9/// Returns true when `raw` parses as a `file:`-scheme URL, or syntactically
10/// starts with `file:` after a possible leading-whitespace strip. Matching is
11/// case-insensitive on the scheme so neither `FILE://` nor `File://` slips
12/// past callers that gate on `file://`.
13pub(crate) fn url_is_file_scheme(raw: &str) -> bool {
14    url::Url::parse(raw)
15        .map(|u| u.scheme().eq_ignore_ascii_case("file"))
16        .unwrap_or_else(|_| {
17            raw.trim_start().to_ascii_lowercase().starts_with("file:")
18        })
19}

21/// Embed an objectId as a JS string literal, quotes included.
22///
23/// objectIds are interpolated into generated snippets that look up
24/// `__telemaco_objects[<here>]`. Escaping `\` and `'` by hand covers the two
25/// characters that close a single-quoted literal and leaves every C0 control
26/// alone - and a raw newline ends a JS string literal just as a stray quote
27/// does. The snippet is then a syntax error, the lookup yields nothing, and
28/// the caller is told nothing: `Runtime.getProperties` answers with an empty
29/// property list and `DOM.describeNode` falls back to node 0, so a client is
30/// handed the document where it asked for an element.
31///
32/// Not an injection vector, and it never was: the worst case is an
33/// unterminated string, i.e. a clean resolution failure. What makes it worth
34/// removing is that the failure is invisible from the outside.
35///
36/// The ids are not all ours to shape. `Runtime.getProperties` mints child ids
37/// as `parentId + "::" + key`, and the key is a property name off a page
38/// object, so the page decides what ends up inside the literal.
39///
40/// JSON string syntax is a subset of JavaScript's, so serialising the id gives
41/// a double-quoted literal that escapes `"`, `\` and every C0 control at once.
42/// Callers interpolate the result *without* adding quotes of their own.
43pub(crate) fn object_id_literal(oid: &str) -> String {
44    serde_json::to_string(oid).unwrap_or_else(|_| "\"\"".to_string())
45}

47/// Truncate `s` to at most `max` bytes, never splitting a UTF-8 character.
48///
49/// `&s[..max]` panics when `max` lands inside a multi-byte character, and the
50/// strings we truncate for log previews are attacker-controlled (raw WebSocket
51/// frames, intercepted URLs). A single frame whose byte `max` straddles a
52/// multi-byte char would otherwise panic the CDP processor task
53/// (`str::floor_char_boundary` would do this but is still unstable).
54pub(crate) fn truncate_on_char_boundary(s: &str, max: usize) -> &str {
55    if s.len() <= max {
56        return s;
57    }
58    let mut end = max;
59    while end > 0 && !s.is_char_boundary(end) {
60        end -= 1;
61    }
62    &s[..end]
63}

65#[cfg(test)]
66mod tests {
67    use super::*;
68
69    #[test]
70    fn matches_plain_file_url() {
71        assert!(url_is_file_scheme("file:///etc/passwd"));
72    }
73
74    #[test]
75    fn matches_case_insensitively() {
76        assert!(url_is_file_scheme("FILE:///etc/passwd"));
77        assert!(url_is_file_scheme("File:///etc/passwd"));
78        assert!(url_is_file_scheme("fIlE:///etc/passwd"));
79    }
80
81    #[test]
82    fn matches_with_leading_whitespace_fallback() {
83        // url::Url::parse rejects leading whitespace, but the syntactic
84        // fallback still catches `   file:...` so callers can't be tricked
85        // into letting it through.
86        assert!(url_is_file_scheme("   file:///etc/passwd"));
87    }
88
89    #[test]
90    fn rejects_http_https_about_data() {
91        assert!(!url_is_file_scheme("http://example.com"));
92        assert!(!url_is_file_scheme("https://example.com"));
93        assert!(!url_is_file_scheme("about:blank"));
94        assert!(!url_is_file_scheme("data:text/plain,hi"));
95        assert!(!url_is_file_scheme(""));
96    }
97
98    #[test]
99    fn rejects_lookalikes_that_are_not_file_scheme() {
100        // The URL parser rejects these (no `://`), so the syntactic fallback
101        // kicks in. `file` appearing anywhere except as the leading scheme
102        // must not match.
103        assert!(!url_is_file_scheme("notfile:///x"));
104        assert!(!url_is_file_scheme("http://file/"));
105    }
106
107    #[test]
108    fn object_id_literal_quotes_and_escapes_what_the_hand_rolled_pair_did() {
109        // The pair this replaced was
110        //   oid.replace('\\', "\\\\").replace('\'', "\\'")
111        // feeding a single-quoted '{}'. A double-quoted JSON literal needs no
112        // escape for `'` and adds one for `"`, so both quote characters are
113        // covered rather than one.
114        assert_eq!(object_id_literal("plain"), r#""plain""#);
115        assert_eq!(object_id_literal(r"x\"), r#""x\\""#);
116        assert_eq!(object_id_literal("a'b"), r#""a'b""#);
117        assert_eq!(object_id_literal("a\"b"), r#""a\"b""#);
118        // Carried over from the dom.rs helper this replaces: the backslash has
119        // to be doubled before the quote is looked at, or `a\'b` comes out with
120        // the escape attached to the wrong character.
121        assert_eq!(object_id_literal(r"a\'b"), r#""a\\'b""#);
122    }
123
124    #[test]
125    fn object_id_literal_escapes_the_controls_the_hand_rolled_pair_missed() {
126        // A raw LF or CR terminates a JS string literal, so these produced a
127        // syntax error and a silent resolution failure rather than a lookup.
128        // NUL parses but truncates the id against anything C-string shaped.
129        assert_eq!(object_id_literal("a\nb"), r#""a\nb""#);
130        assert_eq!(object_id_literal("a\rb"), r#""a\rb""#);
131        assert_eq!(object_id_literal("a\tb"), r#""a\tb""#);
132        assert_eq!(object_id_literal("a\0b"), r#""a\u0000b""#);
133    }
134
135    #[test]
136    fn object_id_literal_survives_the_shape_the_runtime_actually_mints() {
137        // Ids are JSON documents themselves, so the literal has to carry a
138        // second round of quoting without losing the id.
139        let oid = r#"{"injectedScriptId":1,"id":7}"#;
140        assert_eq!(
141            object_id_literal(oid),
142            r#""{\"injectedScriptId\":1,\"id\":7}""#
143        );
144
145        // And a getProperties child id is that, plus `::`, plus a page-chosen
146        // property name - the input this helper exists for.
147        let child = format!("{oid}::lf\nx");
148        assert_eq!(
149            object_id_literal(&child),
150            r#""{\"injectedScriptId\":1,\"id\":7}::lf\nx""#
151        );
152    }
153
154    #[test]
155    fn truncate_never_splits_a_multibyte_char() {
156        // 199 ASCII bytes + '€' (3 bytes, occupying indices 199..=201): byte 200
157        // falls inside the '€'. This is exactly the shape of a malformed CDP
158        // frame that would reach the `warn!("Invalid CDP: ...")` preview.
159        let s = format!("{}€tail", "a".repeat(199));
160        assert!(!s.is_char_boundary(200), "setup: byte 200 splits the € char");
161
162        // The old logging code did `&s[..s.len().min(200)]`, which panics here —
163        // a single crafted frame would take down the CDP processor task.
164        let naive = std::panic::catch_unwind(|| {
165            let _ = &s[..s.len().min(200)];
166        });
167        assert!(naive.is_err(), "raw byte slice at a non-char-boundary must panic");
168
169        // The helper truncates on the boundary before the € instead.
170        let safe = truncate_on_char_boundary(&s, 200);
171        assert!(s.starts_with(safe));
172        assert!(safe.len() <= 200);
173        assert_eq!(safe.len(), 199, "should stop right before the € char");
174    }
175
176    #[test]
177    fn truncate_returns_whole_string_when_short() {
178        assert_eq!(truncate_on_char_boundary("hi", 200), "hi");
179        assert_eq!(truncate_on_char_boundary("", 10), "");
180        // Exact-length and exact-boundary cases are returned unchanged.
181        assert_eq!(truncate_on_char_boundary("abc", 3), "abc");
182    }
