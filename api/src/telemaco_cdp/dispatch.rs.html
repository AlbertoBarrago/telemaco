40pub struct CdpContext {
41    pub pages: Vec<Page>,
42    pub sessions: HashMap<String, String>, // session_id -> page_id
43    /// Current document loader per page. Navigation events and later
44    /// script-initiated Network events must share this id; inventing a loader
45    /// for each fetch breaks DevTools request grouping.
46    pub current_loader_ids: HashMap<String, String>,
47    /// Child frame ids already reported to the client, per page, so each frame
48    /// is announced once and a frame that goes away can be retracted.
49    pub announced_frames: HashMap<String, Vec<String>>,
50    pub pending_events: Vec<CdpEvent>,
51    #[cfg(feature = "render")]
52    pub(crate) screencasts: HashMap<String, ScreencastState>,
53    #[cfg(feature = "render")]
54    next_screencast_session_id: i64,
55    pub default_context: Arc<BrowserContext>,
56    pub browser_contexts: HashMap<String, Arc<BrowserContext>>,
57    page_counter: u32,
58    browser_context_counter: u32,
59    target_session_counter: u64,
60    pub preload_scripts: Vec<(String, String)>, // (identifier, source)
61    pub preload_counter: u32,
62    // Which sessions asked for each `Runtime.addBinding` name. A binding is a
63    // session-scoped subscription in CDP, and a client discards any event whose
64    // sessionId is not one it holds, so the call has to go back to the session
65    // that registered the name rather than to whichever session of the page
66    // happens to come first out of a HashMap.
67    pub binding_sessions: HashMap<String, Vec<String>>, // binding name -> session ids
68    /// Sessions that called Runtime.enable. Console and exception events are
69    /// page-scoped but only delivered to these subscribers.
70    pub runtime_enabled_sessions: HashSet<String>,
71    // World names registered via Page.createIsolatedWorld. After every
72    // navigation Telemaco clears execution contexts (via
73    // Runtime.executionContextsCleared) and must re-emit a
74    // Runtime.executionContextCreated for each registered world, otherwise
75    // Playwright/Puppeteer hang waiting for their utility world to come
76    // back. Stored as plain Strings (not by-page) — for now we only model
77    // a single page in CdpContext anyway.
78    pub isolated_worlds: Vec<String>,
79    // Set of executionContextIds Telemaco has emitted via
80    // Runtime.executionContextCreated. Pre-populated with the default-frame
81    // contexts (`1`, `2`) that Runtime.enable / Page.navigate emit, then
82    // extended each time Page.createIsolatedWorld assigns a fresh id.
83    //
84    // Runtime.evaluate / Runtime.callFunctionOn consult this set to reject
85    // requests targeting an unknown context — matching real Chrome's
86    // "Cannot find context with specified id" CDP error and unblocking the
87    // Playwright locator path described in issue #51.
88    pub valid_context_ids: HashSet<i64>,
89    // Monotonic counter for isolated-world execution context ids. Issue #192:
90    // both `Page.createIsolatedWorld` and `do_navigate` used to hardcode id
91    // 100, so re-creating a context (e.g. Playwright opening a second page or
92    // re-navigating) silently emitted the same id twice and the client's
93    // bookkeeping diverged from the server's. Each fresh isolated context now
94    // claims and increments from this counter so the ids real Chrome would
95    // emit (incrementing, never reused) are mirrored.
96    pub next_isolated_context_id: i64,
97    pub fetch_intercept: FetchInterceptState,
98    pub intercept_tx: Option<tokio::sync::mpsc::UnboundedSender<InterceptedRequest>>,
99    // Open IO streams for Fetch.takeResponseBodyAsStream. Each holds a response
100    // body taken out of the page cache so a large download is streamed
101    // chunk-by-chunk via IO.read and freed on IO.close (issue #360). The store
102    // caps how many bodies (and how many bytes) can be held at once, evicting
103    // the oldest, so an abandoned or disconnected stream cannot leak unbounded.
104    pub io_streams: crate::domains::io::IoStreamStore,
105    /// Serializes V8 work within THIS connection. With the thread-per-connection
106    /// server (#430) each connection runs on its own OS thread, so isolates never
107    /// collide across connections; this per-connection lock keeps a connection's
108    /// own nav task and command dispatch from interleaving two of its pages'
109    /// isolates on that one thread. It is deliberately per-connection, not a
110    /// process-wide lock, so connections run in parallel (measured ~2x at
111    /// concurrency 2, ~3x at 4) instead of serializing all V8 on one mutex.
112    pub v8_lock: Arc<tokio::sync::Mutex<()>>,
113}

229    pub fn create_page_in_context(&mut self, context_id: Option<&str>) -> Result<String, String> {
230        let context = match context_id {
231            Some(id) => self
232                .browser_context(id)
233                .cloned()
234                .ok_or_else(|| format!("Browser context not found: {}", id))?,
235            None => self.default_context.clone(),
236        };
237        self.page_counter += 1;
238        let page_id = format!("page-{}", self.page_counter);
239        let mut page = Page::new(page_id.clone(), context);
240        page.navigate_blank();
241        self.pages.push(page);
242        self.current_loader_ids
243            .insert(page_id.clone(), format!("loader-blank-{page_id}"));
244        Ok(page_id)
245    }

273    pub fn dispose_browser_context(&mut self, id: &str) -> Result<Vec<String>, String> {
274        if id == self.default_context.id {
275            return Err("The default browser context cannot be disposed".to_string());
276        }
277        if self.browser_contexts.remove(id).is_none() {
278            return Err(format!("Browser context not found: {}", id));
279        }
280
281        let page_ids: Vec<String> = self
282            .pages
283            .iter()
284            .filter(|page| page.context.id == id)
285            .map(|page| page.id.clone())
286            .collect();
287        for page_id in &page_ids {
288            self.remove_page(page_id);
289        }
290        Ok(page_ids)
291    }

312    pub fn remove_page(&mut self, id: &str) {
313        let removed_sessions: Vec<String> = self
314            .sessions
315            .iter()
316            .filter(|(_, page_id)| page_id.as_str() == id)
317            .map(|(session_id, _)| session_id.clone())
318            .collect();
319        self.pages.retain(|p| p.id != id);
320        self.current_loader_ids.remove(id);
321        self.announced_frames.remove(id);
322        #[cfg(feature = "render")]
323        {
324            for session_id in &removed_sessions {
325                self.screencasts.remove(session_id);
326            }
327        }
328        for session_id in &removed_sessions {
329            self.runtime_enabled_sessions.remove(session_id);
330        }
331        self.sessions.retain(|_, v| v != id);
332    }

346    pub fn get_session_page_mut(&mut self, session_id: &Option<String>) -> Option<&mut Page> {
347        let page_id = session_id
348            .as_ref()
349            .and_then(|sid| self.sessions.get(sid))
350            .cloned()?;
351
352        let target_has_js = self.pages.iter().any(|p| p.id == page_id && p.has_js());
353
354        if !target_has_js {
355            for page in &mut self.pages {
356                if page.id != page_id && page.has_js() {
357                    page.suspend_js();
358                    break;
359                }
360            }
361            if let Some(target) = self.pages.iter_mut().find(|p| p.id == page_id) {
362                target.resume_js();
363            }
364        }
365
366        self.get_page_mut(&page_id)
367    }

443pub async fn dispatch(req: &CdpRequest, ctx: &mut CdpContext) -> CdpResponse {
444    // headless_chrome (and older Puppeteer) wrap every CDP call inside
445    // Target.sendMessageToTarget. Unwrap and recurse BEFORE acquiring the
446    // per-connection V8 lock — the recursive dispatch will acquire it for the
447    // inner call, and tokio Mutex is not reentrant.
448    if req.method == "Target.sendMessageToTarget" {
449        return dispatch_send_message_to_target(req, ctx).await;
450    }
451
452    // Issue #430: keep this connection's V8 work serialized on its own thread.
453    //
454    // Every CDP handler below may call into a per-Page `JsRuntime` (each owning
455    // its own V8 Isolate). With the thread-per-connection server each connection
456    // runs on its own OS thread, so isolates never collide across connections.
457    // Within one connection, though, a nav task spawned by
458    // `process_with_interception` runs while this processor keeps pumping other
459    // CDP messages, so two of this connection's pages could still interleave V8
460    // work on this one thread and trip
461    // `heap->isolate() == Isolate::TryGetCurrent()`. The per-connection lock
462    // (`ctx.v8_lock`) keeps each handler contiguous: V8 fully exits one Isolate
463    // before the next of this connection's pages is allowed in. It is
464    // per-connection, not process-wide, so other connections run in parallel.
465    //
466    // Optimization: methods that demonstrably never touch V8 bypass the lock
467    // (Puppeteer's newPage() setup issues ~8 such calls). Each listed method was
468    // audited to confirm it never reaches `JsRuntime::execute_script` or DOM
469    // mutation that re-enters V8; `get_session_page_mut` (which can trigger
470    // `suspend_js`/`resume_js`) is NOT in the list.
471    let _v8_guard = if is_v8_free_method(&req.method) {
472        None
473    } else {
474        // Per-connection lock (owned guard, so it does not borrow `ctx`): keeps
475        // this connection's own V8 work contiguous on its thread without
476        // serializing other connections (#430).
477        Some(ctx.v8_lock.clone().lock_owned().await)
478    };
479
480    // Per-command V8 watchdog. The lock above keeps each handler contiguous on
481    // the thread, but it does not bound how long a handler runs: a hung page (a
482    // runaway Runtime.evaluate, a synchronous DOM op) would hold this
483    // connection's V8 lock and wedge its other sessions forever. The one-shot
484    // CLI uses a process-level hard deadline for this; the long-running server
485    // cannot force-exit, so we terminate just the offending isolate instead.
486    // TELEMACO_CDP_COMMAND_TIMEOUT_MS tunes the bound (0 disables); the default
487    // leaves headroom for the slowest legitimate navigation, which already
488    // self-bounds via TELEMACO_NAV_TIMEOUT_MS plus the watchdog-bounded settle.
489    let cmd_budget_ms: u64 = std::env::var("TELEMACO_CDP_COMMAND_TIMEOUT_MS")
490        .ok()
491        .and_then(|s| s.parse().ok())
492        .unwrap_or(60_000);
493    let cmd_watchdog = if cmd_budget_ms == 0 || is_v8_free_method(&req.method) {
494        None
495    } else {
496        ctx.get_session_page(&req.session_id)
497            .and_then(|p| p.isolate_handle())
498            .map(|h| {
499                telemaco_js::cdp_watchdog::arm(h, std::time::Duration::from_millis(cmd_budget_ms))
500            })
501    };
502
503    let (domain, method) = match req.method.split_once('.') {
504        Some((d, m)) => (d, m),
505        None => {
506            return CdpResponse::error(
507                req.id,
508                -32601,
509                format!("Invalid method format: {}", req.method),
510                req.session_id.clone(),
511            );
512        }
513    };
514
515    let result = match domain {
516        "Target" => domains::target::handle(method, &req.params, ctx, &req.session_id).await,
517        "Browser" => domains::browser::handle(method, &req.params).await,
518        "Page" => domains::page::handle(method, &req.params, ctx, &req.session_id).await,
519        "DOM" => domains::dom::handle(method, &req.params, ctx, &req.session_id).await,
520        "DOMSnapshot" => {
521            domains::domsnapshot::handle(method, &req.params, ctx, &req.session_id).await
522        }
523        "Runtime" => domains::runtime::handle(method, &req.params, ctx, &req.session_id).await,
524        "Network" => domains::network::handle(method, &req.params, ctx, &req.session_id).await,
525        "Fetch" => domains::fetch::handle(method, &req.params, ctx, &req.session_id).await,
526        "IO" => domains::io::handle(method, &req.params, ctx).await,
527        "Input" => domains::input::handle(method, &req.params, ctx, &req.session_id).await,
528        "Emulation" => domains::emulation::handle(method, &req.params, ctx, &req.session_id).await,
529        "Storage" => domains::storage::handle(method, &req.params, ctx, &req.session_id).await,
530        "LP" => domains::lp::handle(method, &req.params, ctx, &req.session_id).await,
531        "Accessibility" => {
532            domains::accessibility::handle(method, &req.params, ctx, &req.session_id).await
533        }
534        // Accepted but no-op. Puppeteer's FrameManager.initialize calls
535        // Audits.enable on connect — refusing it breaks puppeteer.connect()
536        // before any user code runs.
537        "Log" | "Performance" | "Security" | "CSS" | "ServiceWorker" | "Inspector" | "Debugger"
538        | "Profiler" | "HeapProfiler" | "Overlay" | "Audits" => Ok(json!({})),
539        _ => Err(format!("Unknown domain: {}", domain)),
540    };
541
542    #[cfg(feature = "render")]
543    if result.is_ok() && domains::page::command_can_change_screencast_frame(&req.method) {
544        if let Err(error) = domains::page::queue_screencast_frame(ctx, &req.session_id, false) {
545            // Frame delivery is an asynchronous side effect in Chromium; it
546            // must not rewrite an otherwise successful command response.
547            tracing::warn!(method = %req.method,
548                "could not produce command-driven screencast frame: {error}");
549        }
550    }
551
552    // Stop the per-command watchdog. If it fired (the handler held V8 past the
553    // budget), V8 is left in a terminating state, so clear that flag before the
554    // next command runs on this page.
555    if let Some(wd) = cmd_watchdog {
556        if telemaco_js::cdp_watchdog::disarm(wd) {
557            tracing::warn!(
558                "CDP command {} held V8 past {}ms; terminated the isolate to free the dispatcher",
559                req.method,
560                cmd_budget_ms
561            );
562            if let Some(page) = ctx.get_session_page_mut(&req.session_id) {
563                page.cancel_v8_termination();
564            }
565        }
566    }
567
568    drain_runtime_events(ctx);
569    drain_binding_calls(ctx);
570    drain_frame_events(ctx);
571
572    match result {
573        Ok(value) => CdpResponse::success(req.id, value, req.session_id.clone()),
574        Err(msg) => {
575            tracing::warn!("CDP error for {}: {}", req.method, msg);
576            CdpResponse::error(req.id, -32601, msg, req.session_id.clone())
577        }
578    }
579}

581pub(crate) fn drain_runtime_events(ctx: &mut CdpContext) {
582    let mut page_to_sessions: HashMap<&str, Vec<&str>> = HashMap::new();
583    for session_id in &ctx.runtime_enabled_sessions {
584        if let Some(page_id) = ctx.sessions.get(session_id) {
585            page_to_sessions
586                .entry(page_id.as_str())
587                .or_default()
588                .push(session_id.as_str());
589        }
590    }
591    for sessions in page_to_sessions.values_mut() {
592        sessions.sort_unstable();
593    }
594
595    let mut events = Vec::new();
596    for page in &mut ctx.pages {
597        let runtime_events = page.take_pending_runtime_events();
598        if runtime_events.is_empty() {
599            continue;
600        }
601        let Some(sessions) = page_to_sessions.get(page.id.as_str()) else {
602            continue;
603        };
604        let execution_context_id = if ctx
605            .current_loader_ids
606            .get(&page.id)
607            .is_some_and(|loader_id| loader_id.starts_with("loader-blank-"))
608        {
609            1
610        } else {
611            2
612        };
613        for runtime_event in runtime_events {
614            for session_id in sessions {
615                let (method, params) = match &runtime_event {
616                    telemaco_js::ops::RuntimeEvent::Console(event) => (
617                        "Runtime.consoleAPICalled",
618                        json!({
619                            "type": event.kind,
620                            "args": event.args,
621                            "executionContextId": execution_context_id,
622                            "timestamp": event.timestamp,
623                        }),
624                    ),
625                    telemaco_js::ops::RuntimeEvent::Exception(event) => (
626                        "Runtime.exceptionThrown",
627                        json!({
628                            "timestamp": event.timestamp,
629                            "exceptionDetails": {
630                                "exceptionId": event.exception_id,
631                                "text": "Uncaught",
632                                "lineNumber": event.line_number,
633                                "columnNumber": event.column_number,
634                                "scriptId": "",
635                                "url": event.url,
636                                "stackTrace": { "callFrames": event.stack_trace },
637                                "executionContextId": execution_context_id,
638                                "exception": {
639                                    "type": "object",
640                                    "subtype": "error",
641                                    "className": event.name,
642                                    "description": event.description,
643                                },
644                            },
645                        }),
646                    ),
647                };
648                events.push(CdpEvent {
649                    method: method.to_string(),
650                    params,
651                    session_id: Some((*session_id).to_string()),
652                });
653            }
654        }
655    }
656    ctx.pending_events.extend(events);
657}

659// Drain every page's binding-call queue (filled by op_binding_called when
660// page JS invokes a `Runtime.addBinding` shim) and turn each entry into a
661// Runtime.bindingCalled CDP event that the writer task forwards to the
662// connected client. Called after every dispatch — binding calls only land
663// in the queue while V8 is running inside a CDP handler, so there is no
664// window in which they could pile up without a draining opportunity.
665pub(crate) fn drain_binding_calls(ctx: &mut CdpContext) {
666    // page_id -> every session on that page. A page commonly has more than one:
667    // Target.createTarget opens a session and the Target.attachToTarget that
668    // follows opens another, so a client that reaches a page the ordinary way
669    // holds two and uses the second.
670    let mut page_to_sessions: HashMap<&str, Vec<&str>> = HashMap::new();
671    for (session_id, page_id) in &ctx.sessions {
672        page_to_sessions
673            .entry(page_id.as_str())
674            .or_default()
675            .push(session_id.as_str());
676    }
677    // ctx.sessions is a HashMap, so fix an order the events can be asserted in.
678    for sessions in page_to_sessions.values_mut() {
679        sessions.sort_unstable();
680    }
681
682    let mut events: Vec<CdpEvent> = Vec::new();
683    for page in &mut ctx.pages {
684        let calls = page.take_pending_binding_calls();
685        if calls.is_empty() {
686            continue;
687        }
688        let Some(page_sessions) = page_to_sessions.get(page.id.as_str()) else {
689            // No session attached — drop the calls; there is no client to
690            // deliver them to.
691            continue;
692        };
693        for (name, payload) in calls {
694            // The sessions that asked for this binding, narrowed to the page the
695            // call came from. Falling back to every session of the page keeps a
696            // binding that was installed without a session (a preload, or a
697            // direct embedder) deliverable rather than silently dropped.
698            let registered = ctx.binding_sessions.get(&name);
699            let targets: Vec<&str> = page_sessions
700                .iter()
701                .copied()
702                .filter(|session| {
703                    registered.is_none_or(|owners| owners.iter().any(|owner| owner == session))
704                })
705                .collect();
706            let targets = if targets.is_empty() {
707                page_sessions.clone()
708            } else {
709                targets
710            };
711            for session_id in targets {
712                events.push(CdpEvent {
713                    method: "Runtime.bindingCalled".into(),
714                    // Use executionContextId=2: the default main-frame context
715                    // emitted post-navigation (see domains/page.rs phase1).
716                    // Puppeteer matches on session_id + binding name and
717                    // tolerates any registered context id.
718                    params: json!({
719                        "name": name,
720                        "payload": payload,
721                        "executionContextId": 2,
722                    }),
723                    session_id: Some(session_id.to_string()),
724                });
725            }
726        }
727    }
728    ctx.pending_events.extend(events);
729}

731// Announce child frames the client has not been told about yet, and retract
732// the ones that are gone.
733//
734// A frame is built when the page settles, which is not necessarily during the
735// navigation that created it: script can add an iframe at any time, and the
736// settle that gives it a realm may belong to a later command. Diffing here,
737// after every dispatch, reports a frame whenever it actually appears instead
738// of only at navigation, and is the same drain point binding calls use.
739pub(crate) fn drain_frame_events(ctx: &mut CdpContext) {
740    // Every session on the page, not just one: a client that reaches a page the
741    // ordinary way holds two of them, because Target.createTarget opens a
742    // session and the Target.attachToTarget that follows opens another. A
743    // client drops any event whose sessionId is not the one it attached with,
744    // so announcing to an arbitrary session is the same as not announcing.
745    let mut page_to_sessions: HashMap<String, Vec<String>> = HashMap::new();
746    for (session_id, page_id) in &ctx.sessions {
747        page_to_sessions
748            .entry(page_id.clone())
749            .or_default()
750            .push(session_id.clone());
751    }
752    // ctx.sessions is a HashMap, so fix an order the events can be asserted in.
753    for sessions in page_to_sessions.values_mut() {
754        sessions.sort();
755    }
756
757    let mut events: Vec<CdpEvent> = Vec::new();
758    let mut announced: HashMap<String, Vec<String>> = HashMap::new();
759    for page in &ctx.pages {
760        let Some(session_ids) = page_to_sessions.get(&page.id) else {
761            continue;
762        };
763        let live = crate::domains::page::child_frame_values(page);
764        let known = ctx.announced_frames.get(&page.id);
765        let live_ids: Vec<String> = live
766            .iter()
767            .map(|frame| frame["id"].as_str().unwrap_or_default().to_string())
768            .collect();
769
770        for frame in &live {
771            let id = frame["id"].as_str().unwrap_or_default();
772            if known.is_some_and(|ids| ids.iter().any(|seen| seen == id)) {
773                continue;
774            }
775            for session_id in session_ids {
776                // Attach before navigate: a client builds its frame from the
777                // attach event and treats a navigation of a frame it has never
778                // seen as a protocol error.
779                events.push(CdpEvent {
780                    method: "Page.frameAttached".into(),
781                    params: json!({
782                        "frameId": id,
783                        "parentFrameId": frame["parentId"].as_str().unwrap_or_default(),
784                    }),
785                    session_id: Some(session_id.clone()),
786                });
787                events.push(CdpEvent {
788                    method: "Page.frameNavigated".into(),
789                    params: json!({ "frame": frame, "type": "Navigation" }),
790                    session_id: Some(session_id.clone()),
791                });
792                // The frame's document scripts have already run by the time it
793                // is in this list, so it is not still loading.
794                events.push(CdpEvent {
795                    method: "Page.frameStoppedLoading".into(),
796                    params: json!({ "frameId": id }),
797                    session_id: Some(session_id.clone()),
798                });
799            }
800        }
801
802        if let Some(known) = known {
803            for id in known {
804                if !live_ids.contains(id) {
805                    for session_id in session_ids {
806                        events.push(CdpEvent {
807                            method: "Page.frameDetached".into(),
808                            params: json!({ "frameId": id, "reason": "remove" }),
809                            session_id: Some(session_id.clone()),
810                        });
811                    }
812                }
813            }
814        }
815        announced.insert(page.id.clone(), live_ids);
816    }
817    ctx.announced_frames.extend(announced);
818    ctx.pending_events.extend(events);
819}

821async fn dispatch_send_message_to_target(req: &CdpRequest, ctx: &mut CdpContext) -> CdpResponse {
822    let session_id = req
823        .params
824        .get("sessionId")
825        .and_then(|v| v.as_str())
826        .map(str::to_string);
827    let message = match req.params.get("message").and_then(|v| v.as_str()) {
828        Some(m) => m,
829        None => {
830            return CdpResponse::error(
831                req.id,
832                -32602,
833                "sendMessageToTarget requires a message string".into(),
834                req.session_id.clone(),
835            );
836        }
837    };
838
839    let inner: CdpRequest = match serde_json::from_str(message) {
840        Ok(r) => r,
841        Err(e) => {
842            return CdpResponse::error(
843                req.id,
844                -32700,
845                format!("sendMessageToTarget message is not a valid CDP request: {e}"),
846                req.session_id.clone(),
847            );
848        }
849    };
850
851    // Override the inner session with the one supplied by the wrapper so
852    // the inner dispatch routes against the right page. Boxing the future
853    // sidesteps the async-fn recursion limit.
854    let inner_with_session = CdpRequest {
855        id: inner.id,
856        method: inner.method.clone(),
857        params: inner.params,
858        session_id: session_id.clone().or(inner.session_id),
859    };
860    let inner_response = Box::pin(dispatch(&inner_with_session, ctx)).await;
861
862    // Re-emit the inner response as the legacy event headless_chrome (and
863    // older Puppeteer) listen for instead of correlating responses by id.
864    let inner_serialized = serde_json::to_string(&inner_response).unwrap_or_else(|_| "{}".into());
865    ctx.pending_events.push(CdpEvent {
866        method: "Target.receivedMessageFromTarget".to_string(),
867        params: json!({
868            "sessionId": session_id.clone().unwrap_or_default(),
869            "message": inner_serialized,
870            "targetId": session_id.clone().unwrap_or_default(),
871        }),
872        session_id: req.session_id.clone(),
873    });
874
875    CdpResponse::success(req.id, json!({}), req.session_id.clone())
876}
