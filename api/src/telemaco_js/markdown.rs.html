1//! Shared markdown extraction script used by the LP.getMarkdown CDP method
2//! and the CLI `--dump markdown` mode. Lives in telemaco-browser so both
3//! telemaco-cdp and telemaco-cli can call it without depending on each other.
4
5/// JS expression that walks `document.body` and returns a markdown string.
6/// Must be evaluated against a Page that has a fully-bootstrapped JS runtime.
7pub const HTML_TO_MARKDOWN_JS: &str = r#"
8(function() {
9    function toMd(el, depth) {
10        if (!el) return '';
11        var out = '';
12        if (el.nodeType === 3) return el.textContent || '';
13        if (el.nodeType !== 1) return '';
14        var tag = (el.tagName || '').toLowerCase();
15        var children = '';
16        var cn = el.childNodes || [];
17        for (var i = 0; i < cn.length; i++) children += toMd(cn[i], depth);
18        children = children.replace(/\n{3,}/g, '\n\n');
19        switch(tag) {
20            case 'h1': return '\n# ' + children.trim() + '\n\n';
21            case 'h2': return '\n## ' + children.trim() + '\n\n';
22            case 'h3': return '\n### ' + children.trim() + '\n\n';
23            case 'h4': return '\n#### ' + children.trim() + '\n\n';
24            case 'h5': return '\n##### ' + children.trim() + '\n\n';
25            case 'h6': return '\n###### ' + children.trim() + '\n\n';
26            case 'p': return '\n' + children.trim() + '\n\n';
27            case 'br': return '\n';
28            case 'hr': return '\n---\n\n';
29            case 'strong': case 'b': return '**' + children + '**';
30            case 'em': case 'i': return '*' + children + '*';
31            case 'code': return '`' + children + '`';
32            case 'pre': return '\n```\n' + children + '\n```\n\n';
33            case 'blockquote': return '\n> ' + children.trim().replace(/\n/g, '\n> ') + '\n\n';
34            case 'a':
35                var href = el.getAttribute('href') || '';
36                if (href && children.trim()) return '[' + children.trim() + '](' + href + ')';
37                return children;
38            case 'img':
39                var src = el.getAttribute('src') || '';
40                var alt = el.getAttribute('alt') || '';
41                return '![' + alt + '](' + src + ')';
42            case 'ul': case 'ol':
43                return '\n' + children + '\n';
44            case 'li':
45                var parent = el.parentNode;
46                var isOrdered = parent && parent.tagName && parent.tagName.toLowerCase() === 'ol';
47                var bullet = isOrdered ? '1. ' : '- ';
48                return bullet + children.trim() + '\n';
49            case 'table': return '\n' + children + '\n';
50            case 'thead': case 'tbody': case 'tfoot': return children;
51            case 'tr':
52                var cells = [];
53                var tds = el.childNodes || [];
54                for (var j = 0; j < tds.length; j++) {
55                    if (tds[j].nodeType === 1) cells.push(toMd(tds[j], depth).trim());
56                }
57                return '| ' + cells.join(' | ') + ' |\n';
58            case 'th': case 'td': return children;
59            case 'script': case 'style': case 'noscript': case 'link': case 'meta': return '';
60            case 'div': case 'section': case 'article': case 'main': case 'aside': case 'nav': case 'header': case 'footer':
61                return '\n' + children;
62            case 'span': return children;
63            default: return children;
64        }
65    }
66    var body = document.body || document.documentElement;
67    var md = toMd(body, 0);
68    md = md.replace(/\n{3,}/g, '\n\n').trim();
69    return md;
70})()
