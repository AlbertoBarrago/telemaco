1use std::sync::Once;
2
3static INIT: Once = Once::new();
4
5/// Apply user-supplied V8 flags exactly once, before the first isolate is
6/// created.
7///
8/// `flags` is a raw V8 flag string in the same form V8/Chromium/Node accept
9/// (e.g. `"--max-old-space-size=4096 --max-semi-space-size=64"`). An empty or
10/// whitespace-only string is a no-op and does not consume the one-shot guard,
11/// so a later non-empty call still takes effect.
12///
13/// V8 ignores `set_flags_from_string` once the platform is initialized, so the
14/// first non-empty call must run before any `JsRuntime` is constructed.
15/// Subsequent calls are silently dropped.
16pub fn set_v8_flags(flags: &str) {
17    let trimmed = flags.trim();
18    if trimmed.is_empty() {
19        return;
20    }
21    INIT.call_once(|| {
22        deno_core::v8::V8::set_flags_from_string(trimmed);
23    });
24}
