1//! The document's input stream for `document.write()`.
2//!
3//! There is one input stream per document, and the tokenizer carries its state across calls.
4//! A construct may therefore be split at any point, even in the middle of a tag name.
5//! <https://html.spec.whatwg.org/multipage/dynamic-markup-insertion.html#dom-document-write>
6//!
7//! This module inserts nothing itself. It creates the nodes and reports where each belongs,
8//! because `Node.appendChild` on the JS side reports the mutation at the same time, registers
9//! window named access, and loads a written stylesheet.
10
11use std::cell::RefCell;
12use std::collections::{HashMap, HashSet};
13
14use html5ever::driver::{parse_fragment, ParseOpts, Parser};
15use html5ever::tendril::TendrilSink;
16use html5ever::tree_builder::Tracer;
17use html5ever::{local_name, ns, QualName};
18use telemaco_dom::{DomTree, NodeData, NodeId};

20/// Collects every node the TreeBuilder still holds.
21///
22/// `TreeSink::pop` would be the obvious signal for "done" and does not deliver: of html5ever's
23/// three ways to drain the stack of open elements, only one reports to the sink. An end tag
24/// usually goes through `pop_until`, which does not.
25#[derive(Default)]
26struct RetainedNodes(RefCell<HashSet<NodeId>>);
27
28impl Tracer for RetainedNodes {
29    type Handle = NodeId;
30
31    fn trace_handle(&self, node: &NodeId) {
32        self.0.borrow_mut().insert(*node);
33    }
34}
35
36/// `parent` is `None` for a node at the head of the input stream, which belongs at the
37/// insertion point.
38pub(crate) struct Placement {
39    pub(crate) parent: Option<NodeId>,
40    pub(crate) node: NodeId,
41}
42
43/// A `<script>` runs as soon as it is inserted, so a half-written one must not.
44/// A `<template>` keeps its children in its own contents document, which a child walk does
45/// not reach, so it can only be copied as a whole.
46fn needs_to_be_complete(source: &DomTree, node: NodeId) -> bool {
47    source
48        .get_node(node)
49        .and_then(|n| n.as_element().map(|name| name.local.clone()))
50        .is_some_and(|local| local == local_name!("script") || local == local_name!("template"))
51}
52
53pub(crate) struct DocumentWriteStream {
54    parser: Parser<DomTree>,
55    /// Maps a node of the parser tree to its copy in the document. A node missing here has
56    /// not been handed over yet.
57    handed_over: HashMap<NodeId, NodeId>,
58    /// Staging for copying a whole subtree, reused.
59    staging: Option<NodeId>,
60}
61
62impl DocumentWriteStream {
63    pub(crate) fn new() -> Self {
64        // Scripting enabled, as with the fragment parser behind innerHTML.
65        let context = QualName::new(None, ns!(html), local_name!("body"));
66        DocumentWriteStream {
67            parser: parse_fragment(DomTree::new(), ParseOpts::default(), context, vec![], true),
68            handed_over: HashMap::new(),
69            staging: None,
70        }
71    }
72
73    /// Pushes a call's arguments into the input stream and mirrors what the parser made of them.
74    /// Returns the nodes to insert, parents before children.
75    pub(crate) fn write(&mut self, html: &str, dom: &DomTree) -> Vec<Placement> {
76        self.parser.process(html.into());
77
78        let retained = RetainedNodes::default();
79        self.parser.tokenizer.sink.trace_handles(&retained);
80        let retained = retained.0.into_inner();
81
82        // Separate fields: the parser tree is read, the mapping is written.
83        let source = &self.parser.tokenizer.sink.sink;
84        let handed_over = &mut self.handed_over;
85        let staging = &mut self.staging;
86
87        let root = source.fragment_root();
88        let mut placements = Vec::new();
89        // Only look at what is new. A walk over all children would cost as much per call as
90        // the input stream written so far is long, so quadratic over a thousand calls. Measured at
91        // 5000 calls: 852 ms versus 4130 ms.
92        let mut stack: Vec<NodeId> = fresh_children(source, root, handed_over);
93
94        while let Some(current) = stack.pop() {
95            let node = match source.get_node(current) {
96                Some(node) => node,
97                None => continue,
98            };
99
100            if let Some(&copy) = handed_over.get(&current) {
101                // A text node at the end of the input stream grows with every call and stays the
102                // same node. Elements no longer change after their creation.
103                if node.is_text() {
104                    let text = source.text_content(current);
105                    dom.with_node_mut(copy, |n| {
106                        if let NodeData::Text { contents } = &mut n.data {
107                            *contents = text;
108                        }
109                    });
110                }
111                for child in fresh_children(source, current, handed_over) {
112                    stack.push(child);
113                }
114                continue;
115            }
116
117            let parent = match node.parent {
118                Some(p) if p == root => None,
119                // If the parent is held back, this one waits along.
120                Some(p) => match handed_over.get(&p) {
121                    Some(&copy) => Some(copy),
122                    None => continue,
123                },
124                None => continue,
125            };
126
127            if needs_to_be_complete(source, current) {
128                if retained.contains(&current) {
129                    continue;
130                }
131                let container = *staging.get_or_insert_with(|| dom.new_node(NodeData::Document));
132                let Some(copy) = dom.import_node_from(container, source, current) else {
133                    continue;
134                };
135                dom.detach(copy);
136                map_subtree(source, current, dom, copy, handed_over);
137                placements.push(Placement { parent, node: copy });
138                continue;
139            }
140
141            // Handed over shallow and left to grow in place. That makes an element that is
142            // never closed appear at once instead of never.
143            let copy = dom.new_node(node.data.clone());
144            handed_over.insert(current, copy);
145            placements.push(Placement { parent, node: copy });
146            for child in fresh_children(source, current, handed_over) {
147                stack.push(child);
148            }
149        }
150
151        placements
152    }
153}

155/// The children of `parent` still to do, as a stack: the top element is processed first.
156///
157/// The walk goes backward from the last child and stops at the first one already handed over.
158/// The parser only appends at the back, so everything before it is done. This one already
159/// handed-over child comes back along, because a text node at the end of the input stream keeps
160/// growing and an open element still receives children.
161fn fresh_children(
162    source: &DomTree,
163    parent: NodeId,
164    handed_over: &HashMap<NodeId, NodeId>,
165) -> Vec<NodeId> {
166    let mut stack = Vec::new();
167    let mut current = source.get_node(parent).and_then(|n| n.last_child);
168    while let Some(id) = current {
169        stack.push(id);
170        if handed_over.contains_key(&id) {
171            break;
172        }
173        current = source.get_node(id).and_then(|n| n.prev_sibling);
174    }
175    stack
176}

178/// After copying, both trees have the same shape, so a lockstep walk aligns the nodes.
179fn map_subtree(
180    source: &DomTree,
181    source_node: NodeId,
182    dom: &DomTree,
183    copy: NodeId,
184    handed_over: &mut HashMap<NodeId, NodeId>,
185) {
186    let mut pairs = vec![(source_node, copy)];
187    while let Some((from, to)) = pairs.pop() {
188        handed_over.insert(from, to);
189        let from_children = source.children(from);
190        let to_children = dom.children(to);
191        for (from_child, to_child) in from_children.into_iter().zip(to_children) {
192            pairs.push((from_child, to_child));
193        }
194    }
195}
