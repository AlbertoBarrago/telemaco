25impl ImportMap {
26    pub(crate) fn parse(input: &str, base_url: &str) -> Result<Self, String> {
27        let base = ModuleSpecifier::parse(base_url)
28            .map_err(|e| format!("Invalid import map base URL {}: {}", base_url, e))?;
29        let parsed: serde_json::Value =
30            serde_json::from_str(input).map_err(|e| format!("Invalid import map JSON: {}", e))?;
31        let object = parsed
32            .as_object()
33            .ok_or_else(|| "Import map top level must be an object".to_string())?;
34
35        let imports = match object.get("imports") {
36            Some(value) => SpecifierMap::parse(
37                value
38                    .as_object()
39                    .ok_or_else(|| "Import map \"imports\" must be an object".to_string())?,
40                &base,
41            ),
42            None => SpecifierMap::default(),
43        };
44
45        let mut scopes = Vec::new();
46        if let Some(value) = object.get("scopes") {
47            let scope_object = value
48                .as_object()
49                .ok_or_else(|| "Import map \"scopes\" must be an object".to_string())?;
50            for (scope_prefix, scope_imports) in scope_object {
51                // Unlike keys and addresses in a module specifier map, scope
52                // prefixes use ordinary URL parsing. In particular, a bare
53                // relative value such as "feature/" is valid here.
54                let Ok(normalized_scope) = base.join(scope_prefix) else {
55                    continue;
56                };
57                let scope_imports = scope_imports.as_object().ok_or_else(|| {
58                    format!(
59                        "Import map scope \"{}\" must contain an object",
60                        scope_prefix
61                    )
62                })?;
63                scopes.push((
64                    normalized_scope.to_string(),
65                    SpecifierMap::parse(scope_imports, &base),
66                ));
67            }
68            // The most specific applicable scope is consulted first.
69            scopes.sort_by(|left, right| right.0.cmp(&left.0));
70        }
71
72        // Integrity enforcement is owned by the module fetch layer. Still
73        // validate the top-level shape here: a malformed integrity member
74        // invalidates the complete import map in Chromium/the HTML algorithm,
75        // rather than leaving its `imports` member active.
76        if let Some(value) = object.get("integrity") {
77            value
78                .as_object()
79                .ok_or_else(|| "Import map \"integrity\" must be an object".to_string())?;
80        }
81
82        Ok(Self {
83            imports,
84            scopes,
85            resolved_modules: Vec::new(),
86        })
87    }
88
89    pub(crate) fn merge(&mut self, mut new_map: Self) {
90        // Multiple import maps are merged even after module graphs have
91        // started. New rules which could change an already-observed
92        // (referrer, specifier) resolution are removed first; unrelated rules
93        // remain available to later graphs.
94        for record in &self.resolved_modules {
95            new_map
96                .imports
97                .remove_rules_affecting(&record.specifier, record.as_url_is_special);
98            for (scope_prefix, scope_imports) in &mut new_map.scopes {
99                if scope_applies(scope_prefix, &record.referrer) {
100                    scope_imports
101                        .remove_rules_affecting(&record.specifier, record.as_url_is_special);
102                }
103            }
104        }
105
106        self.imports.merge(new_map.imports);
107        for (scope_prefix, new_imports) in new_map.scopes {
108            if let Some((_, imports)) = self
109                .scopes
110                .iter_mut()
111                .find(|(existing, _)| existing == &scope_prefix)
112            {
113                imports.merge(new_imports);
114            } else {
115                self.scopes.push((scope_prefix, new_imports));
116            }
117        }
118        self.scopes.sort_by(|left, right| right.0.cmp(&left.0));
119    }

121    pub(crate) fn resolve(
122        &mut self,
123        specifier: &str,
124        referrer: &ModuleSpecifier,
125    ) -> Result<ModuleSpecifier, String> {
126        let as_url = resolve_url_like(specifier, referrer);
127        let normalized = as_url
128            .as_ref()
129            .map(ToString::to_string)
130            .unwrap_or_else(|| specifier.to_string());
131        let serialized_referrer = referrer.to_string();
132
133        for (scope_prefix, scope_imports) in &self.scopes {
134            if scope_applies(scope_prefix, &serialized_referrer) {
135                if let Some(resolved) = scope_imports.resolve_match(&normalized, as_url.as_ref())? {
136                    self.remember_resolution(serialized_referrer, normalized, as_url.as_ref());
137                    return Ok(resolved);
138                }
139            }
140        }
141
142        if let Some(resolved) = self.imports.resolve_match(&normalized, as_url.as_ref())? {
143            self.remember_resolution(serialized_referrer, normalized, as_url.as_ref());
144            return Ok(resolved);
145        }
146        match as_url {
147            Some(resolved) => {
148                self.remember_resolution(serialized_referrer, normalized, Some(&resolved));
149                Ok(resolved)
150            }
151            None => Err(format!(
152                "Bare module specifier \"{}\" was not remapped by the import map",
153                specifier
154            )),
155        }
156    }
157
158    fn remember_resolution(
159        &mut self,
160        referrer: String,
161        specifier: String,
162        as_url: Option<&ModuleSpecifier>,
163    ) {
164        let resolution = ResolvedModule {
165            referrer,
166            specifier,
167            as_url_is_special: as_url.is_none_or(is_special_url),
168        };
169        if !self.resolved_modules.contains(&resolution) {
170            self.resolved_modules.push(resolution);
171        }
172    }
173}

175impl SpecifierMap {
176    fn parse(object: &serde_json::Map<String, serde_json::Value>, base: &ModuleSpecifier) -> Self {
177        let mut entries = HashMap::with_capacity(object.len());
178        let mut prefixes = Vec::new();
179        for (key, value) in object {
180            let Some(normalized_key) = normalize_key(key, base) else {
181                continue;
182            };
183            let address = value
184                .as_str()
185                .and_then(|address| resolve_url_like(address, base))
186                .filter(|address| !key.ends_with('/') || address.as_str().ends_with('/'));
187            if normalized_key.ends_with('/') {
188                prefixes.push(normalized_key.clone());
189            }
190            entries.insert(normalized_key, address);
191        }
192        prefixes.sort_by(|left, right| right.cmp(left));
193        Self { entries, prefixes }
194    }
195
196    fn merge(&mut self, new_map: Self) {
197        for (key, address) in new_map.entries {
198            if !self.entries.contains_key(&key) {
199                if key.ends_with('/') {
200                    self.prefixes.push(key.clone());
201                }
202                self.entries.insert(key, address);
203            }
204        }
205        self.prefixes.sort_by(|left, right| right.cmp(left));
206    }
207
208    fn remove_rules_affecting(&mut self, resolved: &str, as_url_is_special: bool) {
209        self.entries.retain(|key, _| {
210            key != resolved
211                && !(as_url_is_special && key.ends_with('/') && resolved.starts_with(key.as_str()))
212        });
213        self.prefixes
214            .retain(|key| self.entries.contains_key(key.as_str()));
215    }
216
217    fn resolve_match(
218        &self,
219        normalized: &str,
220        as_url: Option<&ModuleSpecifier>,
221    ) -> Result<Option<ModuleSpecifier>, String> {
222        if let Some(address) = self.entries.get(normalized) {
223            return address.clone().map(Some).ok_or_else(|| {
224                format!(
225                    "Module specifier \"{}\" is blocked by import map entry \"{}\"",
226                    normalized, normalized
227                )
228            });
229        }
230
231        if as_url.is_some_and(|url| !is_special_url(url)) {
232            return Ok(None);
233        }
234
235        for key in self
236            .prefixes
237            .iter()
238            .filter(|key| normalized.starts_with(key.as_str()))
239        {
240            let address = self
241                .entries
242                .get(key)
243                .ok_or_else(|| format!("Import map prefix \"{}\" has no matching address", key))?;
244            let address = address.as_ref().ok_or_else(|| {
245                format!(
246                    "Module specifier \"{}\" is blocked by import map prefix \"{}\"",
247                    normalized, key
248                )
249            })?;
250            let after_prefix = &normalized[key.len()..];
251            let resolved = address.join(after_prefix).map_err(|e| {
252                format!(
253                    "Module specifier \"{}\" could not resolve through import map prefix \"{}\": {}",
254                    normalized, key, e
255                )
256            })?;
257            if !resolved.as_str().starts_with(address.as_str()) {
258                return Err(format!(
259                    "Module specifier \"{}\" backtracks above import map prefix \"{}\"",
260                    normalized, key
261                ));
262            }
263            return Ok(Some(resolved));
264        }
265        Ok(None)
266    }
267}
268
269fn normalize_key(key: &str, base: &ModuleSpecifier) -> Option<String> {
270    if key.is_empty() {
271        return None;
272    }
273    Some(
274        resolve_url_like(key, base)
275            .map(|url| url.to_string())
276            .unwrap_or_else(|| key.to_string()),
277    )
278}
279
280fn resolve_url_like(specifier: &str, base: &ModuleSpecifier) -> Option<ModuleSpecifier> {
281    if specifier.starts_with('/') || specifier.starts_with("./") || specifier.starts_with("../") {
282        base.join(specifier).ok()
283    } else {
284        ModuleSpecifier::parse(specifier).ok()
285    }
286}
287
288fn scope_applies(scope_prefix: &str, referrer: &str) -> bool {
289    scope_prefix == referrer || (scope_prefix.ends_with('/') && referrer.starts_with(scope_prefix))
290}
291
292fn is_special_url(url: &ModuleSpecifier) -> bool {
293    matches!(
294        url.scheme(),
295        "ftp" | "file" | "http" | "https" | "ws" | "wss"
296    )
297}

304    fn exact_prefix_and_url_like_keys_are_normalized_against_the_map_base() {
305        let mut map = ImportMap::parse(
306            r#"{
307                "imports": {
308                    "pkg": "../vendor/pkg.js",
309                    "pkg/": "../vendor/pkg/",

312            }"#,
313            "https://example.test/app/maps/import-map.json",

321            "https://example.test/app/vendor/pkg.js",
322        );
