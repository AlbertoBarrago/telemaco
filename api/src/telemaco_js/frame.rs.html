1//! Child frame realms.
2//!
3//! An iframe is a separate browsing context: its own JavaScript realm, its own
4//! DOM tree, and its own origin. Without this, a frame's HTML is fetched and its
5//! body dropped into a detached document in the *parent's* realm, so no script
6//! inside a frame ever runs (issue #600).
7//!
8//! A frame realm is a second `v8::Context` in the page's isolate. Three things
9//! make that practical:
10//!
11//! - The startup snapshot already contains the whole bootstrap, so a restored
12//!   context arrives with every DOM class installed. A realm costs a context
13//!   restore, not a re-parse.
14//! - The realm's op table is filled from the page realm's, so every shim in the
15//!   frame can call ops.
16//! - Each realm registers its document in `RealmStates`, and an op looks up the
17//!   realm that called it. Making a realm current around the host's calls into
18//!   it is not enough, because a frame's timers and settled promises re-enter
19//!   JavaScript straight from the event loop.
20//!
21//! Staying in one isolate is what lets same-origin frames share objects with
22//! their parent, the way `iframe.contentWindow.document` does in a browser. A
23//! second isolate could never do that.
24

32/// One child browsing context: its own realm, document and origin, living in
33/// the page's isolate.
34pub struct FrameRealm {
35    context: deno_core::v8::Global<deno_core::v8::Context>,
36    modules: deno_core::ModuleRealm,
37    /// Held so the frame's entry can be taken out again when the frame dies.
38    realms: Rc<std::cell::RefCell<RealmStates>>,
39    frame_id: u32,
40    parent_frame_id: u32,
41    url: String,
42    origin: String,
43}
44
45impl Drop for FrameRealm {
46    fn drop(&mut self) {
47        self.realms.borrow_mut().forget(&self.context);
48    }
49}

51impl FrameRealm {
52    /// Builds a frame realm around an already-fetched document.
53    ///
54    /// The frame inherits the page's browser identity and its shared resources,
55    /// by copying them from the parent rather than by being told them, so the
56    /// two cannot drift apart.
57    pub fn new(
58        parent: &mut TelemacoJsRuntime,
59        frame_id: u32,
60        parent_frame_id: u32,
61        url: &str,
62        html: &str,
63    ) -> Option<Self> {
64        let context = parent.create_realm_context()?;
65        if !parent.share_ops_with_realm(&context) {
66            return None;
67        }
68
69        // Only a same-origin frame is reachable from the page. Cross-origin
70        // keeps its own security token, so V8 answers `undefined` for any
71        // property the page tries to read out of it, and nothing about it is
72        // published below.
73        let origin = origin_of(url);
74        let same_origin = origin != "null" && origin == parent.page_origin();
75        if same_origin {
76            parent.share_security_token_with_realm(&context);
77        }
78
79        let mut state = TelemacoState::new();
80        state.dom = Some(parse_html(html));
81        state.url = url.to_string();
82        state.frame_id = frame_id;
83        parent.share_resources_with(&mut state);
84
85        let state = Rc::new(std::cell::RefCell::new(state));
86        let modules = parent.attach_frame_module_realm(context.clone(), &state);
87        parent.copy_identity_to_realm(&context);
88        let realms = parent.realm_states();
89        realms.borrow_mut().register(
90            context.clone(),
91            frame_id,
92            state,
93        );
94
95        let realm = FrameRealm {
96            context,
97            modules,
98            realms,
99            frame_id,
100            parent_frame_id,
101            url: url.to_string(),
102            origin,
103        };
104        // Both ids before init, not after: init is what installs `parent` and
105        // `top`, and a document that runs even one script believing it is
106        // top-level has already taken the wrong branch.
107        realm
108            .run(
109                parent,
110                &format!(
111                    "globalThis.__telemaco_frameId = {frame_id};\
112                     globalThis.__telemaco_parentFrameId = {parent_frame_id};\
113                     globalThis.__telemaco_init();"
114                ),
115            )
116            .ok()?;
117        // Only after init, so the document the page reaches through
118        // `contentDocument` is the initialized one.
119        if same_origin {
120            parent.publish_realm_objects(&realm.context, frame_id);
121        }
122        Some(realm)
123    }

125    /// Fires the frame document's lifecycle events, in spec order.
126    ///
127    /// The page's own document gets these; a frame's did not, and a document
128    /// that is never told it finished loading will not run any of the work
129    /// scripts defer until then. That is most of what a widget does: a frame
130    /// can talk to its parent perfectly and still never build its interface,
131    /// which looks like a rendering problem and is a lifecycle one.
132    pub fn dispatch_load_events(&self, parent: &mut TelemacoJsRuntime) -> Result<(), String> {
133        self.execute_script(
134            parent,
135            "globalThis.__documentReadyState__ = 'interactive';\
136             try { document.dispatchEvent(new Event('DOMContentLoaded', \
137                 { bubbles: false, cancelable: false })); } catch (_) {}\
138             try { window.dispatchEvent(new Event('DOMContentLoaded', \
139                 { bubbles: false, cancelable: false })); } catch (_) {}\
140             globalThis.__documentReadyState__ = 'complete';\
141             try { document.dispatchEvent(new Event('readystatechange')); } catch (_) {}\
142             if (typeof window.onload === 'function') { try { window.onload(); } catch (_) {} }\
143             try { window.dispatchEvent(new Event('load', \
144                 { bubbles: false, cancelable: false })); } catch (_) {}",
145        )
146    }

246    /// Runs the frame document's classic scripts, in document order.
247    ///
248    /// `load_external` resolves a `src=` script to its source text; returning
249    /// `None` skips it, which is what a failed subresource fetch looks like to
250    /// the page. One script throwing does not stop the ones after it, matching
251    /// how a browser treats separate classic scripts.
252    ///
253    /// Modules are deferred until all parser-discovered classic scripts ran.
254    /// `run_document_modules` executes their graphs before document load events.
255    ///
256    /// Returns one message per script that failed or was skipped.
257    pub fn run_document_scripts(
258        &self,
259        parent: &mut TelemacoJsRuntime,
260        load_external: impl Fn(&str) -> Option<String>,
261    ) -> Vec<String> {
262        let scripts = match self.list_scripts(parent) {
263            Ok(scripts) => scripts,
264            Err(error) => return vec![error],
265        };
266
267        let mut problems = Vec::new();
268        for (index, script) in scripts.iter().enumerate() {
269            if !script.is_classic() {
270                continue;
271            }
272
273            let (name, source) = if script.src.is_empty() {
274                (format!("inline {index}"), script.text.clone())
275            } else {
276                let resolved = self.resolve(&script.src);
277                match load_external(&resolved) {
278                    Some(source) => (resolved, source),
279                    None => {
280                        problems.push(format!("frame script {resolved} could not be loaded"));
281                        continue;
282                    }
283                }
284            };
285            if source.trim().is_empty() {
286                continue;
287            }
288            if let Err(error) = self.execute_script(parent, &source) {
289                problems.push(format!("frame script {name} failed: {error}"));
290            }
291        }
292        problems
293    }

295    /// Executes deferred module roots in the frame's own cache and context.
296    /// One deadline covers all roots, and lifecycle events follow completion.
297    pub async fn run_document_modules(
298        &self,
299        parent: &mut TelemacoJsRuntime,
300        budget_ms: u64,
301    ) -> Vec<String> {
302        let scripts = match self.list_scripts(parent) {
303            Ok(scripts) => scripts,
304            Err(error) => return vec![error],
305        };
306        let deadline = tokio::time::Instant::now() + std::time::Duration::from_millis(budget_ms);
307        let mut problems = Vec::new();
308        for script in scripts.into_iter().filter(|script| script.type_attribute == "module") {
309            let Some(remaining) = deadline.checked_duration_since(tokio::time::Instant::now()) else {
310                problems.push("Frame module budget exhausted".to_string());
311                break;
312            };
313            let (url, code) = if script.src.is_empty() {
314                (self.url.clone(), Some(script.text))
315            } else {
316                (self.resolve(&script.src), None)
317            };
318            if let Err(error) = parent.load_frame_module(
319                &self.modules, &url, code, remaining.as_millis().max(1) as u64,
320            ).await {
321                problems.push(format!("frame module {url} failed: {error}"));
322            }
323        }
324        problems
325    }

327    /// Absolute URLs of the frame's `src=` classic scripts, in document order.
328    ///
329    /// A caller that fetches over the network needs the list before running
330    /// anything, because `run_document_scripts` resolves sources synchronously.
331    pub fn external_script_urls(&self, parent: &mut TelemacoJsRuntime) -> Vec<String> {
332        self.list_scripts(parent)
333            .unwrap_or_default()
334            .iter()
335            .filter(|script| script.is_classic() && !script.src.is_empty())
336            .map(|script| self.resolve(&script.src))
337            .collect()
338    }
339
340    /// Resolves a subresource URL against the frame's own document URL, not the
341    /// parent's. A relative `src` in a frame is relative to the frame.
342    fn resolve(&self, src: &str) -> String {
343        url::Url::parse(&self.url)
344            .and_then(|base| base.join(src))
345            .map(|url| url.to_string())
346            .unwrap_or_else(|_| src.to_string())
347    }
348
349    fn list_scripts(&self, parent: &mut TelemacoJsRuntime) -> Result<Vec<DocumentScript>, String> {
350        let listed = self.evaluate(
351            parent,
352            r#"[...document.querySelectorAll('script')].map(node => ({
353                src: node.getAttribute('src') || '',
354                type: (node.getAttribute('type') || '').toLowerCase(),
355                text: node.textContent || '',
356            }))"#,
357        );
358        match listed {
359            Ok(value) => Ok(serde_json::from_value(value).unwrap_or_default()),
360            Err(error) => Err(format!("could not list frame scripts: {error}")),
361        }
362    }

365#[derive(serde::Deserialize)]
366struct DocumentScript {
367    src: String,
368    #[serde(rename = "type")]
369    type_attribute: String,
370    text: String,
371}
372
373impl DocumentScript {
374    /// An empty type, or a JavaScript MIME type, is a classic script. Anything
375    /// else is data or a module.
376    fn is_classic(&self) -> bool {
377        self.type_attribute.is_empty()
378            || matches!(
379                self.type_attribute.as_str(),
380                "text/javascript" | "application/javascript" | "text/ecmascript"
381            )
382    }
383}
384
385/// Embeds a string in JavaScript source as a literal, so a payload holding
386/// quotes or newlines cannot end the literal and be read as code.
387fn encode_json_argument(value: &str) -> String {
388    serde_json::to_string(value).unwrap_or_else(|_| "\"\"".to_string())
389}
390
391/// Serializes an origin the way `location.origin` does, using `"null"` for
392/// schemes that have no tuple origin.
393fn origin_of(url: &str) -> String {
394    match url::Url::parse(url) {
395        Ok(parsed) => {
396            let origin = parsed.origin();
397            if origin.is_tuple() {
398                origin.ascii_serialization()
399            } else {
400                "null".to_string()
401            }
402        }
403        Err(_) => "null".to_string(),
404    }
405}

1215        // unsafe: the page still names its window, so V8 keeps the context
1216        // alive and the read still answers. This is exactly why a discarded
1217        // frame has to have its entry removed rather than merely dropped, and
1218        // what `Page::release_detached_frames` is for.
1219        assert_eq!(
1220            parent
1221                .evaluate("globalThis.__telemaco_frameObjects[1].window.marker")
1222                .unwrap(),
1223            serde_json::json!("child"),
1224        );
1225        // The page's own DOM work still resolves against the page.
1226        assert_eq!(
1227            parent.evaluate("document.body.innerHTML").unwrap(),
1228            serde_json::json!(""),
1229        );
1230        // Dropping the page's reference is what lets the frame be collected.
1231        parent
1232            .execute_script("p", "delete globalThis.__telemaco_frameObjects[1];")
1233            .unwrap();
1234        assert_eq!(
1235            parent
1236                .evaluate("globalThis.__telemaco_frameObjects[1] === undefined")
1237                .unwrap(),
1238            serde_json::json!(true),
1239        );
1240    }
