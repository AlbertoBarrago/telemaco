1//! The agent: serves the remote protocol on one byte stream.
2//!
3//! Under Tailcat, `tailcat serve -- telemaco remote agent` starts one agent
4//! process per connection with the connection as stdio, so an agent handles
5//! exactly one session and exits. Stdout is the protocol channel: nothing
6//! else may ever be written to it. Diagnostics go to stderr, which tailcat
7//! forwards to the terminal running `remote serve`.
8

19/// Output is forwarded in chunks of at most this many bytes (about 43 KiB
20/// once base64 encoded, far below the frame cap).
21const OUTPUT_CHUNK: usize = 32 * 1024;
22/// After the program exits, how long to keep collecting output still in the
23/// pipes. Bounded because a background grandchild can hold them open forever.
24const DRAIN_AFTER_EXIT: Duration = Duration::from_secs(1);
25
26/// What this agent is and what it permits. Fixed for the agent's lifetime:
27/// nothing a client sends can change it.
28#[derive(Debug, Clone)]
29pub struct AgentConfig {
30    pub telemaco_version: String,
31    /// Whether `exec` requests are honored. Off unless the operator started
32    /// `remote serve --allow-exec`.
33    pub allow_exec: bool,
34    /// When set, one audit line per exec (program name only, never the
35    /// arguments, which may carry secrets) goes to stderr, tagged with this
36    /// peer label. Stderr is the operator's log under `tailcat serve`.
37    pub audit_peer: Option<String>,
38    /// Ports `remote serve` told tailcat to forward, reported in status so a
39    /// client can check before forwarding. Informational: tailcat enforces
40    /// what is actually reachable.
41    pub forwarded_ports: Vec<u16>,
42}

44/// One frame (or the end of the stream) as read by the session's reader.
45type Incoming = Result<Option<Vec<u8>>, ProtocolError>;
46
47/// Serves one session to completion. Returns `Ok(())` when the client
48/// disconnects cleanly after the handshake.
49///
50/// A single reader future pulls frames off the stream for the whole session
51/// and hands them over a channel. Reading a frame directly inside a
52/// `select!` would not be cancel-safe: losing the race to a chunk of exec
53/// output mid-frame would desynchronize the stream. `mpsc::Receiver::recv`
54/// is cancel-safe, so the session can wait on it alongside a running exec.
55pub async fn serve<S>(stream: &mut S, config: &AgentConfig) -> Result<(), ProtocolError>
56where
57    S: AsyncRead + AsyncWrite + Unpin,
58{
59    let (mut rd, mut wr) = tokio::io::split(stream);
60    let (tx, mut rx) = mpsc::channel::<Incoming>(1);
61    let reader = async move {
62        loop {
63            let frame = protocol::read_frame(&mut rd).await;
64            let last = !matches!(frame, Ok(Some(_)));
65            if tx.send(frame).await.is_err() || last {
66                break;
67            }
68        }
69    };
70    let session = async {
71        handshake(&mut rx, &mut wr, config).await?;
72        session_loop(&mut rx, &mut wr, config).await
73    };
74    tokio::pin!(session);
75    // If the reader finishes first it has already queued EOF or an error;
76    // the session still has to consume it, so keep awaiting the session.
77    tokio::select! {
78        result = &mut session => result,
79        () = reader => session.await,
80    }
81}

83type Writer<'a, S> = WriteHalf<&'a mut S>;

85async fn next_frame(rx: &mut mpsc::Receiver<Incoming>) -> Incoming {
86    rx.recv().await.unwrap_or(Ok(None))
87}
88
89async fn session_loop<S>(
90    rx: &mut mpsc::Receiver<Incoming>,
91    wr: &mut Writer<'_, S>,
92    config: &AgentConfig,
93) -> Result<(), ProtocolError>
94where
95    S: AsyncRead + AsyncWrite + Unpin,
96{
97    loop {
98        let body = match next_frame(rx).await {
99            Ok(Some(body)) => body,
100            Ok(None) => return Ok(()),
101            // Framing is lost (or the peer is hostile): report and hang up.
102            Err(e @ (ProtocolError::FrameTooLarge(_) | ProtocolError::EmptyFrame)) => {
103                let _ = reply_error(wr, ErrorCode::BadRequest, e.to_string()).await;
104                return Err(e);
105            }
106            Err(e) => return Err(e),
107        };
108        let request = match protocol::decode::<Request>(&body) {
109            Ok(r) => r,
110            // A well-framed but invalid message: say so and keep serving.
111            Err(e) => {
112                reply_error(wr, ErrorCode::BadRequest, e.to_string()).await?;
113                continue;
114            }
115        };
116        let response = match request {
117            Request::Hello { .. } => Response::Error(RemoteError::new(
118                ErrorCode::BadRequest,
119                "handshake already done",
120            )),
121            Request::Ping { nonce } => Response::Pong { nonce },
122            Request::Status {} => Response::Status(status(config).await),
123            Request::Cancel {} => Response::Error(RemoteError::new(
124                ErrorCode::BadRequest,
125                "no exec is running",
126            )),
127            Request::Exec(req) => match exec(req, rx, wr, config).await? {
128                ExecFlow::Continue => continue,
129                ExecFlow::ClientGone => return Ok(()),
130            },
131        };
132        protocol::send(wr, &response).await?;
133    }
134}

136async fn handshake<S>(
137    rx: &mut mpsc::Receiver<Incoming>,
138    wr: &mut Writer<'_, S>,
139    config: &AgentConfig,
140) -> Result<(), ProtocolError>
141where
142    S: AsyncRead + AsyncWrite + Unpin,
143{
144    let first = next_frame(rx).await?.ok_or(ProtocolError::Closed)?;
145    match protocol::decode::<Request>(&first) {
146        Ok(Request::Hello { version }) if version == PROTOCOL_VERSION => {
147            let hello = Response::Hello {
148                version: PROTOCOL_VERSION,
149                telemaco_version: config.telemaco_version.clone(),
150            };
151            protocol::send(wr, &hello).await
152        }
153        Ok(Request::Hello { version }) => {
154            let msg = format!("agent speaks protocol v{PROTOCOL_VERSION}, client sent v{version}");
155            reply_error(wr, ErrorCode::UnsupportedVersion, msg).await?;
156            Err(ProtocolError::VersionMismatch {
157                ours: PROTOCOL_VERSION,
158                theirs: version,
159            })
160        }
161        Ok(_) => {
162            reply_error(wr, ErrorCode::BadRequest, "expected hello").await?;
163            Err(ProtocolError::Unexpected("request before hello".into()))
164        }
165        Err(e) => {
166            reply_error(wr, ErrorCode::BadRequest, e.to_string()).await?;
167            Err(e)
168        }
169    }
170}

172async fn reply_error<W>(
173    wr: &mut W,
174    code: ErrorCode,
175    message: impl Into<String>,
176) -> Result<(), ProtocolError>
177where
178    W: AsyncWrite + Unpin,
179{
180    protocol::send(wr, &Response::Error(RemoteError::new(code, message))).await
181}

183enum ExecFlow {
184    /// The exec is settled (ran, or was refused); keep serving the session.
185    Continue,
186    /// The client disconnected mid-exec; the program was killed.
187    ClientGone,
188}

190/// Runs one program with no shell, streaming its output, until it exits, the
191/// deadline passes, or the client cancels or disconnects. The child is
192/// `kill_on_drop`, so every early return (including write errors to a vanished
193/// client) also kills it.
194///
195/// Limitation: only the direct child is killed. Grandchildren it started in
196/// the background keep running, as they would after an `ssh` session ends.
197async fn exec<S>(
198    req: ExecRequest,
199    rx: &mut mpsc::Receiver<Incoming>,
200    wr: &mut Writer<'_, S>,
201    config: &AgentConfig,
202) -> Result<ExecFlow, ProtocolError>
203where
204    S: AsyncRead + AsyncWrite + Unpin,
205{
206    if !config.allow_exec {
207        reply_error(
208            wr,
209            ErrorCode::Forbidden,
210            "exec is disabled on this agent; the operator must start it with \
211             `telemaco remote serve --allow-exec`",
212        )
213        .await?;
214        return Ok(ExecFlow::Continue);
215    }
216    if let Err(e) = req.validate() {
217        reply_error(wr, ErrorCode::BadRequest, e).await?;
218        return Ok(ExecFlow::Continue);
219    }
220    let program = protocol::sanitize_for_display(&req.program);
221    if let Some(peer) = &config.audit_peer {
222        eprintln!(
223            "telemaco agent: {peer} exec {program} ({} args)",
224            req.args.len()
225        );
226    }
227
228    let mut cmd = tokio::process::Command::new(&req.program);
229    cmd.args(&req.args)
230        .envs(req.env.iter().map(|v| (&v.name, &v.value)))
231        .stdin(std::process::Stdio::null())
232        .stdout(std::process::Stdio::piped())
233        .stderr(std::process::Stdio::piped())
234        .kill_on_drop(true);
235    if let Some(cwd) = &req.cwd {
236        cmd.current_dir(cwd);
237    }
238    let mut child = match cmd.spawn() {
239        Ok(c) => c,
240        Err(e) => {
241            reply_error(
242                wr,
243                ErrorCode::ExecFailed,
244                format!("failed to start {program}: {e}"),
245            )
246            .await?;
247            return Ok(ExecFlow::Continue);
248        }
249    };
250    let (Some(mut out), Some(mut err)) = (child.stdout.take(), child.stderr.take()) else {
251        reply_error(wr, ErrorCode::Internal, "child stdio was not piped").await?;
252        return Ok(ExecFlow::Continue);
253    };
254
255    let deadline = async {
256        match req.timeout_secs {
257            Some(s) => tokio::time::sleep(Duration::from_secs(s)).await,
258            None => std::future::pending().await,
259        }
260    };
261    tokio::pin!(deadline);
262    let mut out_buf = vec![0u8; OUTPUT_CHUNK];
263    let mut err_buf = vec![0u8; OUTPUT_CHUNK];
264    let (mut out_open, mut err_open) = (true, true);
265
266    // Every branch here is cancel-safe: pipe reads, `recv`, `wait`, sleep.
267    let (status, end) = loop {
268        tokio::select! {
269            r = out.read(&mut out_buf), if out_open => match r {
270                Ok(0) | Err(_) => out_open = false,
271                Ok(n) => protocol::send(wr, &Response::output(OutputStream::Stdout, &out_buf[..n])).await?,
272            },
273            r = err.read(&mut err_buf), if err_open => match r {
274                Ok(0) | Err(_) => err_open = false,
275                Ok(n) => protocol::send(wr, &Response::output(OutputStream::Stderr, &err_buf[..n])).await?,
276            },
277            status = child.wait() => break (status?, ExecEnd::Exited),
278            () = &mut deadline => {
279                child.kill().await?;
280                break (child.wait().await?, ExecEnd::TimedOut);
281            }
282            frame = next_frame(rx) => {
283                child.kill().await?;
284                let status = child.wait().await?;
285                match frame.map(|f| f.map(|b| protocol::decode::<Request>(&b))) {
286                    Ok(Some(Ok(Request::Cancel {}))) => break (status, ExecEnd::Cancelled),
287                    // Anything else mid-exec is a protocol violation: stop
288                    // the program, say why, and end the exec as cancelled.
289                    Ok(Some(_)) => {
290                        reply_error(wr, ErrorCode::BadRequest, "only cancel is valid while an exec runs").await?;
291                        break (status, ExecEnd::Cancelled);
292                    }
293                    Ok(None) | Err(_) => return Ok(ExecFlow::ClientGone),
294                }
295            }
296        }
297    };
298
299    if end == ExecEnd::Exited {
300        // Forward what is still buffered in the pipes, briefly.
301        let drain = async {
302            while out_open || err_open {
303                tokio::select! {
304                    r = out.read(&mut out_buf), if out_open => match r {
305                        Ok(0) | Err(_) => out_open = false,
306                        Ok(n) => protocol::send(wr, &Response::output(OutputStream::Stdout, &out_buf[..n])).await?,
307                    },
308                    r = err.read(&mut err_buf), if err_open => match r {
309                        Ok(0) | Err(_) => err_open = false,
310                        Ok(n) => protocol::send(wr, &Response::output(OutputStream::Stderr, &err_buf[..n])).await?,
311                    },
312                }
313            }
314            Ok::<_, ProtocolError>(())
315        };
316        if let Ok(result) = tokio::time::timeout(DRAIN_AFTER_EXIT, drain).await {
317            result?;
318        }
319    }
320
321    let exit = ExecExit {
322        code: status.code(),
323        signal: exit_signal(&status),
324        end,
325    };
326    protocol::send(wr, &Response::Exited(exit)).await?;
327    Ok(ExecFlow::Continue)
328}

352/// The machine's host name via the `hostname` utility (present on Linux,
353/// macOS and Windows), which avoids an FFI call and a new dependency.
354/// Any failure just means "unknown".
355async fn hostname() -> Option<String> {
356    let out = tokio::time::timeout(
357        std::time::Duration::from_secs(2),
358        tokio::process::Command::new("hostname")
359            .stdin(std::process::Stdio::null())
360            .stderr(std::process::Stdio::null())
361            .kill_on_drop(true)
362            .output(),
363    )
364    .await
365    .ok()?
366    .ok()?;
367    if !out.status.success() {
368        return None;
369    }
370    let name = String::from_utf8_lossy(&out.stdout).trim().to_string();
371    (!name.is_empty() && name.len() <= 255).then_some(name)
372}

374/// Short, non-secret label for the peer from Tailcat's `$TAILCAT_PEER_KEY`
375/// (a public node key), for the operator's log. Truncated to keep log lines
376/// readable; the full key is not needed to recognize a peer.
377pub fn peer_label(raw: Option<&str>) -> String {
378    match raw {
379        Some(key) if !key.is_empty() => {
380            let visible: String = key.chars().take(16).collect();
381            format!("{}...", protocol::sanitize_for_display(&visible))
382        }
383        _ => "unknown peer".to_string(),
384    }
385}

687        #[tokio::test]
688        async fn client_disconnect_kills_the_program() {
689            let (mut c, agent) = agent(true);
690            let pidfile = std::env::temp_dir()
691                .join(format!("telemaco-remote-exec-{}.pid", std::process::id()));
692            {
693                let mut client = RemoteClient::handshake(&mut c).await.unwrap();
694                let script = format!("echo $$ > '{}'; exec sleep 30", pidfile.display());
695                let run = client.exec(
696                    req("sh", &["-c", &script]),
697                    std::future::pending(),
698                    Duration::ZERO,
699                    |_, _| {},
700                );
701                // Give the program time to start, then walk away mid-exec.
702                let _ = tokio::time::timeout(Duration::from_millis(500), run).await;
703            }
704            drop(c);
705            // The agent must notice the disconnect promptly, not when the
706            // program would have finished on its own.
707            tokio::time::timeout(Duration::from_secs(5), agent)
708                .await
709                .expect("agent did not react to the client disconnecting")
710                .unwrap()
711                .unwrap();
712            let pid = std::fs::read_to_string(&pidfile)
713                .unwrap()
714                .trim()
715                .to_string();
716            let alive = std::process::Command::new("kill")
717                .args(["-0", &pid])
718                .stderr(std::process::Stdio::null())
719                .status()
720                .unwrap()
721                .success();
722            assert!(!alive, "program {pid} outlived its client");
723        }
