1//! Port forwarding over Tailcat: `tailcat forward --bind=<ip> <addr> L:R`.
2//!
3//! Tailcat owns the listener and the tunnel; Telemaco only starts it, reads
4//! the `# forwarding <local> -> remote <target>` line it prints once the
5//! listener is up, and tears it down. The server side exposes a port only if
6//! `remote serve --forward-port` named it.
7
8use std::collections::VecDeque;
9use std::net::{IpAddr, SocketAddr};
10use std::process::Stdio;
11use std::sync::{Arc, Mutex};
12use std::time::Duration;

14use tokio::process::Child;
15use tokio::sync::mpsc;

19use crate::address::TailcatAddress;

21/// What to forward: a local listener to a port on the remote's localhost.
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub struct ForwardRequest {
24    pub remote_port: u16,
25    /// `0` asks the OS for a free port.
26    pub local_port: u16,
27    /// Listen address. Loopback unless the user explicitly chose otherwise.
28    pub bind: IpAddr,
29}
30
31/// A running forwarder. Dropping it kills the forwarding process.
32#[derive(Debug)]
33pub struct Forwarder {
34    child: Child,
35    local: SocketAddr,
36    spec: ProcessSpec,
37    stderr_tail: Arc<Mutex<VecDeque<String>>>,
38}
39
40impl Forwarder {
41    pub(crate) fn spec(
42        cli: &TailcatCli,
43        addr: &TailcatAddress,
44        req: &ForwardRequest,
45    ) -> ProcessSpec {
46        cli.spec()
47            .arg("forward")
48            .arg(format!("--bind={}", req.bind))
49            .secret(addr.clone())
50            .arg(format!("{}:{}", req.local_port, req.remote_port))
51    }
52
53    /// Starts tailcat and waits for it to report the local listener.
54    pub(crate) async fn start(
55        spec: ProcessSpec,
56        startup: Duration,
57    ) -> Result<(Self, mpsc::Receiver<String>), TransportError> {
58        let mut child = spec
59            .command()
60            .stdin(Stdio::null())
61            .stdout(Stdio::null())
62            .stderr(Stdio::piped())
63            .spawn()
64            .map_err(|e| spec.spawn_error(e))?;
65        let Some(stderr) = child.stderr.take() else {
66            return Err(spec.spawn_error(std::io::Error::other("child stderr was not piped")));
67        };
68        let stderr_tail = Arc::new(Mutex::new(VecDeque::new()));
69        let (tx, mut raw) = mpsc::channel(256);
70        tokio::spawn(relay_stderr(stderr, tx, Arc::clone(&stderr_tail)));
71
72        // Watch the relayed lines for the listener announcement, then hand
73        // the remaining lines to the caller through a fresh channel.
74        let wait_for_listener = async {
75            while let Some(line) = raw.recv().await {
76                if let Some(local) = parse_listening(&line) {
77                    return Some(local);
78                }
79            }
80            None
81        };
82        match tokio::time::timeout(startup, wait_for_listener).await {
83            Ok(Some(local)) => {
84                let (tx, rx) = mpsc::channel(256);
85                tokio::spawn(async move {
86                    while let Some(line) = raw.recv().await {
87                        if tx.send(line).await.is_err() {
88                            break;
89                        }
90                    }
91                });
92                Ok((
93                    Self {
94                        child,
95                        local,
96                        spec,
97                        stderr_tail,
98                    },
99                    rx,
100                ))
101            }
102            Ok(None) => Err(exit_report(&mut child, &spec, &stderr_tail).await),
103            Err(_) => {
104                let _ = child.kill().await;
105                Err(TransportError::Timeout {
106                    program: spec.display_name(),
107                    seconds: startup.as_secs(),
108                })
109            }
110        }
111    }
112
113    /// The address local clients connect to (with the port the OS picked,
114    /// if the request asked for port 0).
115    pub fn local_addr(&self) -> SocketAddr {
116        self.local
117    }
118
119    /// Resolves when the forwarder exits, with its status and recent stderr.
120    pub async fn wait(&mut self) -> TransportError {
121        exit_report(&mut self.child, &self.spec, &self.stderr_tail).await
122    }
123
124    /// Stops forwarding and reaps the process.
125    pub async fn shutdown(mut self) -> Result<(), TransportError> {
126        match self.child.kill().await {
127            Ok(()) => Ok(()),
128            Err(e) if e.kind() == std::io::ErrorKind::InvalidInput => Ok(()),
129            Err(e) => Err(e.into()),
130        }
131    }
132}
133
134/// Parses `# forwarding 127.0.0.1:54321 -> remote localhost:5432`.
135fn parse_listening(line: &str) -> Option<SocketAddr> {
136    let rest = line.strip_prefix("# forwarding ")?;
137    let (local, _) = rest.split_once(" -> ")?;
138    local.parse().ok()
139}

145    #[test]
146    fn listener_line_parsing() {
147        assert_eq!(
148            parse_listening("# forwarding 127.0.0.1:54321 -> remote localhost:5432"),
149            Some("127.0.0.1:54321".parse().unwrap())
150        );
151        assert_eq!(
152            parse_listening("# forwarding [::1]:8080 -> remote localhost:8080"),
153            Some("[::1]:8080".parse().unwrap())
154        );
155        assert_eq!(
156            parse_listening("# Selected bootstrap relay region 303"),
157            None
158        );
159        assert_eq!(parse_listening("# forwarding garbage -> remote x"), None);
160    }

177        #[test]
178        fn argv_is_bind_then_address_then_mapping() {
179            let cli = TailcatCli::locate(Some(&fake_bin("tailcat", "exit 0"))).unwrap();
180            let spec = Forwarder::spec(&cli, &TailcatAddress::parse(SAMPLE).unwrap(), &req());
181            let argv: Vec<String> = spec
182                .args
183                .iter()
184                .map(|a| match a {
185                    Arg::Plain(s) => s.to_string_lossy().into_owned(),
186                    Arg::Secret(_) => "<addr>".into(),
187                })
188                .collect();
189            assert_eq!(argv, ["forward", "--bind=127.0.0.1", "<addr>", "0:5432"]);
190        }

192        #[tokio::test]
193        async fn start_reports_the_listener_and_stops_cleanly() {
194            let body = r##"[ "$1" = version ] && { echo v0.7.0; exit 0; }
195echo "# forwarding 127.0.0.1:54321 -> remote localhost:5432" >&2
196exec sleep 30"##;
197            let cli = TailcatCli::locate(Some(&fake_bin("tailcat", body))).unwrap();
198            let t = TailcatTransport::new(cli, TailcatAddress::parse(SAMPLE).unwrap());
199            let (fwd, _lines) = t.forward(req()).await.unwrap();
200            assert_eq!(fwd.local_addr(), "127.0.0.1:54321".parse().unwrap());
201            fwd.shutdown().await.unwrap();
202        }
